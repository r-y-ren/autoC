
# Certified same-turn sale proceeds may fund a species substitution.
_K28S_PLAN_PARENT=_cs_plan
_K28S_REPORT={'funded_plans':0,'budget_declines':0,'mixed_buy_declines':0}
def _cs_plan(obs,action,k,mode):
    if mode!='SHEEP':return _K28S_PLAN_PARENT(obs,action,k,mode)
    farm=obs['farms'][int(obs['player'])];stock=projected_shed(action,FarmView(obs))
    cash=float(farm['money']);hires=int(farm['hires_today']);land=len(farm['unlocked_quadrants'])
    params={key:dict(value) for key,value in _R37_MARKET_PARAMS.items()}
    for item,patch in (obs['market'].get('params') or {}).items():
        if item in params:params[item].update(patch)
    inventory=dict(obs['market']['inventory']);minimum_cash=cash
    for index,order in enumerate(action.get('market',[])):
        if not order:continue
        op=order[0]
        if op=='BUY_PRODUCT':
            _K28S_REPORT['mixed_buy_declines']+=1;return None
        if op=='HIRE':cash-=_v219_fib(hires);hires+=1
        elif op=='BUY_LAND' and land<4:cash-=(1000,2000,4000)[land-1];land+=1
        elif len(order)>=3:
            item=order[1];n=max(0,int(order[2]))
            if op=='BUY_ANIMAL':cash-=n*(500 if item=='COW' else {'GOOSE':300,'COW':400,'SHEEP':500}[item])
            elif op=='BUY_SEED':cash-=n*{'WHEAT':10,'CARROT':20,'TOMATO':50,'STRAWBERRY':100,'MELON':80}[item]
            elif op=='SELL':
                n=min(n,int(stock.get(item,0)))
                for u in range(n):
                    opposing_prefix=u if index==0 else 100
                    price=_r37_market_price(item,inventory[item]+opposing_prefix,params)
                    cash+=price
                    if price>1:inventory[item]+=1
                stock[item]=int(stock.get(item,0))-n
        minimum_cash=min(minimum_cash,cash)
    if minimum_cash<_K28S_RESERVE:
        _K28S_REPORT['budget_declines']+=1;return None
    result=_K28S_PLAN_PARENT(obs,action,k,mode)
    if result is not None:_K28S_REPORT['funded_plans']+=1
    return result
