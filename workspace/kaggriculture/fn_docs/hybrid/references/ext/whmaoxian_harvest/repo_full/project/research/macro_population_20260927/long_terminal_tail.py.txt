
# Kaggriculture final-day physical planning. No opponent private state is used.
_LT_PARENT=phase_b_agent
_LT_PLAN=_PLANNER_NS['plan_terminal']
_LT_PLANS={};_LT_LAST={}
_LT_REPORT={'calls':0,'declined':0,'accepted':0,'changed':0,'aborted':0,'errors':0,'max_ms':0.0}

def _lt_call(name,*args,**kwargs):
    original_start=_PLANNER_NS['START']
    _PLANNER_NS['START']=_LT_START
    try:return _PLANNER_NS[name](*args,**kwargs)
    finally:_PLANNER_NS['START']=original_start

def _lt_forecast(obs,configuration,first):
    seat=int(obs['player']);chassis=_IMPL.chassis
    state=chassis.players.get(seat)
    if not state or state.get('pending') or state.get('route')!=2:return None
    if _V219_STATES.get(seat,{}).get('committed'):return None
    shadow=copy.copy(chassis);shadow.players=copy.deepcopy(chassis.players)
    shadow.diagnostics={k:0 for k in chassis.diagnostics}
    projected=copy.deepcopy(obs);baseline=[];observations=[]
    for step in range(_LT_START,719):
        projected['step']=step;projected['day']=29;projected['hour']=step%24
        observations.append(copy.deepcopy(projected))
        if step==_LT_START:action=copy.deepcopy(first)
        elif step==718:action=_parent_liquidate(projected['farms'][seat],projected['private'],projected['market']['prices'])
        else:action=shadow.act(projected,configuration)
        orders=action.get('market',[])
        if step<718 and (len(orders)!=9 or {o[1] for o in orders if len(o)>1}!=set(PRODUCTS) or any(len(o)!=3 or o[0]!='SELL' or type(o[2]) is not int or o[2]<100 for o in orders)):return None
        if any(shadow.diagnostics.values()):return None
        run=_lt_call('simulate',projected,configuration,[action])
        if run['actions'][0]!=action:return None
        baseline.append(action)
        run['farm']['money']=projected['farms'][seat]['money']
        projected['farms'][seat]=run['farm'];projected['private']=run['private']
    offset=712-_LT_START
    if offset>0:
        old_plan=_LT_PLAN(observations[offset],configuration,baseline[offset:],max_simulations=64,passes=1,proposals_per_actor=4)
        if old_plan.get('accepted'):baseline[offset:]=old_plan['actions']
    return baseline

def long_terminal_agent(observation,configuration=None):
    step=int(observation['step']);seat=int(observation['player'])
    previous=_LT_LAST.get(seat)
    if step==0 or previous is not None and step<=previous:
        _LT_PLANS.pop(seat,None)
        if step==0:
            for key in _LT_REPORT:_LT_REPORT[key]=0
    _LT_LAST[seat]=step
    plan=_LT_PLANS.get(seat)
    if plan and plan.get('accepted') and _LT_START<=step<=718:
        if previous!=step-1:plan.update(abandoned=True,reason='nonconsecutive callback')
        try:
            action=_lt_call('terminal_action',observation,configuration,plan['baseline'][step-_LT_START],plan)
            _LT_REPORT['changed']+=int(action!=plan['baseline'][step-_LT_START])
            if plan.get('abandoned') and not plan.get('counted_abort'):
                _LT_REPORT['aborted']+=1;plan['counted_abort']=True
            return action
        except Exception:
            _LT_REPORT['errors']+=1
            return _lt_call('_recover_observed',observation,configuration,plan['baseline'][step-_LT_START],plan)
    action=_LT_PARENT(observation,configuration)
    if _LT_MODE=='control' or step!=_LT_START:return action
    _LT_REPORT['calls']+=1
    try:
        baseline=_lt_forecast(observation,configuration,action)
        if baseline is None:
            _LT_REPORT['declined']+=1;return action
        plan=_lt_call('plan_terminal',observation,configuration,baseline,max_simulations=_LT_BUDGET,passes=2,proposals_per_actor=_LT_PROPOSALS)
        _LT_REPORT['max_ms']=max(_LT_REPORT['max_ms'],float(plan.get('planning_ms',0)))
        if not plan.get('accepted'):return action
        _LT_PLANS[seat]=plan;_LT_REPORT['accepted']+=1
        result=_lt_call('terminal_action',observation,configuration,action,plan)
        _LT_REPORT['changed']+=int(result!=action)
        return result
    except Exception:
        _LT_REPORT['errors']+=1;return action
_MP_REPORT=_LT_REPORT
long_terminal_agent.telemetry=_LT_REPORT
agent=long_terminal_agent
