    if _MP_GENOME['expansion']=='none':return False

def _hd2_schedule(animal,placed_day,day_from):
    if not _MP_GENOME['initial_care']:
        return _MP_ORIGINAL_SCHEDULE(animal,placed_day,day_from)
    spec=_HD2_SPEC[animal];result={};cap={'GOOSE':4,'COW':6,'SHEEP':6}[animal]
    for day in range(max(day_from,placed_day+spec['first']),30):
        age=day-placed_day
        if (age-spec['first'])%spec['interval']:continue
        # The production boundary consumes past care before adding today's care.
        care_days=spec['first']-1 if age==spec['first'] else spec['interval']
        result[day]=min(float(cap),1.0+care_days*_HD2_CARE)
    return result

def macro_population_agent(observation,configuration=None):
    step=int(observation['step']);seat=int(observation['player'])
    if step==0:
        for key in _MP_REPORT:_MP_REPORT[key]=0
    action=_MP_PARENT(observation,configuration)
    _MP_REPORT['turns']+=1
    cap=int(_MP_GENOME['hire_cap']);farm=observation['farms'][seat]
    if not cap or step<144:return action
    urgent=any(isinstance(t,dict) and t.get('animal') and t.get('consecutive_unfed',0)>0
               for row in farm['tiles'] for t in row)
    if urgent:
        _MP_REPORT['cap_veto_feed_risk']+=1;return action
    orders=[list(o) for o in action.get('market',[])];hires=int(farm['hires_today'])
    changed=False
    for index,order in enumerate(orders):
        if order!=['HIRE']:continue
        if hires>=cap:
            orders[index]=[];changed=True
            _MP_REPORT['hire_requests_removed']+=1
        else:hires+=1
    if changed:action=dict(action,market=orders)
    return action
macro_population_agent.telemetry=_MP_REPORT
