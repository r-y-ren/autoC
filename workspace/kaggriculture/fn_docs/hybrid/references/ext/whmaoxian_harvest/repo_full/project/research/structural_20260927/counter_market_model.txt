
# One-turn market model: quotes are simultaneous, commits follow player order.
# Opponent inventories are explicit scenarios, never hidden live observations.
_MC_SEEDS={'WHEAT':10,'CARROT':20,'TOMATO':50,'STRAWBERRY':100,'MELON':80}
_MC_ANIMALS={'COW':400,'SHEEP':500,'GOOSE':300}
def _mc_run(obs,available,mine,theirs,rival_stock=None,cache=None):
    seat=int(obs['player']);farms=obs['farms'];params=_v44y_params(obs)
    warehouse=[dict(available),dict(rival_stock or {})]
    if rival_stock is None:
        demand={}
        for o in theirs:
            if len(o)>=3 and o[0]=='SELL':demand[o[1]]=demand.get(o[1],0)+max(0,int(o[2]))
        total=sum(demand.values());scale=min(1.0,100/max(1,total))
        warehouse[1]={k:int(v*scale) for k,v in demand.items()}
    money=[float(farms[seat]['money']),float(farms[1-seat]['money'])]
    hires=[int(farms[seat]['hires_today']),int(farms[1-seat]['hires_today'])]
    land=[len(farms[seat]['unlocked_quadrants']),len(farms[1-seat]['unlocked_quadrants'])]
    inventory=dict(obs['market']['inventory']);orders=[mine[:10],theirs[:10]]
    cache={} if cache is None else cache
    def price(item,level):
        key=(item,level)
        if key not in cache:cache[key]=max(1,int(_r37_market_price(item,level,params)))
        return cache[key]
    for slot in range(max(map(len,orders),default=0)):
        active=[None,None]
        for p in (0,1):
            o=orders[p][slot] if slot<len(orders[p]) else []
            if not o:continue
            op=o[0]
            if op=='HIRE':
                cost=_v219_fib(hires[p])
                if money[p]>=cost:money[p]-=cost;hires[p]+=1
            elif op=='BUY_LAND' and land[p]<4:
                cost=(1000,2000,4000)[land[p]-1]
                if money[p]>=cost:money[p]-=cost;land[p]+=1
            elif len(o)>=3:
                item=o[1];quantity=max(0,int(o[2]))
                if op=='BUY_SEED' and item in _MC_SEEDS:
                    n=min(quantity,max(0,int(money[p]//_MC_SEEDS[item])))
                    money[p]-=n*_MC_SEEDS[item]
                elif op=='BUY_ANIMAL' and item in _MC_ANIMALS:
                    n=min(quantity,max(0,int(money[p]//_MC_ANIMALS[item])),max(0,100-sum(warehouse[p].values())))
                    money[p]-=n*_MC_ANIMALS[item];warehouse[p][item]=warehouse[p].get(item,0)+n
                elif op=='SELL' and item in inventory:active[p]=[op,item,quantity]
                elif op=='BUY_PRODUCT' and item in ('WHEAT','FERTILIZER'):active[p]=[op,item,quantity]
        while any(a is not None and a[2]>0 for a in active):
            quoted=[None,None]
            for p,a in enumerate(active):
                if a is not None and a[2]>0:
                    op,item,n=a;quoted[p]=price(item,int(inventory[item])-(op=='BUY_PRODUCT'))
            changed=False
            for p in (0,1):
                a=active[p]
                if a is None or a[2]<=0:continue
                op,item,n=a;quote=quoted[p]
                if op=='SELL':
                    if warehouse[p].get(item,0)<=0:active[p]=None;continue
                    warehouse[p][item]-=1;money[p]+=quote
                    if quote>1:inventory[item]+=1
                else:
                    if money[p]<quote or sum(warehouse[p].values())>=100:active[p]=None;continue
                    warehouse[p][item]=warehouse[p].get(item,0)+1
                    money[p]-=quote;inventory[item]-=1
                a[2]-=1;changed=True
            if not changed:break
    return dict(money=money,warehouse=warehouse,inventory=inventory,margin=money[0]-money[1])
