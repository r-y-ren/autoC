
# Bounded commodity-order experiment inside the Kaggriculture toy economy.
# Numerical forest inference uses public observations and causal history only.
import struct as _cy_struct
_CY_PARENT=phase_b_agent
_CY_STATE={}
_CY_REPORT={'predictions':0,'buy_cycles':0,'sell_cycles':0,'units':0,'errors':0}
def _cy_predict(features):
    x=_cy_struct.unpack('43f',_cy_struct.pack('43f',*features))
    values=[0.0]*4
    for tree in _CY_FOREST:
        node=0
        while tree[node][0]>=0:
            row=tree[node];node=row[2] if x[row[0]]<=row[1] else row[3]
        for j in range(4):values[j]+=tree[node][4+j]
    return [max(0.0,min(1.0,v/len(_CY_FOREST))) for v in values]

def _cy_choose(obs,action,history):
    step=int(obs['step']);seat=int(obs['player'])
    if not _CY_START<=step<_CY_END:return action
    if any(o for o in action.get('market',[])):return action
    cash=float(obs['farms'][seat]['money'])
    if cash<_CY_MIN_CASH:return action
    projected=projected_shed(action,FarmView(obs));room=100-sum(projected.values())
    params=_v44y_params(obs);choices=[]
    for item in _CY_ALLOWED_ITEMS:
        inventory=int(obs['market']['inventory'][item])
        if _r37_market_price(item,inventory+220,params)<=1:continue
        prediction=_cy_predict(cycle_features(obs,item,history))
        _CY_REPORT['predictions']+=1
        for direction in (0,1):
            if _CY_SIDE!='both' and _CY_SIDE!=('buy' if direction==0 else 'sell'):continue
            probability=prediction[direction]
            if probability<_CY_THRESHOLD or probability-prediction[1-direction]<.25:continue
            conditional=40*prediction[2+direction]/max(probability,.001)
            quantity=min(_CY_MAX,max(1,int(conditional+.5)))
            quantity=min(quantity,room if direction==0 else max(0,int(projected.get(item,0))))
            if quantity<=0:continue
            high_quote=max(1,_r37_market_price(item,inventory-220,params))
            if cash<1000+quantity*high_quote:continue
            first='BUY_PRODUCT' if direction==0 else 'SELL'
            second='SELL' if direction==0 else 'BUY_PRODUCT'
            score=probability*quantity*max(1,abs(int(obs['market']['prices'][item])-_r37_market_price(item,inventory+quantity,params)))
            choices.append((score,item,direction,quantity,[[first,item,quantity],[second,item,quantity]]))
    if not choices:return action
    _,item,direction,quantity,orders=max(choices,key=lambda c:c[0])
    if _CY_SHADOW:return action
    _CY_REPORT['buy_cycles' if direction==0 else 'sell_cycles']+=1
    _CY_REPORT['units']+=quantity
    return dict(action,market=orders)

def commodity_cycle_agent(observation,configuration=None):
    step=int(observation['step']);seat=int(observation['player'])
    history=_CY_STATE.get(seat)
    if history is None or step<=history.get('step',-1):
        history=_CY_STATE[seat]={}
        if step==0:
            for key in _CY_REPORT:_CY_REPORT[key]=0
    original=_CY_PARENT(observation,configuration)
    try:action=_cy_choose(observation,original,history)
    except Exception:
        _CY_REPORT['errors']+=1;action=original
    cycle_commit(observation,history)
    return action
_MP_REPORT=_CY_REPORT
commodity_cycle_agent.telemetry=_CY_REPORT
agent=commodity_cycle_agent
kaggle_submission_agent=commodity_cycle_agent
