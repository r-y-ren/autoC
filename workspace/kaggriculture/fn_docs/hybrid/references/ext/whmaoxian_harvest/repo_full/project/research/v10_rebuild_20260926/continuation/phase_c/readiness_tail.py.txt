
# Phase C: infer cargo that cannot yet reach the market from PUBLIC state only.
_C_STATE = {}
_C_DELIVERY_WINDOW = 2
_C_READY_REPORT = {'observed_harvests': 0, 'blocked_checks': 0, 'errors': 0}
_C_ACCESS = ((4, 4), (4, 5), (5, 4), (5, 5))
_C_PARENT = phase_b_agent

def _c_tiles(farm):
    result = {}
    for y, row in enumerate(farm['tiles']):
        for x, tile in enumerate(row):
            if not isinstance(tile, dict):
                continue
            item = tile.get('crop') or _OR2_ANIMAL.get(tile.get('animal'))
            amount = int(tile.get('yield_units', 0))
            if item and amount > 0:
                identity = (tile.get('crop') or tile.get('animal'),
                            tile.get('planted_day', tile.get('placed_day', -1)))
                result[(x, y)] = (item, identity, amount, tile.get('kind'))
    return result

def _c_observe(obs):
    step = int(obs['step']); seat = int(obs['player'])
    farm = obs['farms'][1-seat]
    positions = [tuple(farm['farmer'])] + [tuple(p) for p in farm['hands']]
    state = _C_STATE.get(seat)
    if state is None or step == 0 or state['step'] != step-1:
        state = {'step': step-1, 'positions': [], 'tiles': {}, 'cargo': {}}
        _C_STATE[seat] = state
    cargo = state['cargo']
    old_positions = state['positions']
    if step % 24 == 0:
        cargo.clear()  # The engine banks all inventories at midnight.
    else:
        for actor in list(cargo):
            if actor >= len(positions):
                cargo.pop(actor, None)
            elif actor < len(old_positions):
                if positions[actor] == old_positions[actor] and positions[actor] in _C_ACCESS:
                    cargo.pop(actor, None)  # A drop was possible; do not assume hidden cargo.
        current_tiles = _c_tiles(farm)
        for position, old in state['tiles'].items():
            item, identity, quantity, kind = old
            x, y = position; tile = farm['tiles'][y][x]
            current = current_tiles.get(position)
            cleared = tile is None and kind == 'PLANT' and item not in _OR2_ONGOING
            same = isinstance(tile, dict) and tile.get('kind') != 'WEED'
            now_id = ((tile.get('crop') or tile.get('animal')),
                      tile.get('planted_day', tile.get('placed_day', -1))) if same else None
            harvested = cleared or (same and now_id == identity and int(tile.get('yield_units', 0)) == 0)
            if not harvested:
                continue
            actors = [i for i, p in enumerate(positions) if i < len(old_positions)
                      and p == position and old_positions[i] == position]
            if len(actors) == 1:
                inventory = cargo.setdefault(actors[0], {})
                inventory[item] = inventory.get(item, 0) + quantity
                _C_READY_REPORT['observed_harvests'] += 1
    blocked = {}
    for actor, inventory in cargo.items():
        if actor >= len(positions):
            continue
        x, y = positions[actor]
        distance = min(abs(x-a)+abs(y-b) for a, b in _C_ACCESS)
        earliest_delivery = min(distance, 24-step % 24)
        if earliest_delivery > _C_DELIVERY_WINDOW:
            for item, quantity in inventory.items():
                blocked[item] = blocked.get(item, 0) + quantity
    state.update(step=step, positions=positions, tiles=_c_tiles(farm), blocked=blocked)

def _c_ready(obs, item, estimate):
    blocked = _C_STATE.get(int(obs['player']), {}).get('blocked', {}).get(item, 0)
    if blocked:
        _C_READY_REPORT['blocked_checks'] += 1
    return max(0, int(estimate)-blocked)

def phase_c_agent(observation, configuration=None):
    if int(observation['step']) == 0:
        for key in _C_READY_REPORT:
            _C_READY_REPORT[key] = 0
    _c_observe(observation)
    return _C_PARENT(observation, configuration)

phase_c_agent.telemetry = _C_READY_REPORT
agent = phase_c_agent
kaggle_submission_agent = phase_c_agent
