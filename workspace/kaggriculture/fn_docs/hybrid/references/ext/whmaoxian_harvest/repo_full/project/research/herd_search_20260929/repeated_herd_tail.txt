
# Research-only repeated cow-to-sheep selection. Public observations only.
_K29_HERD_PARENT=phase_b_agent
_K29_CS_DEFAULTS=(_CS_FROM,_CS_TO,_CS_OPTIONS,_CS_SHOP_RULE,_CS_RATIO,_CS_MIN_GAIN)
_K29_SCHEDULE_PARENT=_hd2_schedule
_K29_HERD_REPORT={'eligible':0,'rearms':0,'cash_veto':0,'turns':0}
_MP_REPORT=_K29_HERD_REPORT

def _hd2_schedule(animal,placed_day,day_from):
    if not _K29_CFG['care_model']:
        return _K29_SCHEDULE_PARENT(animal,placed_day,day_from)
    spec=_HD2_SPEC[animal];out={};cap={'GOOSE':4,'COW':6,'SHEEP':6}[animal]
    for day in range(max(day_from,placed_day+spec['first']),30):
        age=day-placed_day
        if (age-spec['first'])%spec['interval']:continue
        cared=max(0,spec['first']-1) if age==spec['first'] else spec['interval']
        out[day]=min(float(cap),1.0+cared*_HD2_CARE)
    return out

def repeated_herd_agent(observation,configuration=None):
    global _CS_FROM,_CS_TO,_CS_OPTIONS,_CS_SHOP_RULE,_CS_RATIO,_CS_MIN_GAIN
    step=int(observation['step']);seat=int(observation['player'])
    if step==0:
        for key in _K29_HERD_REPORT:_K29_HERD_REPORT[key]=0
    _K29_HERD_REPORT['turns']+=1
    _CS_FROM,_CS_TO,_CS_OPTIONS,_CS_SHOP_RULE,_CS_RATIO,_CS_MIN_GAIN=_K29_CS_DEFAULTS
    farm=observation['farms'][seat];shops=observation['town']['unlocked_shops']
    animals=[t.get('animal') for row in farm['tiles'] for t in row if isinstance(t,dict)]
    cows=animals.count('COW');sheep=animals.count('SHEEP')
    sheep+=int(observation['private']['shed'].get('SHEEP',0))
    sheep+=sum(int(inv.get('SHEEP',0)) for inv in observation['private']['inventories'])
    milk=sum(s in ('PIZZA_SHOP','ICE_CREAM_SHOP','SMOOTHIE_SHOP') for s in shops)
    eligible=('YARN_STORE' in shops and (_K29_CFG['milk'] or milk==0)
              and _K29_CFG['start']<=step<_K29_CFG['end']
              and cows>=_K29_CFG['min_cows'] and sheep<_K29_CFG['max_sheep'])
    state=_CS_STATE.get(seat)
    if state and state.get('mode') not in (None,'SHEEP'):eligible=False
    if eligible:
        _K29_HERD_REPORT['eligible']+=1
        _CS_FROM=_K29_CFG['start'];_CS_TO=_K29_CFG['end']
        _CS_OPTIONS=('SHEEP',);_CS_SHOP_RULE=''
        _CS_RATIO=_K29_CFG['ratio'];_CS_MIN_GAIN=_K29_CFG['gain']
        tape=_cs_tape(seat,step)
        planned=any(len(o)>=3 and o[:2]==['BUY_ANIMAL','COW'] and int(o[2])>0 for o in tape.get('market',[]))
        if planned and float(farm['money'])<_K29_CFG['cash']:
            _CS_FROM=720;_K29_HERD_REPORT['cash_veto']+=1
        elif planned and state and _K29_CFG['repeat']:
            state['decided']=False
            _K29_HERD_REPORT['rearms']+=1
    return _K29_HERD_PARENT(observation,configuration)
repeated_herd_agent.telemetry=_K29_HERD_REPORT
agent=repeated_herd_agent
kaggle_submission_agent=repeated_herd_agent
