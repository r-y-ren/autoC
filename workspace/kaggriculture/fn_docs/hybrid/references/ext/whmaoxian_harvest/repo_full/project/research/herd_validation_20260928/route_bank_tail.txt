_RT28_STATE={}
_RT28_REPORT={'switches':0,'matches':0,'inventory_vetoes':0,'route':-1,'errors':0}
_MP_REPORT=_RT28_REPORT
_PREEMPT_ENABLED=False

def _v17_r5_counter(obs,action,step):return action

def _v17_md_counter(obs,action,step):return action

def _rt28_key(farm):
    return json.dumps({k:v for k,v in farm.items() if k!='money'},sort_keys=True,separators=(',',':'))

def _rt28_resources(private,required):
    for field in ('shed','seeds'):
        actual=private.get(field,{})
        if any(actual.get(k,0)<v for k,v in required.get(field,{}).items()):return False
    inventories=private.get('inventories',[])
    for i,inv in enumerate(required.get('inventories',[])):
        if i>=len(inventories):return False
        if any(inventories[i].get(k,0)<v for k,v in inv.items()):return False
    return True

def _rt28_score(obs,snapshot):
    actual=obs['town']['unlocked_shops'];reference=snapshot['shops']
    score=100.0 if actual==reference else 0.0
    for shop in set(actual+reference):score-=12*abs(actual.count(shop)-reference.count(shop))
    for item,price in snapshot['prices'].items():
        score-=abs(float(obs['market']['prices'].get(item,price))-price)/max(1,price)
    return score

def state_aligned_route_agent(obs,configuration=None):
    global _ACTIONS
    seat=int(obs['player']);step=int(obs['step'])
    if step==0 or seat not in _RT28_STATE:
        _RT28_STATE[seat]={'route':_RT28_ROOT}
        for key in _RT28_REPORT:_RT28_REPORT[key]=0
        _NP28_REPORT['errors']=0
    state=_RT28_STATE[seat];current=state['route']
    if step in (72,144) and _RT28_MODE!='fixed':
        key=_rt28_key(obs['farms'][seat]);choices=[]
        for index,record in enumerate(_RT28_BANK):
            snapshot=record['snapshots'][str(step)]
            if snapshot['farm_key']!=key:continue
            _RT28_REPORT['matches']+=1
            if not _rt28_resources(obs['private'],snapshot['private']):
                _RT28_REPORT['inventory_vetoes']+=1;continue
            if _RT28_MODE=='exact_shops' and obs['town']['unlocked_shops']!=snapshot['shops']:continue
            score=_rt28_score(obs,snapshot)+(0.1 if index==current else 0)
            choices.append((score,-index,index))
        if choices:
            chosen=max(choices)[2]
            _RT28_REPORT['switches']+=int(chosen!=current);state['route']=chosen
    _ACTIONS=_RT28_BANK[state['route']]['actions']
    action=_E279_PUBLIC_AGENT(obs)
    _RT28_REPORT['route']=state['route'];_RT28_REPORT['errors']=_NP28_REPORT['errors']
    return action
state_aligned_route_agent.telemetry=_RT28_REPORT
agent=state_aligned_route_agent
