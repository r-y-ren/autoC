
# Finite-horizon husbandry: do not consume grain after its last possible benefit.
_TF_PARENT=phase_b_agent
_TF_REPORT={'feed_saved':0,'care_replaced':0,'harvest_requests':0,'fertilizer_requests':0,'errors':0}
_MP_REPORT=_TF_REPORT
_TF_SPEC={'GOOSE':(4,1),'COW':(8,2),'SHEEP':(6,3)}
def _tf_no_feed_benefit(tile,day):
    if not isinstance(tile,dict) or tile.get('animal') not in _TF_SPEC:return False
    if day>=29:return True
    if day!=28 or tile.get('consecutive_unfed',0)!=0:return False
    first,interval=_TF_SPEC[tile['animal']]
    age=29-int(tile['placed_day'])
    return age<first or (age-first)%interval!=0

def terminal_feed_agent(observation,configuration=None):
    step=int(observation['step']);day=step//24;seat=int(observation['player'])
    if step==0:
        for key in _TF_REPORT:_TF_REPORT[key]=0
    action=_TF_PARENT(observation,configuration)
    if step<672 or step>=712:return action
    farm=observation['farms'][seat];positions=[farm['farmer']]+list(farm['hands'])
    commands=[list(action.get('farmer') or ['PASS'])]+[list(c) for c in action.get('hands',[])]
    changed=False;claimed=set()
    for actor,pos in enumerate(positions[:len(commands)]):
        tile=farm['tiles'][pos[1]][pos[0]];cmd=commands[actor]
        if _TF_FEED and cmd==['FEED'] and _tf_no_feed_benefit(tile,day):
            commands[actor]=['PASS'];_TF_REPORT['feed_saved']+=1;changed=True
        if _TF_CARE and cmd==['CARE'] and isinstance(tile,dict) and tile.get('animal') and day>=28:
            key=tuple(pos)
            if key in claimed:continue
            if int(tile.get('yield_units',0))>0:
                commands[actor]=['HARVEST'];_TF_REPORT['harvest_requests']+=1
            elif tile.get('fertilizer_available'):
                commands[actor]=['COLLECT_FERTILIZER'];_TF_REPORT['fertilizer_requests']+=1
            else:continue
            claimed.add(key);_TF_REPORT['care_replaced']+=1;changed=True
    if changed:return dict(action,farmer=commands[0],hands=commands[1:])
    return action
terminal_feed_agent.telemetry=_TF_REPORT
agent=terminal_feed_agent
kaggle_submission_agent=terminal_feed_agent
