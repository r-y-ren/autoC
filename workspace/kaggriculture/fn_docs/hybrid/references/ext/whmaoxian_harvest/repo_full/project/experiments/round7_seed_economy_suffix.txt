

# Local round7 experiment, 2026-09-22. Preserve Apache-2.0 upstream notices.
# Cap late CARROT purchases by the remaining scheduled planting opportunities.
# This uses the agent's own public fixed schedule, never replay future outcomes.
_SE_PARENT = agent
_SE_REPORT = {"seed_units_avoided": 0, "seed_cap_errors": 0}

def agent(observation, configuration=None):
    step = int(observation["step"])
    if step == 0:
        _SE_REPORT.update(seed_units_avoided=0, seed_cap_errors=0)
    action = _SE_PARENT(observation, configuration)
    if step < 648:
        return action
    try:
        seat = int(observation["player"])
        # The carrot substitution layer stops at day 28. In the terminal route,
        # every future carrot comes from a scheduled WHEAT/CARROT planting slot.
        remaining = 0
        for t in range(step + 1, min(719, _CA_TO * 24)):
            plan = _ca_tape(seat, t)
            units = [plan.get("farmer") or ["PASS"]] + list(plan.get("hands") or [])
            remaining += sum(bool(c) and len(c) > 1 and c[0] == "PLANT" and c[1] in ("WHEAT", "CARROT") for c in units)
        units = [action.get("farmer") or ["PASS"]] + list(action.get("hands") or [])
        demand = sum(bool(c) and c[:2] == ["PLANT", "CARROT"] for c in units)
        have = int(observation["private"]["seeds"].get("CARROT", 0))
        # Official atomic seed validation drops all requests if underfunded.
        after_plant = have - demand if have >= demand else have
        allowed = max(0, remaining - after_plant)
        market, avoided = [], 0
        for order in action.get("market") or []:
            if len(order) >= 3 and order[:2] == ["BUY_SEED", "CARROT"]:
                take = min(max(0, int(order[2])), allowed)
                allowed -= take
                avoided += int(order[2]) - take
                if take:
                    market.append(["BUY_SEED", "CARROT", take])
            else:
                market.append(order)
        if avoided:
            action = dict(action, market=market)
            # The parent's accounting credits seed buffer purchases before this
            # wrapper; keep its actual-purchase credit consistent with the cut.
            state = _CA_STATE.get(seat)
            if state is not None:
                state["spare_carrot"] = max(0, state.get("spare_carrot", 0) - avoided)
            _SE_REPORT["seed_units_avoided"] += avoided
    except Exception:
        _SE_REPORT["seed_cap_errors"] += 1
    return action

agent.telemetry = _SE_REPORT
agent = globals().pop("agent")
kaggle_submission_agent = agent
