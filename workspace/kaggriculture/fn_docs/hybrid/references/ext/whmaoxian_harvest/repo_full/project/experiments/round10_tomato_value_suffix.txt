
# Round 10 isolated experiment: value a physically scheduled ten-tomato project.
# The public V219 worker/market implementation above is unchanged.  The only
# intervention is its day-18 qualification.  V9's 19-tile branch is held off so
# that a ten-tile forecast is never used to authorise a different-size project.
import itertools as _r10tv_it

_R10TV_PARENT = round9_slack_agent
_R10TV_REPORT = {
    'calls': 0, 'approved': 0, 'rejected': 0, 'projected_revenue': 0,
    'projected_cost': 0, 'projected_margin': 0, 'rival_price_loss': 0,
    'fertilizer_mode': 0, 'errors': 0, 'reason': '',
}
_R10TV_TARGETS = tuple((x, y) for y in (5, 6) for x in range(5, 10))


def _r10tv_structural(obs, native):
    """The old project's physical preconditions, without its price/shop gates."""
    if int(obs['step']) != 432:
        return 'wrong_day'
    seat = int(obs['player'])
    farm = obs['farms'][seat]
    if len(farm['tiles']) != 10 or set(farm['unlocked_quadrants']) != {'NW', 'NE', 'SW'}:
        return 'land_layout'
    if any(farm['tiles'][y][x] != 'LOCKED' for x, y in _R10TV_TARGETS):
        return 'target_occupied'
    private = obs['private']
    if private['seeds'].get('TOMATO', 0) or private['shed'].get('TOMATO', 0):
        return 'existing_tomato_stock'
    if any(isinstance(t, dict) and t.get('crop') == 'TOMATO'
           for row in farm['tiles'] for t in row):
        return 'existing_tomato_crop'
    for tape in _IMPL.chassis.routes.values():
        for action in tape[432:719]:
            if any(o and o[0] == 'BUY_LAND' for o in action.get('market', [])):
                return 'native_land_conflict'
            if any(c == ['PLANT', 'TOMATO']
                   for c in [action.get('farmer')] + action.get('hands', [])):
                return 'native_tomato_conflict'
    # The V219 transaction needs four free market slots: land, ten seeds,
    # and two new hands.  It will recheck the exact current queue before buying.
    tape18 = _v219_native_day(native, 18)
    if not any(len(a.get('market', [])) + 4 <= MAX_ORDERS for a in tape18[:4]):
        return 'order_slots'
    hands = max(len(a.get('hands', [])) for a in tape18)
    first_labor = _v219_fib(hands) + _v219_fib(hands + 1)
    if float(farm['money']) < 4500 + first_labor + 3000:
        return 'first_day_liquidity'
    return ''


def _r10tv_labor(native, fertilize):
    """Upper-cost crew plan for the existing V219 route, by actual native day."""
    total = 0
    for day in range(18, 30):
        tape = _v219_native_day(native, day)
        base = max(len(a.get('hands', [])) for a in tape)
        if day in (18, 24, 29):
            count = 2
        elif day in (26, 27, 28):
            count = 3
        else:
            count = 1
        # If the day-27 route optimiser cannot cover fertilising in three
        # people, V219 hires a fourth dedicated hand.
        if fertilize and day == 27:
            count += 1
        total += sum(_v219_fib(base + i) for i in range(count))
    return total


def _r10tv_rival_supply(obs):
    """Visible planted tomatoes, including one physically possible replant."""
    daily = {day: 0 for day in range(18, 30)}
    farm = obs['farms'][1 - int(obs['player'])]
    for row in farm['tiles']:
        for tile in row:
            if not isinstance(tile, dict) or tile.get('crop') != 'TOMATO':
                continue
            born = int(tile.get('planted_day', 30))
            first, last = born + 8, born + 11
            for day in range(max(18, first), min(29, last) + 1):
                daily[day] += 1
            # Assume that a rival maintaining a visible tomato site may replant
            # after this crop ends; do not assume instant maturation.
            for day in range(max(18, last + 9), min(29, last + 12) + 1):
                daily[day] += 1
            if first <= 18 <= last:
                daily[18] += max(0, int(tile.get('yield_units', 0)))
    return daily


def _r10tv_price_params(obs, item):
    p = dict(_R37_MARKET_PARAMS[item])
    p.update(obs['market'].get('params', {}).get(item, {}))
    return {item: p}


def _r10tv_economy(obs, native, fertilize, extra_shops):
    """Incremental delivered bank and visible rival-price effect to day 29."""
    tomato_params = _r10tv_price_params(obs, 'TOMATO')
    inventory_base = float(obs['market']['inventory']['TOMATO'])
    inventory_project = inventory_base
    tomato_shops = sum(s in ('PIZZA_SHOP', 'FARMERS_MARKET')
                        for s in obs['town']['unlocked_shops'])
    rival = _r10tv_rival_supply(obs)
    own_revenue = rival_loss = 0.0
    # Ten plants yield one unit per day on days 26..29, plus one if fertilised.
    # Reserve one unit per day for missed harvest or delivery.  The V219 route
    # must still confirm actual production in the league.
    own_daily = (19 if fertilize else 9)
    unlock_days = (21, 24)
    for day in range(18, 30):
        if day in unlock_days:
            tomato_shops += extra_shops[unlock_days.index(day)]
        town_drain = 1 + 6 * tomato_shops
        inventory_base -= town_drain
        inventory_project -= town_drain
        rival_qty = rival[day]
        for unit in range(rival_qty):
            before = _r37_market_price('TOMATO', inventory_base + unit, tomato_params)
            after = _r37_market_price('TOMATO', inventory_project + unit, tomato_params)
            rival_loss += max(0, before - after)
        inventory_base += rival_qty
        inventory_project += rival_qty
        if 26 <= day <= 29:
            for unit in range(own_daily):
                own_revenue += _r37_market_price(
                    'TOMATO', inventory_project + unit, tomato_params)
            inventory_project += own_daily
    cost = 4500 + _r10tv_labor(native, fertilize)
    if fertilize:
        fert_params = _r10tv_price_params(obs, 'FERTILIZER')
        fert_inventory = float(obs['market']['inventory']['FERTILIZER'])
        # V219 buys at least ten fertilizer units on each of days 24 and 27.
        # Reserve the upper quote after those units enter the buy queue.
        unit_cost = _r37_market_price('FERTILIZER', fert_inventory - 20, fert_params) + 5
        cost += 20 * unit_cost
    return own_revenue, cost, rival_loss


def _r10tv_decision(obs, native):
    # Known timetable has at most two future shop draws.  Each draw has two
    # tomato buyers among eight equally likely public shop types.
    slots = min(2, 8 - len(obs['town']['unlocked_shops']))
    scenarios = []
    for draws in _r10tv_it.product((0, 1), repeat=slots):
        later = tuple(draws) + (0,) * (2 - slots)
        p = (0.25 ** sum(draws)) * (0.75 ** (slots - sum(draws)))
        scenarios.append((later, p))
    nofert = [(_r10tv_economy(obs, native, False, draws), p)
              for draws, p in scenarios]
    fert = [(_r10tv_economy(obs, native, True, draws), p)
            for draws, p in scenarios]
    nofert_revenue = sum(e[0] * p for e, p in nofert)
    fert_revenue = sum(e[0] * p for e, p in fert)
    nofert_cost = nofert[0][0][1]
    fert_cost = fert[0][0][1]
    # The native V219 fertilizer switch itself requires a large benefit over
    # input cost.  Forecast the same two-to-one condition before crediting
    # extra tomato output; its actual price gate rechecks later.
    use_fert = fert_revenue - nofert_revenue >= 2 * (fert_cost - nofert_cost)
    cases = fert if use_fert else nofert
    revenue = fert_revenue if use_fert else nofert_revenue
    cost = fert_cost if use_fert else nofert_cost
    variance = sum(p * (e[0] - revenue) ** 2 for e, p in cases)
    # Half a standard deviation guards against unseen future shop draws.
    # Do not credit rival-price damage: it is uncertain, so the resulting
    # own-bank gain is a conservative lower bound on modelled bank margin.
    margin = revenue - cost - 0.5 * variance ** 0.5
    rival_loss = sum(e[2] * p for e, p in cases)
    return margin > 0, revenue, cost, margin, rival_loss, int(use_fert)


def _v219_qualifies(obs, native):
    _R10TV_REPORT['calls'] += 1
    reason = _r10tv_structural(obs, native)
    if reason:
        _R10TV_REPORT['rejected'] += 1
        _R10TV_REPORT['reason'] = reason
        return False
    approved, revenue, cost, margin, rival_loss, fert = _r10tv_decision(obs, native)
    _R10TV_REPORT.update(projected_revenue=round(revenue), projected_cost=round(cost),
                         projected_margin=round(margin), rival_price_loss=round(rival_loss),
                         fertilizer_mode=fert, reason='' if approved else 'negative_net_margin')
    _R10TV_REPORT['approved' if approved else 'rejected'] += 1
    return approved

def round10_tomato_value_agent(observation, configuration=None):
    if int(observation['step']) == 0:
        _R10TV_REPORT.update(calls=0, approved=0, rejected=0,
                             projected_revenue=0, projected_cost=0,
                             projected_margin=0, rival_price_loss=0,
                             fertilizer_mode=0, errors=0, reason='')
