
# Reject weak historical-template forecasts instead of always trusting the best one.
_PG_PARENT=phase_b_agent
_PG_FORECAST_PARENT=_v92_p_forecast
_PG_REPORT={'forecast_calls':0,'accepted_templates':0,'empty_forecasts':0,'disabled_calls':0,'errors':0}
_MP_REPORT=_PG_REPORT

def _v92_p_forecast(obs,st):
    step=int(obs['step']);lo=step-240;seen=st['obs']
    shops=tuple(obs['town']['unlocked_shops'][:2]);candidates=_v92_p_pair(shops)
    recent=[(t,i) for t,i in seen if t>=lo];scored=[]
    _PG_REPORT['forecast_calls']+=1
    for ordinal,ev in candidates:
        matches=0;false=0
        for (t,i),q in ev.items():
            if lo<=t<step-1:
                if (t,i) in seen or (t-1,i) in seen or (t+1,i) in seen:matches+=1
                else:false+=1
        missed=sum((t,i) not in ev and (t-1,i) not in ev and (t+1,i) not in ev for t,i in recent)
        score=matches-.5*false-.5*missed
        if matches>=_PG_MATCH and matches/max(1,matches+false)>=_PG_PRECISION and score>=_PG_SCORE:
            scored.append((score,ev))
    scored.sort(key=lambda r:-r[0]);selected=[ev for _,ev in scored[:_V92_P_TOP]]
    _PG_REPORT['accepted_templates']+=len(selected)
    _PG_REPORT['empty_forecasts']+=int(not selected)
    return selected
