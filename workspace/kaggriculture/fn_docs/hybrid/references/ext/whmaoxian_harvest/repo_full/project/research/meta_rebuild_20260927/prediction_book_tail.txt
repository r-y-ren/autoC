
# Separate causal predictor observations from the legacy race heuristics.
_PB_PARENT=phase_b_agent
_PB_BOOK={}
_PB_REPORT={'corrected_items':0,'corrected_units':0,'errors':0}
_MP_REPORT=_PB_REPORT

def prediction_book_agent(observation,configuration=None):
    seat=int(observation['player']);step=int(observation['step'])
    if step==0:
        _PB_BOOK.pop(seat,None)
        for key in _PB_REPORT:_PB_REPORT[key]=0
    action=_PB_PARENT(observation,configuration)
    previous=_V9_RACE.get(seat,{}).get('prev')
    if previous and previous.get('step')==step:
        snapshot=dict(previous);snapshot['own']=dict(previous['own']);snapshot['left']=dict(previous['left'])
        stock=projected_shed(action,FarmView(observation));orders=action.get('market',[])[:10]
        bought={o[1] for o in orders if len(o)>=3 and o[0]=='BUY_PRODUCT'}
        for item in V9_RACE_ITEMS:
            if item in bought:continue
            requested=sum(max(0,int(o[2])) for o in orders if len(o)>=3 and o[:2]==['SELL',item])
            sold=min(max(0,int(stock.get(item,0))),requested)
            old=int(snapshot['own'].get(item,0))
            if sold!=old:
                _PB_REPORT['corrected_items']+=1;_PB_REPORT['corrected_units']+=abs(sold-old)
            snapshot['own'][item]=sold;snapshot['left'][item]=max(0,int(stock.get(item,0))-sold)
        _PB_BOOK[seat]={'prev':snapshot}
    return action
prediction_book_agent.telemetry=_PB_REPORT
agent=prediction_book_agent
kaggle_submission_agent=prediction_book_agent
