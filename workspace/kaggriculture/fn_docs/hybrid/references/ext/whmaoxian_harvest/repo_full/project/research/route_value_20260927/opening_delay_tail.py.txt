
# A virtual-game opening study: preserve five wheat before the first pickup.
_OP_PARENT=phase_b_agent
_OP_REPORT={'deferred_units':0,'topup_units':0,'changed_turns':0}
def opening_delay_agent(observation,configuration=None):
    step=int(observation['step']);seat=int(observation['player'])
    if step==0:
        for key in _OP_REPORT:_OP_REPORT[key]=0
    action=_OP_PARENT(observation,configuration)
    if step>3:return action
    orders=[list(o) for o in action.get('market',[])]
    if step<_OP_DUE:
        for index,order in enumerate(orders):
            if len(order)>=3 and order[:2]==['BUY_PRODUCT','WHEAT']:
                quantity=min(max(0,int(order[2])),_OP_INITIAL if step==0 else 0)
                _OP_REPORT['deferred_units']+=max(0,int(order[2]))-quantity
                orders[index]=['BUY_PRODUCT','WHEAT',quantity] if quantity else []
    if step>=_OP_DUE:
        private=observation['private']
        grain=int(private['shed'].get('WHEAT',0))+sum(int(b.get('WHEAT',0)) for b in private['inventories'])
        incoming=sum(int(o[2]) for o in orders if len(o)>=3 and o[:2]==['BUY_PRODUCT','WHEAT'])
        outgoing=sum(int(o[2]) for o in orders if len(o)>=3 and o[:2]==['SELL','WHEAT'])
        quantity=max(0,5-grain-incoming+outgoing)
        if quantity and len(orders)<10:
            orders.append(['BUY_PRODUCT','WHEAT',quantity]);_OP_REPORT['topup_units']+=quantity
    if orders!=action.get('market',[]):_OP_REPORT['changed_turns']+=1
    return dict(action,market=orders)
opening_delay_agent.telemetry=_OP_REPORT
_MP_REPORT=_OP_REPORT
agent=opening_delay_agent
kaggle_submission_agent=opening_delay_agent
