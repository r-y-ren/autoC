
# Earlier and repeatable cow-to-sheep decisions using only current public state.
_K28E_PARENT=phase_b_agent
_K28E_NATIVE=(_CS_FROM,_CS_TO,_CS_OPTIONS,_CS_SHOP_RULE,_CS_RATIO,_CS_MIN_GAIN)
_K28E_PLAN_PARENT=_cs_plan
_K28E_REPORT={'eligible_turns':0,'reopened':0,'budget_declines':0,'plans':0}
_MP_REPORT=_K28E_REPORT

def _cs_plan(obs,action,k,mode):
    if mode=='SHEEP' and _K28E_BUDGET:
        farm=obs['farms'][int(obs['player'])];cost=0
        hires=int(farm['hires_today']);land=len(farm['unlocked_quadrants'])
        for order in action.get('market',[]):
            if not order:continue
            op=order[0]
            if op=='HIRE':cost+=_v219_fib(hires);hires+=1
            elif op=='BUY_LAND' and land<4:cost+=(1000,2000,4000)[land-1];land+=1
            elif len(order)>=3:
                item=order[1];n=max(0,int(order[2]))
                if op=='BUY_ANIMAL':cost+=n*{'GOOSE':300,'COW':400,'SHEEP':500}[item]
                elif op=='BUY_SEED':cost+=n*{'WHEAT':10,'CARROT':20,'TOMATO':50,'STRAWBERRY':100,'MELON':80}[item]
                elif op=='BUY_PRODUCT':cost+=n*{'WHEAT':35,'FERTILIZER':110}.get(item,1000000)
        if farm['money']<cost+100*k+50:
            _K28E_REPORT['budget_declines']+=1;return None
    result=_K28E_PLAN_PARENT(obs,action,k,mode)
    _K28E_REPORT['plans']+=int(result is not None)
    return result

def early_herd_agent(observation,configuration=None):
    global _CS_FROM,_CS_TO,_CS_OPTIONS,_CS_SHOP_RULE,_CS_RATIO,_CS_MIN_GAIN
    step=int(observation['step']);seat=int(observation['player'])
    if step==0:
        for key in _K28E_REPORT:_K28E_REPORT[key]=0
    _CS_FROM,_CS_TO,_CS_OPTIONS,_CS_SHOP_RULE,_CS_RATIO,_CS_MIN_GAIN=_K28E_NATIVE
    shops=observation['town']['unlocked_shops']
    milk=any(s in ('PIZZA_SHOP','ICE_CREAM_SHOP','SMOOTHIE_SHOP') for s in shops)
    eligible='YARN_STORE' in shops and (not _K28E_NOMILK or not milk) and 72<=step<_K28E_TO
    state=_CS_STATE.get(seat)
    if state and len(state.get('sites',{}))>=_K28E_CAP:eligible=False
    if eligible:
        _CS_FROM=72;_CS_TO=_K28E_TO;_CS_OPTIONS=('SHEEP',)
        _CS_SHOP_RULE='';_CS_RATIO=_K28E_RATIO;_CS_MIN_GAIN=_K28E_GAIN
        _K28E_REPORT['eligible_turns']+=1
        if _K28E_REPEAT and state and state.get('mode') in (None,'SHEEP') and not state.get('broken'):
            plan=state.get('plan') or {};last=max((t for t,i in plan.get('places',{})),default=-1)
            tape=_cs_tape(seat,step)
            count=sum(int(o[2]) for o in tape.get('market',[]) if len(o)>=3 and o[:2]==['BUY_ANIMAL','COW'])
            room=_K28E_CAP-len(state.get('sites',{}))
            if 0<count<=room and step>last and not state.get('pending'):
                state['decided']=False;_K28E_REPORT['reopened']+=1
    return _K28E_PARENT(observation,configuration)
early_herd_agent.telemetry=_K28E_REPORT
agent=early_herd_agent
kaggle_submission_agent=early_herd_agent
