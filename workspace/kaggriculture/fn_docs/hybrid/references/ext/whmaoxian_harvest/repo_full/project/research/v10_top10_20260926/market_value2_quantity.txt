
def _mv_quantity(obs,item,available,original,params):
    step=int(obs['step']);seat=int(obs['player']);horizon=_MV_HORIZON
    shops=obs['town']['unlocked_shops']
    per_tick=sum((2 if len(_OR2_SHOPS.get(s,()))==1 else 1) for s in shops if item in _OR2_SHOPS.get(s,()))
    demand=per_tick*sum(t%4==0 for t in range(step,min(719,step+horizon)))
    demand+=sum(t%24==0 for t in range(step,min(719,step+horizon)))
    stored=max(0,int(_OR2_STATE.get(seat,{}).get('stock',{}).get(item,0)))
    mature=0
    for row in obs['farms'][1-seat]['tiles']:
        for tile in row:
            if not isinstance(tile,dict):continue
            product=tile.get('crop') or {'COW':'MILK','SHEEP':'WOOL','GOOSE':'EGG'}.get(tile.get('animal'))
            if product==item:mature+=max(0,int(tile.get('yield_units',0)))
    rival=min(80,int(stored*_MV_STOCK_WEIGHT+mature*_MV_MATURE_WEIGHT+.5))
    scenarios=((0,_MV_ZERO_PROB),(rival,(1-_MV_ZERO_PROB)*.7),(min(100,rival*2),(1-_MV_ZERO_PROB)*.3))
    inventory=int(obs['market']['inventory'][item]);maximum=max(original,min(available,_MV_LOT))
    total=available
    low=inventory-demand-2;high=inventory+total+102
    cumulative=[0];growth=[0]
    for level in range(low,high+1):
        price=max(1,_r37_market_price(item,level,params))
        cumulative.append(cumulative[-1]+price);growth.append(growth[-1]+int(price>1))
    def revenue(level,quantity):
        index=level-low
        return cumulative[index+quantity]-cumulative[index]
    def after(level,quantity):
        index=level-low
        return level+growth[index+quantity]-growth[index]
    before=demand//2;remaining=demand-before;scores=[]
    for quantity in range(maximum+1):
        now=revenue(inventory,quantity);post=after(inventory,quantity)
        value=0.0
        for other,weight in scenarios:
            rival_revenue=revenue(post-before,other)
            ending=after(post-before,other)-remaining
            our_later=revenue(ending,total-quantity)
            value+=weight*(now+our_later-_MV_RIVAL_WEIGHT*rival_revenue)
        scores.append(value)
    best=max(range(len(scores)),key=lambda q:(scores[q],-abs(q-original)))
    if scores[best]<=scores[original]+_MV_MIN_GAIN:return original
    return best
