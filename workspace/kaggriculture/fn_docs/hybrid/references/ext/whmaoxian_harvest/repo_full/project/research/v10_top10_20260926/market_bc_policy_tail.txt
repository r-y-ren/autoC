
# Local Kaggriculture experiment: bounded SELL advice from arithmetic trees.
# No network, credentials, opponent-private state, or external files at runtime.
import struct as _mbc_struct
_MBC_PARENT = phase_b_agent
_MBC_STATE = {}
_MBC_STATS = dict(predictions=0, proposed=0, changed=0, extra=0, held=0, errors=0)
_MBC_CFG = dict(mode='shadow', threshold=0.70, hold_threshold=0.10,
                max_extra=8, start=216, end=696, min_cash=8000, hold=False)

def _mbc_predict(features):
    if len(features) != 45:
        raise ValueError('market feature count mismatch')
    x = _mbc_struct.unpack('45f', _mbc_struct.pack('45f', *features))
    sums = [0.0, 0.0]
    for tree in _MBC_FOREST:
        node = 0
        while tree[node][0] >= 0:
            row = tree[node]
            node = row[2] if x[row[0]] <= row[1] else row[3]
        sums[0] += tree[node][4]
        sums[1] += tree[node][5]
    return [max(0.0, min(1.0, value / len(_MBC_FOREST))) for value in sums]

def _mbc_sold(action, available):
    return {item: min(max(0, int(available.get(item, 0))), sum(
        max(0, int(o[2])) for o in action.get('market', [])[:10]
        if len(o) >= 3 and o[:2] == ['SELL', item])) for item in _MBC_ITEMS}

def _mbc_advise(obs, action, context):
    cfg = _MBC_CFG; step = int(obs['step'])
    available = context['projected']
    orders = [list(o) for o in action.get('market', [])]
    initial = _mbc_sold(action, available)
    if not cfg['start'] <= step < cfg['end'] or step % 24 == 23:
        return action
    if context['own']['money'] < cfg['min_cash']:
        return action
    if any(o and o[0] == 'BUY_PRODUCT' for o in orders):
        return action
    picked = {c[1] for c in [action.get('farmer', [])] + action.get('hands', [])
              if len(c) > 1 and c[0] == 'PICKUP'}
    proposals = []
    for item in _MBC_ITEMS:
        stock = max(0, int(available.get(item, 0)))
        if not stock or item in picked:
            continue
        probability, fraction = _mbc_predict(_mbc_features(context, item))
        _MBC_STATS['predictions'] += 1
        indices = [i for i, o in enumerate(orders)
                   if len(o) >= 3 and o[:2] == ['SELL', item]]
        original = initial[item]; desired = original
        if len(indices) > 1 or int(obs['market']['prices'][item]) <= 1:
            continue
        if probability >= cfg['threshold']:
            desired = min(stock, original + cfg['max_extra'],
                          max(original, int(stock * fraction + 0.5)))
        elif cfg['hold'] and probability <= cfg['hold_threshold']:
            if context['total'] < 60 and not context['non_sells']:
                desired = max(0, original - min(2, original))
        if desired != original:
            proposals.append((probability, item, original, desired, indices))
    _MBC_STATS['proposed'] += len(proposals)
    if cfg['mode'] == 'shadow':
        return action
    for probability, item, original, desired, indices in sorted(proposals, reverse=True):
        if indices:
            orders[indices[0]] = ['SELL', item, desired] if desired else []
        elif desired and len(orders) < 10:
            orders.append(['SELL', item, desired])
        else:
            continue
        _MBC_STATS['extra'] += max(0, desired - original)
        _MBC_STATS['held'] += max(0, original - desired)
    if orders == action.get('market', []):
        return action
    _MBC_STATS['changed'] += 1
    return dict(action, market=orders)

def _mbc_sync_observers(obs, action, sold):
    seat = int(obs['player']); step = int(obs['step'])
    previous = _OR2_STATE.get(seat, {}).get('prev')
    if previous and previous.get('step') == step:
        previous.setdefault('own', {}).update(sold)
    race = _RACE_STATE.get(seat, {})
    if race.get('step') == step and race.get('prev_action') is not None:
        race['prev_action'] = action

def market_bc_agent(observation, configuration=None):
    seat = int(observation['player']); step = int(observation['step'])
    history = _MBC_STATE.get(seat)
    if history is None or step <= history.get('last_step', -1):
        history = _MBC_STATE[seat] = {}
    if step == 0:
        for key in _MBC_STATS:
            _MBC_STATS[key] = 0
    original = _MBC_PARENT(observation, configuration)
    try:
        available = projected_shed(original, FarmView(observation))
        context = _mbc_context(observation, original, available, history)
        action = _mbc_advise(observation, original, context)
        sold = _mbc_sold(action, available)
        _mbc_commit(context, sold)
        if action is not original:
            _mbc_sync_observers(observation, action, sold)
        return action
    except Exception:
        _MBC_STATS['errors'] += 1
        return original
market_bc_agent.telemetry = _MBC_STATS
