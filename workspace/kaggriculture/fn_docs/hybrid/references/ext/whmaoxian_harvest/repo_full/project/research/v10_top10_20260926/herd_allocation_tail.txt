
# Test demand-sensitive livestock allocation on existing, verified worker routes.
_HA_PARENT=phase_b_agent
_HA_PLAN=_cs_plan
_HA_REPORT={'budget_declines':0,'errors':0}
def _cs_plan(obs,action,k,mode):
    if mode=='SHEEP':
        farm=obs['farms'][int(obs['player'])];cost=0;hires=int(farm['hires_today'])
        land=len(farm['unlocked_quadrants'])
        for order in action.get('market',[]):
            if not order:continue
            if order[0]=='HIRE':cost+=_v219_fib(hires);hires+=1
            elif order[0]=='BUY_LAND' and land<4:cost+=(1000,2000,4000)[land-1];land+=1
            elif len(order)>=3:
                op,item,q=order[:3];q=max(0,int(q))
                if op=='BUY_SEED':cost+=q*{'WHEAT':10,'CARROT':20,'TOMATO':50,'STRAWBERRY':100,'MELON':80}[item]
                elif op=='BUY_ANIMAL':cost+=q*{'GOOSE':300,'COW':400,'SHEEP':500}[item]
                elif op=='BUY_PRODUCT':cost+=q*(int(obs['market']['prices'][item])+10)
        # No contemporaneous sale proceeds assumed; protect committed purchases.
        if farm['money']<cost+100*k+200:
            _HA_REPORT['budget_declines']+=1
            return None
    return _HA_PLAN(obs,action,k,mode)
def herd_allocation_agent(observation,configuration=None):
    if int(observation['step'])==0:
        for key in _HA_REPORT:_HA_REPORT[key]=0
    return _HA_PARENT(observation,configuration)
herd_allocation_agent.telemetry=_HA_REPORT
agent=herd_allocation_agent
kaggle_submission_agent=herd_allocation_agent
