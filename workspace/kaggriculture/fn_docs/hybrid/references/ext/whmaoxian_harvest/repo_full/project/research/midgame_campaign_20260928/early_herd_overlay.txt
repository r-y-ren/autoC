
# Extend the existing transaction-confirmed herd controller, not its transport.
_K28Y_PARENT=phase_b_agent
_K28Y_NATIVE=dict(_Y_CFG)
_K28Y_TARGET_PARENT=_y_target
_K28Y_CONTEXT={'day':0,'sheep':0}
_K28Y_REPORT=dict(eligible_turns=0,swaps=0,pick=0,place=0,harvest=0,boost=0,errors=0)
_MP_REPORT=_K28Y_REPORT

def _y_target(kind,shops,cfg):
    day=_K28Y_CONTEXT['day']
    if (kind=='COW' and _K28Y_GENOME['from']<=day<=_K28Y_GENOME['to'] and
        'YARN_STORE' in shops and not any(s in ('PIZZA_SHOP','ICE_CREAM_SHOP','SMOOTHIE_SHOP') for s in shops) and
        _K28Y_CONTEXT['sheep']<_K28Y_GENOME['target']):return 'SHEEP'
    if _K28Y_NATIVE['days'][0]<=day<=_K28Y_NATIVE['days'][1]:
        return _K28Y_TARGET_PARENT(kind,shops,_K28Y_NATIVE)
    return None

def early_herd_overlay_agent(observation,configuration=None):
    global _Y_CFG
    step=int(observation['step']);seat=int(observation['player']);day=step//24
    if step==0:
        for key in _K28Y_REPORT:_K28Y_REPORT[key]=0
        for key in _Y_REPORT:_Y_REPORT[key]=0
    farm=observation['farms'][seat];private=observation['private']
    sheep=sum(isinstance(t,dict) and t.get('animal')=='SHEEP' for row in farm['tiles'] for t in row)
    sheep+=int(private['shed'].get('SHEEP',0))+sum(int(v.get('SHEEP',0)) for v in private['inventories'])
    _K28Y_CONTEXT.update(day=day,sheep=sheep)
    shops=observation['town']['unlocked_shops']
    eligible=(_K28Y_GENOME['from']<=day<=_K28Y_GENOME['to'] and 'YARN_STORE' in shops and
              not any(s in ('PIZZA_SHOP','ICE_CREAM_SHOP','SMOOTHIE_SHOP') for s in shops) and
              sheep<_K28Y_GENOME['target'])
    _Y_CFG=dict(_K28Y_NATIVE)
    if eligible:
        _Y_CFG['days']=(min(_K28Y_NATIVE['days'][0],_K28Y_GENOME['from']),max(_K28Y_NATIVE['days'][1],_K28Y_GENOME['to']))
        _Y_CFG['margin']=_K28Y_GENOME['reserve']
        _K28Y_REPORT['eligible_turns']+=1
    action=_K28Y_PARENT(observation,configuration)
    for key in ('swaps','pick','place','harvest','boost','errors'):_K28Y_REPORT[key]=_Y_REPORT.get(key,0)
    return action
early_herd_overlay_agent.telemetry=_K28Y_REPORT
agent=early_herd_overlay_agent
kaggle_submission_agent=early_herd_overlay_agent
