
# Outcome-supervised, bounded game decisions. Runtime uses only current player view.
_VP_PARENT=phase_b_agent
_VP_STATE={}
_VP_REPORT={'predictions':0,'changes':0,'units':0,'errors':0}
def _vp_predict(features):
    total=0.0;prob=0.0
    for tree in _VP_FOREST:
        i=0
        while tree[i][0]>=0:
            n=tree[i];i=n[2] if features[n[0]]<=n[1] else n[3]
        total+=tree[i][4];prob+=tree[i][5]
    return total/len(_VP_FOREST),prob/len(_VP_FOREST)

def _vp_propose(obs,action,context,state):
    step=int(obs['step'])
    if step-state['last_changed']<_VP_COOLDOWN:return action
    if any(o and o[0]!='SELL' for o in action.get('market',[])):return action
    best=None
    for item,quantities,original,spare in vp_options(obs,action,context['projected']):
        features=_mbc_features(context,item)
        for quantity in quantities:
            value,prob=_vp_predict(features+[float(quantity),float(original),float(spare)])
            _VP_REPORT['predictions']+=1
            if value>=_VP_MIN_GAIN and prob>=_VP_PROB:
                row=(value,-quantity,item,quantity)
                if best is None or row>best:best=row
    if best is None or _VP_SHADOW:return action
    result=vp_action(action,best[2],best[3]);state['last_changed']=step
    _VP_REPORT['changes']+=1;_VP_REPORT['units']+=best[3]
    return result

def value_agent(observation,configuration=None):
    step=int(observation['step']);seat=int(observation['player'])
    state=_VP_STATE.get(seat)
    if state is None or step<=state['history'].get('last_step',-1):
        state=_VP_STATE[seat]={'last_changed':-1000,'history':{}}
        if step==0:
            for k in _VP_REPORT:_VP_REPORT[k]=0
    action=_VP_PARENT(observation,configuration)
    try:
        projected=projected_shed(action,FarmView(observation))
        context=_mbc_context(observation,action,projected,state['history'])
        result=_vp_propose(observation,action,context,state)
        if result is not action:vp_sync(observation,result,globals())
        sold={item:min(max(0,int(projected.get(item,0))),sum(max(0,int(o[2])) for o in result.get('market',[])[:10] if len(o)>=3 and o[:2]==['SELL',item])) for item in VP_ITEMS}
        _mbc_commit(context,sold)
        return result
    except Exception:
        _VP_REPORT['errors']+=1
        return action
value_agent.telemetry=_VP_REPORT
agent=value_agent
kaggle_submission_agent=value_agent
_MP_REPORT=_VP_REPORT
