_RB_PARENT=[v for v in list(globals().values()) if callable(v)][-1]
_RB_REPORT={'opening_changed':0,'errors':0}
_MP_REPORT=_RB_REPORT

def rebased_opening_agent(observation,configuration=None):
    step=int(observation['step'])
    if step==0:
        for key in _RB_REPORT:_RB_REPORT[key]=0
    action=_RB_PARENT(observation,configuration)
    if step==0:
        orders=action.get('market',[])
        if orders[:2]==[['BUY_PRODUCT','WHEAT',20],['SELL','WHEAT',15]]:
            prefix=[['BUY_PRODUCT','WHEAT',_RB_BUY]]
            if _RB_BUY>5:prefix.append(['SELL','WHEAT',_RB_BUY-5])
            action=dict(action,market=prefix+orders[2:])
            _RB_REPORT['opening_changed']=1
    return action
rebased_opening_agent.telemetry=_RB_REPORT
