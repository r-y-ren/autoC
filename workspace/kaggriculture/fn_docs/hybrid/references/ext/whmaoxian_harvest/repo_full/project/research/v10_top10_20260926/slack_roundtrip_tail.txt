
# Side trips fit wholly inside known PASS runs and restore original coordinates.
_SD_PARENT=phase_b_agent
_SD_STATE={}
_SD_REPORT={'trips':0,'care':0,'fertilizer':0,'harvest':0,'contract_errors':0,'errors':0}
_SD_MOVES={'NORTH':(0,-1),'SOUTH':(0,1),'EAST':(1,0),'WEST':(-1,0)}
def _sd_walk(start,end):
    x,y=start;tx,ty=end;out=[]
    while x!=tx:
        out.append(['EAST' if x<tx else 'WEST']);x+=1 if x<tx else -1
    while y!=ty:
        out.append(['SOUTH' if y<ty else 'NORTH']);y+=1 if y<ty else -1
    return out

def _sd_reservations(obs,action,native):
    step=int(obs['step']);farm=obs['farms'][int(obs['player'])]
    positions=[list(farm['farmer'])]+[list(p) for p in farm['hands']]
    reserved=set();deadlines={i:min(719,(step//24+1)*24) for i in range(len(positions))}
    for t in range(step,min(719,(step//24+1)*24)):
        if t==step:a=action
        else:a=_IMPL.chassis.routes[2 if t>=648 else native['route']][t]
        commands=[a.get('farmer') or ['PASS']]+list(a.get('hands') or [])
        for i,cmd in enumerate(commands[:len(positions)]):
            if not cmd or cmd==['PASS']:continue
            if t>step:deadlines[i]=min(deadlines[i],t)
            if cmd[0] in _SD_MOVES:
                dx,dy=_SD_MOVES[cmd[0]];positions[i]=[max(0,min(9,positions[i][0]+dx)),max(0,min(9,positions[i][1]+dy))]
            elif cmd[0] in ('CARE','HARVEST','COLLECT_FERTILIZER'):
                reserved.add((tuple(positions[i]),cmd[0]))
    return reserved,deadlines

def slack_roundtrip_agent(observation,configuration=None):
    step=int(observation['step']);seat=int(observation['player']);day=step//24
    state=_SD_STATE.get(seat)
    if state is None or step<=state['last']:
        state=_SD_STATE[seat]={'last':-1,'active':{}}
        if step==0:
            for key in _SD_REPORT:_SD_REPORT[key]=0
    state['last']=step;farm=observation['farms'][seat]
    positions=[farm['farmer']]+list(farm['hands']);active=state['active']
    virtual=copy.deepcopy(observation) if active else observation
    for actor,trip in list(active.items()):
        if actor>=len(positions) or tuple(positions[actor])!=tuple(trip['expected']):
            _SD_REPORT['contract_errors']+=1;active.pop(actor,None);continue
        if not trip['commands']:
            if tuple(positions[actor])!=tuple(trip['anchor']):_SD_REPORT['contract_errors']+=1
            active.pop(actor,None);continue
        if actor==0:virtual['farms'][seat]['farmer']=list(trip['anchor'])
        else:virtual['farms'][seat]['hands'][actor-1]=list(trip['anchor'])
    action=_SD_PARENT(virtual,configuration)
    commands=[action.get('farmer') or ['PASS']]+list(action.get('hands') or [])
    commands+=[['PASS'] for _ in range(len(positions)-len(commands))]
    native=_IMPL.chassis.players.get(seat);selected=set()
    if native and _SD_FROM<=day<29 and step%24>=8 and not _V219_STATES.get(seat,{}).get('committed'):
        reserved,deadlines=_sd_reservations(virtual,action,native)
        stock_total=sum(observation['private']['shed'].values())+sum(sum(b.values()) for b in observation['private']['inventories'])
        for actor,pos in enumerate(positions):
            if actor in active or commands[actor]!=['PASS']:continue
            if _SD_FARMER_ONLY and actor!=0:continue
            slack=deadlines[actor]-step;best=None
            for y,row in enumerate(farm['tiles']):
                for x,tile in enumerate(row):
                    if not isinstance(tile,dict) or tile.get('animal') not in _R9S_PRODUCT:continue
                    site=(x,y);distance=abs(pos[0]-x)+abs(pos[1]-y)
                    if site in selected or distance>_SD_DISTANCE or 2*distance+1>slack:continue
                    kind=tile['animal'];product=_R9S_PRODUCT[kind];tasks=[]
                    first=int(tile['placed_day'])+_R9S_FIRST[kind]
                    next_yield=first if day<first else day+((-day+first)%_R9S_PERIOD[kind])
                    held=int(tile.get('yield_units',0))+int(tile.get('pending_care_bonus',0))
                    if tile.get('fed_today') and not tile.get('cared_today') and next_yield<30 and held<5:
                        tasks.append(('CARE',float(observation['market']['prices'][product]),0))
                    if _SD_COLLECT and tile.get('fertilizer_available') and stock_total<85:
                        tasks.append(('COLLECT_FERTILIZER',float(observation['market']['prices']['FERTILIZER']),1))
                    qty=int(tile.get('yield_units',0))
                    if _SD_HARVEST and qty>0 and stock_total+qty<85:
                        tasks.append(('HARVEST',qty*.3*float(observation['market']['prices'][product]),qty))
                    for op,value,quantity in tasks:
                        if (site,op) in reserved or value<5:continue
                        score=value/(2*distance+1)
                        if best is None or score>best[0]:best=(score,site,op,quantity)
            if best:
                _,site,op,qty=best;selected.add(site);stock_total+=qty
                queue=_sd_walk(pos,site)+[[op]]+_sd_walk(site,pos)
                active[actor]={'anchor':list(pos),'expected':list(pos),'commands':queue,'site':site,'deadline':deadlines[actor]}
                _SD_REPORT['trips']+=1
    for actor,trip in active.items():
        if not trip['commands']:continue
        if commands[actor]!=['PASS']:_SD_REPORT['contract_errors']+=1
        command=trip['commands'].pop(0);commands[actor]=command
        if command[0] in _SD_MOVES:
            dx,dy=_SD_MOVES[command[0]];trip['expected']=[positions[actor][0]+dx,positions[actor][1]+dy]
        else:
            trip['expected']=list(positions[actor])
            key={'CARE':'care','COLLECT_FERTILIZER':'fertilizer','HARVEST':'harvest'}.get(command[0])
            if key:_SD_REPORT[key]+=1
    return dict(action,farmer=commands[0],hands=commands[1:])
slack_roundtrip_agent.telemetry=_SD_REPORT
agent=slack_roundtrip_agent
kaggle_submission_agent=slack_roundtrip_agent
