
# Sell-only supply competition ablation. Keep production, inputs and order positions.
_LF_PARENT=phase_b_agent
_LF_ITEMS=('MILK','WOOL','STRAWBERRY','EGG','MELON','CARROT','TOMATO')
_LF_ANIMALS={'COW':'MILK','SHEEP':'WOOL','GOOSE':'EGG'}
_LF_REPORT={'changed_turns':0,'extra_units':0,'gated_items':0,'errors':0}
_MP_REPORT=_LF_REPORT

def supply_competition_agent(observation,configuration=None):
    step=int(observation['step']);seat=int(observation['player'])
    if step==0:
        for k in _LF_REPORT:_LF_REPORT[k]=0
    action=_LF_PARENT(observation,configuration)
    if not 144<=step<712:return action
    if _LF_MODE=='similar' and _r37_similarity(observation)<.85:return action
    orders=[list(o) for o in action.get('market',[])];view=FarmView(observation)
    stock=projected_shed(action,view);counts=[]
    for farm in observation['farms']:
        c={}
        for row in farm['tiles']:
            for t in row:
                if isinstance(t,dict):
                    item=t.get('crop') or _LF_ANIMALS.get(t.get('animal'))
                    if item:c[item]=c.get(item,0)+1
        counts.append(c)
    commands=[action.get('farmer') or ['PASS']]+list(action.get('hands',[]))
    blocked={c[1] for c in commands if len(c)>=2 and c[0]=='PICKUP'}
    blocked.update(o[1] for o in orders if len(o)>=2 and o[0]=='BUY_PRODUCT')
    extra=0
    for item in _LF_ITEMS:
        if item in blocked or int(view.prices.get(item,0))<=1:continue
        if _LF_MODE=='supply' and counts[1-seat].get(item,0)<_LF_RATIO*max(1,counts[seat].get(item,0)):
            _LF_REPORT['gated_items']+=1;continue
        original=sum(max(0,int(o[2])) for o in orders if len(o)>=3 and o[:2]==['SELL',item])
        amount=max(0,int(stock.get(item,0))-original)
        if not amount:continue
        hit=next((o for o in orders if len(o)>=3 and o[:2]==['SELL',item]),None)
        if hit is not None:hit[2]=int(hit[2])+amount
        elif len(orders)<10:orders.append(['SELL',item,amount])
        else:continue
        extra+=amount
    if not extra:return action
    result=dict(action,market=orders)
    previous=_OR2_STATE.get(seat,{}).get('prev')
    if previous and previous.get('step')==step:
        for item in _LF_ITEMS:
            requested=sum(max(0,int(o[2])) for o in orders if len(o)>=3 and o[:2]==['SELL',item])
            previous.setdefault('own',{})[item]=min(max(0,int(stock.get(item,0))),requested)
    race=_RACE_STATE.get(seat)
    if race and race.get('step')==step and race.get('prev_action') is not None:race['prev_action']=result
    _LF_REPORT['changed_turns']+=1;_LF_REPORT['extra_units']+=extra
    return result
supply_competition_agent.telemetry=_LF_REPORT
agent=supply_competition_agent
kaggle_submission_agent=supply_competition_agent
