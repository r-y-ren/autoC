
# Scenario valuation only: preserve the parent policy's globals after each estimate.
def _k28v_schedule(animal,placed_day,day_from):
    spec=_HD2_SPEC[animal];cap={'GOOSE':4,'COW':6,'SHEEP':6}[animal];out={}
    for day in range(max(day_from,placed_day+spec['first']),30):
        age=day-placed_day
        if (age-spec['first'])%spec['interval']:continue
        care_days=spec['first']-1 if age==spec['first'] else spec['interval']
        out[day]=min(float(cap),1.0+care_days*_HD2_CARE)
    return out

def _k28v_gain(k,obs,st):
    global _HD2_FUTURE,_hd2_schedule
    saved_future,saved_schedule=_HD2_FUTURE,_hd2_schedule
    values=[]
    try:
        if _K28V_CORRECT_FIRST:_hd2_schedule=_k28v_schedule
        for scenario in _K28V_FUTURES:
            _HD2_FUTURE=float(scenario)
            values.append(_hd2_ev('SHEEP',k,obs,st)[0]-_hd2_ev('COW',k,obs,st)[0])
    finally:
        _HD2_FUTURE=saved_future;_hd2_schedule=saved_schedule
    return min(values) if _K28V_ROBUST else sum(values)/len(values)
