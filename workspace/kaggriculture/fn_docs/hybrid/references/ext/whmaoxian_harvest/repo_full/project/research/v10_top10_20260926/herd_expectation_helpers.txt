
# Enumerate unknown future shop outcomes from the documented uniform draw rule.
# These are probability scenarios, not future observations or seed lookup.
import itertools as _he_itertools
import math as _he_math
_HE_GRID_CACHE={}
_HE_REPORT={'evaluations':0,'scenario_paths':0,'errors':0}
def _he_scenarios(item,day,shops):
    key=(item,day,len(shops))
    if key in _HE_GRID_CACHE:return _HE_GRID_CACHE[key]
    dates=[d for d in range(day+1,30) if d%3==0 and d<=24][:max(0,8-len(shops))]
    matches=[p for p in _HD2_SHOP_TYPES.values() if item in p]
    probability=len(matches)/len(_HD2_SHOP_TYPES)
    multipliers={2 if len(p)==1 else 1 for p in matches};assert len(multipliers)==1
    multiplier=next(iter(multipliers));result=[]
    for bits in _he_itertools.product((0,1),repeat=len(dates)):
        count=sum(bits);weight=probability**count*(1-probability)**(len(dates)-count)
        result.append((dict(zip(dates,(b*multiplier for b in bits))),weight))
    _HE_GRID_CACHE[key]=result
    return result

def _he_trade(inventory,mine,theirs,price):
    revenue_me=0;revenue_other=0
    for j in range(max(mine,theirs)):
        quote=price(inventory)
        if j<mine:revenue_me+=quote
        if j<theirs:revenue_other+=quote
        if quote>1:inventory+=int(j<mine)+int(j<theirs)
    return revenue_me,revenue_other,inventory
