
# Replace a proven current no-op at the shed, without moving a worker.
_K29P_PARENT=phase_b_agent
_K29P_PRODUCTS=('CARROT','TOMATO','STRAWBERRY','MELON','EGG','MILK','WOOL')
_K29P_REPORT={'changes':0,'units_deposited':0,'extra_sales':0,'physical_rejections':0}
_MP_REPORT=_K29P_REPORT

def _k29p_sim(obs,action):
    seat=int(obs['player']);farm=copy.deepcopy(obs['farms'][seat]);private=copy.deepcopy(obs['private'])
    peak=sum(private['shed'].values());step=int(obs['step'])
    for actor,c in enumerate(([action.get('farmer') or ['PASS']]+list(action.get('hands') or []))[:len(farm['hands'])+1]):
        _UNIT_NS['_apply_unit_action'](farm,private,actor,c,len(farm['tiles']),step//24,24,100)
        peak=max(peak,sum(private['shed'].values()))
    return farm,private,peak

def noop_deposit_agent(observation,configuration=None):
    step=int(observation['step']);seat=int(observation['player'])
    if step==0:
        for key in _K29P_REPORT:_K29P_REPORT[key]=0
    action=_K29P_PARENT(observation,configuration)
    if step<144 or step>=696 or step%24<_K29P_CFG['hour']:return action
    farm=observation['farms'][seat];private=observation['private'];positions=[farm['farmer']]+farm['hands']
    possible=[a for a,p in enumerate(positions) if tuple(p) in ((4,4),(5,4),(4,5),(5,5)) and any(private['inventories'][a].get(i,0)>0 for i in _K29P_PRODUCTS)]
    if not possible:return action
    base_farm,base_private,peak=_k29p_sim(observation,action)
    if peak>=100:return action
    local_farm=copy.deepcopy(farm);local_private=copy.deepcopy(private);extra_load=0;changed=0
    units=[list(action.get('farmer') or ['PASS'])]+[list(c) for c in action.get('hands',[])]
    units+=[['PASS'] for _ in range(max(0,len(positions)-len(units)))]
    for actor,command in enumerate(units[:len(positions)]):
        before=copy.deepcopy((local_farm,local_private)) if actor in possible else None
        _UNIT_NS['_apply_unit_action'](local_farm,local_private,actor,command,len(farm['tiles']),step//24,24,100)
        if before is None or (local_farm,local_private)!=before:continue
        room=max(0,100-peak-extra_load)
        bag=local_private['inventories'][actor]
        items=[i for i in _K29P_PRODUCTS if bag.get(i,0)>0]
        if not room or not items:continue
        item=max(items,key=lambda i:observation['market']['prices'][i]*min(room,bag[i]))
        quantity=min(room,bag[item]);replacement=['PLACE',item,quantity]
        _UNIT_NS['_apply_unit_action'](local_farm,local_private,actor,replacement,len(farm['tiles']),step//24,24,100)
        units[actor]=replacement;extra_load+=quantity;changed+=1
    if not changed:return action
    def totals(state):
        out=dict(state['shed'])
        for bag in state['inventories']:
            for item,n in bag.items():out[item]=out.get(item,0)+n
        return out
    before=totals(base_private);after=totals(local_private)
    if local_farm!=base_farm or local_private['seeds']!=base_private['seeds'] or any(after.get(i,0)<n for i,n in before.items()):
        _K29P_REPORT['physical_rejections']+=1;return action
    market=[list(o) for o in action.get('market',[])];stock=local_private['shed'];extra={}
    if _K29P_CFG['sell']:
        for item in sorted(_K29P_PRODUCTS,key=lambda i:-observation['market']['prices'][i]):
            increase=max(0,stock.get(item,0)-base_private['shed'].get(item,0))
            planned=sum(max(0,int(o[2])) for o in market if len(o)>=3 and o[:2]==['SELL',item])
            quantity=min(increase,max(0,stock.get(item,0)-planned))
            if quantity<=0 or observation['market']['prices'][item]<=1:continue
            hit=next((o for o in market if len(o)>=3 and o[:2]==['SELL',item]),None)
            if hit is not None:quantity=min(quantity,max(0,100-int(hit[2])));hit[2]+=quantity
            elif len(market)<10:market.append(['SELL',item,quantity])
            else:quantity=0
            if quantity:extra[item]=quantity
    native=_IMPL.chassis.players.get(seat)
    if native and _K29P_CFG['book']:
        debts=native['sell_state'].setdefault('r36_debts',{})
        for item,quantity in extra.items():
            for t in range(step+1,min(719,step+49)):
                planned=sum(min(100,max(0,int(o[2]))) for o in _adv_future(seat,t) if len(o)>=3 and o[:2]==['SELL',item])
                old=debts.get(t,{}).get(item,0);take=min(quantity,max(0,planned-old))
                if take:debts.setdefault(t,{})[item]=old+take;quantity-=take
                if quantity<=0:break
    previous=_OR2_STATE.get(seat,{}).get('prev')
    if previous and previous.get('step')==step:
        for item in _ADV_ITEMS:
            requested=sum(min(100,max(0,int(o[2]))) for o in market if len(o)>=3 and o[:2]==['SELL',item])
            previous.setdefault('own',{})[item]=min(stock.get(item,0),requested)
    _K29P_REPORT['changes']+=changed;_K29P_REPORT['units_deposited']+=extra_load
    _K29P_REPORT['extra_sales']+=sum(extra.values())
    return dict(action,farmer=units[0],hands=units[1:],market=market)
noop_deposit_agent.telemetry=_K29P_REPORT
agent=noop_deposit_agent
kaggle_submission_agent=noop_deposit_agent
