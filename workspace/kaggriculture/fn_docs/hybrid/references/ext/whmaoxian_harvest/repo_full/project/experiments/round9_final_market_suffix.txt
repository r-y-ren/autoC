_R9M_PARENT = round8_production_fusion_agent
_R9M_REPORT = {'changes':0,'errors':0}
def round9_final_market_agent(observation,configuration=None):
    if int(observation['step'])==0:
        for key in _R9M_REPORT:_R9M_REPORT[key]=0
    action=_R9M_PARENT(observation,configuration)
    if int(observation['step'])>=216 and _ig_standard(configuration):
        result=_v44y_reorder(observation,action)
        _R9M_REPORT['changes']+=int(result!=action)
        return result
    return action
round9_final_market_agent.telemetry=_R9M_REPORT
