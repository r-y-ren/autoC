
# Kaggriculture: remove obsolete queued farming tasks without moving workers.
_WQ_PARENT=phase_b_agent
_WQ_ORIGINAL=Chassis._weed_repair
_WQ_REPORT={'removed':0,'deduplicated':0,'calls':0,'errors':0}
def _wq_obsolete(command,tile,inventory,day):
    if not command:return True
    op=command[0];kind=tile.get('kind') if isinstance(tile,dict) else None
    if op=='DIG':return tile is None or kind=='PLANT' or isinstance(tile,dict) and bool(tile.get('animal'))
    if op=='PLANT':return tile is not None and kind!='WEED'
    if op in ('BUILD_COOP','BUILD_PASTURE'):return tile is not None and kind!='WEED'
    if op=='WATER':return kind!='PLANT' or bool(tile.get('watered_today'))
    if op=='CARE':return not isinstance(tile,dict) or not tile.get('animal') or bool(tile.get('cared_today'))
    if op=='HARVEST':return not isinstance(tile,dict) or int(tile.get('yield_units',0))<=0
    if op=='FERTILIZE':return kind!='PLANT' or int(tile.get('fertilized_until_day',-1))>=day+2
    if op=='PLACE' and len(command)>1:return int(inventory.get(command[1],0))<=0
    return False

def _wq_repair(self,action,view,state,route,step):
    if step>=648 or _WQ_MODE=='control':return _WQ_ORIGINAL(self,action,view,state,route,step)
    _WQ_REPORT['calls']+=1
    for actor,queue in list(state['pending'].items()):
        if actor>=len(view.positions):continue
        pos=tuple(view.positions[actor]);tile=_tile_at(view.tiles,pos);seen=set();keep=[]
        for target,command in queue:
            key=(tuple(target),tuple(command))
            obsolete=tuple(target)!=pos or _wq_obsolete(command,tile,view.inv(actor),step//24)
            if _WQ_MODE=='dig_only':obsolete=bool(command) and command[0]=='DIG' and obsolete
            if _WQ_MODE=='dedup':obsolete=False
            if _WQ_MODE=='chains':obsolete=obsolete or not command or command[0] not in ('PLANT','WATER','BUILD_COOP','BUILD_PASTURE','PLACE')
            if _WQ_MODE=='none':obsolete=True
            if obsolete:
                _WQ_REPORT['removed']+=1;continue
            if key in seen:
                _WQ_REPORT['deduplicated']+=1;continue
            seen.add(key);keep.append((target,command))
        if keep:state['pending'][actor]=keep
        else:state['pending'].pop(actor,None)
    return _WQ_ORIGINAL(self,action,view,state,route,step)
Chassis._weed_repair=_wq_repair

def weed_queue_agent(observation,configuration=None):
    if int(observation['step'])==0:
        for key in _WQ_REPORT:_WQ_REPORT[key]=0
    return _WQ_PARENT(observation,configuration)
_MP_REPORT=_WQ_REPORT
weed_queue_agent.telemetry=_WQ_REPORT
agent=weed_queue_agent
kaggle_submission_agent=weed_queue_agent
