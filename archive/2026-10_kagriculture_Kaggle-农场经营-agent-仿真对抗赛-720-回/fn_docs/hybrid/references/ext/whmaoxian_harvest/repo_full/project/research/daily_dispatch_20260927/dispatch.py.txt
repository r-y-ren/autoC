
# Observation-driven daily dispatcher. The original agent owns the opening only.
_DP_PARENT=phase_b_agent
_DP_STATE={}
_DP_REPORT={'turns':0,'assignments':0,'feed':0,'water':0,'harvest':0,'drop':0,'plant':0,'idle':0}
_DP_HOME=((4,4),(5,4),(4,5),(5,5))
_DP_CROPS={'WHEAT':(2,4,6,10),'CARROT':(2,3,4,20),'TOMATO':(8,8,4,50),'STRAWBERRY':(10,10,4,100),'MELON':(10,12,6,80)}
_DP_ANIMAL={'GOOSE':('EGG',4,1,4),'COW':('MILK',8,2,6),'SHEEP':('WOOL',6,3,6)}
def _dp_dist(a,b):return abs(a[0]-b[0])+abs(a[1]-b[1])
def _dp_home(p):return min(_DP_HOME,key=lambda q:(_dp_dist(p,q),q))
def _dp_walk(p,q):
    if p[0]!=q[0]:return ['EAST' if p[0]<q[0] else 'WEST']
    if p[1]!=q[1]:return ['SOUTH' if p[1]<q[1] else 'NORTH']
    return ['PASS']
def _dp_crop(day,prices,grain,animals):
    if day>27:return None
    options=[]
    for crop in ('WHEAT','CARROT'):
        first,last,cap,cost=_DP_CROPS[crop]
        if day+last>29:continue
        price=prices[crop]*(1.3 if crop=='WHEAT' and grain<2*animals else 1)
        options.append(((cap*price-cost)/(last+1),crop))
    return max(options)[1] if options else None

def _dp_bundle(tile,day,hour,prices,crop):
    ops=[];value=0.0;need={}
    if tile=='LOCKED':return ops,value,need
    if tile is None or isinstance(tile,dict) and tile.get('kind')=='WEED':
        if crop is None:return ops,value,need
        if tile is not None:ops.append(['DIG'])
        ops.append(['PLANT',crop]);need['seed:'+crop]=1
        first,last,cap,cost=_DP_CROPS[crop]
        value=_DP_PLANT_WEIGHT*max(0,cap*prices[crop]-cost)
        return ops,value,need
    if not isinstance(tile,dict):return ops,value,need
    if 'animal' in tile:
        item,first,interval,cap=_DP_ANIMAL[tile['animal']]
        held=int(tile.get('yield_units',0));age=day-int(tile['placed_day'])
        if held>0:ops.append(['HARVEST']);value+=held*prices[item]
        if day<29 and not tile.get('fed_today'):
            ops.append(['FEED']);need['WHEAT']=1
            value+=_DP_FEED_WEIGHT*(120+600*int(tile.get('consecutive_unfed',0)>0))
        if day<29 and age<30 and not tile.get('cared_today'):
            ops.append(['CARE']);value+=prices[item]/interval
        if tile.get('fertilizer_available'):
            ops.append(['COLLECT_FERTILIZER']);value+=prices['FERTILIZER']
        return ops,value,need
    if tile.get('kind')!='PLANT':return ops,value,need
    item=tile['crop'];first,last,cap,cost=_DP_CROPS[item]
    age=day-int(tile['planted_day']);held=max(0,int(tile.get('yield_units',0)))
    ongoing=item in ('TOMATO','STRAWBERRY');interval=1 if item=='TOMATO' else 2
    remaining=max(0,719-(day*24+hour));expiry=int(tile.get('max_lifespan_step',-1))
    ripe=age>=first and held>0
    harvest=ripe and (ongoing or age>=last or held>=cap or day==29 or 0<=expiry<=day*24+hour+6)
    produces=ongoing and age+1>=first and (age+1-first)%interval==0 and (age+1-first)//interval<4
    grows=not ongoing and (last+1)//2<=age<=last and held<cap
    water=not tile.get('watered_today') and day<29 and (tile.get('consecutive_unwatered',0)>0 or grows or produces)
    if day==29 and grows and harvest and not tile.get('watered_today'):water=True
    exhausted=ongoing and expiry>=0 and held==0
    if exhausted:
        if crop is not None:
            ops=[['DIG'],['PLANT',crop]];need['seed:'+crop]=1
            value=_DP_PLANT_WEIGHT*max(0,_DP_CROPS[crop][2]*prices[crop]-_DP_CROPS[crop][3])
        return ops,value,need
    if water:
        if (grows or produces) and tile.get('fertilized_until_day',-1)<day and prices[item]>1.1*prices['FERTILIZER']:
            ops.append(['FERTILIZE']);need['FERTILIZER']=1
        ops.append(['WATER'])
        value+=(prices[item] if grows or produces else 20)
        if tile.get('consecutive_unwatered',0)>0:value+=200+held*prices[item]
    if harvest:
        ops.append(['HARVEST']);value+=held*prices[item]
    return ops,value,need

def _dp_target(obs,farm,private,actor,claimed,crop):
    day=int(obs['step'])//24;hour=int(obs['step'])%24;prices=obs['market']['prices']
    pos=([farm['farmer']]+farm['hands'])[actor];bag=private['inventories'][actor]
    time_left=719-int(obs['step']) if day==29 else 24-hour
    best=None
    for y,row in enumerate(farm['tiles']):
        for x,tile in enumerate(row):
            q=(x,y)
            if q in claimed:continue
            ops,value,needs=_dp_bundle(tile,day,hour,prices,crop)
            if not ops or value<=0:continue
            distance=_dp_dist(pos,q);supply=0
            if any(needs.get(i,0)>bag.get(i,0) for i in ('WHEAT','FERTILIZER')):
                home=_dp_home(pos);supply=_dp_dist(pos,home)+_dp_dist(home,q)-distance+1
            cost=distance+len(ops)+supply
            if day==29:cost+=_dp_dist(q,_dp_home(q))+1
            if cost>time_left:continue
            score=value/(max(1,cost)**_DP_DISTANCE_POWER)
            if distance==0:score*=1.1
            key=(score,-cost,-y,-x)
            if best is None or key>best[0]:best=(key,q)
    return None if best is None else best[1]

def _dp_command(obs,farm,private,actor,state,claimed,crop,requests,total_load):
    step=int(obs['step']);day,hour=divmod(step,24)
    pos=tuple(([farm['farmer']]+farm['hands'])[actor]);bag=private['inventories'][actor]
    home=_dp_home(pos);premium=sum(bag.values())-min(2,bag.get('WHEAT',0))-min(1,bag.get('FERTILIZER',0))
    final_return=day==29 and sum(bag.values()) and 719-step<=_dp_dist(pos,home)+2
    delivery=(premium>=_DP_DELIVERY and hour<20) or (total_load>=85 and premium>0) or final_return
    if delivery or actor in state['returning']:
        state['goals'].pop(actor,None);state['returning'].add(actor)
        if pos!=home:return _dp_walk(pos,home)
        state['returning'].discard(actor)
        if sum(bag.values()):return ['DROP']
    target=state['goals'].get(actor)
    if target is not None:
        tile=farm['tiles'][target[1]][target[0]]
        ops,_,need=_dp_bundle(tile,day,hour,obs['market']['prices'],crop)
        if not ops:claimed.discard(target);state['goals'].pop(actor,None);target=None
    if target is None:
        target=_dp_target(obs,farm,private,actor,claimed,crop)
        if target is not None:state['goals'][actor]=target;claimed.add(target);_DP_REPORT['assignments']+=1
    if target is None:
        if premium>0 or day==29 and sum(bag.values()):
            state['returning'].add(actor)
            return ['DROP'] if pos==home else _dp_walk(pos,home)
        return ['PASS']
    tile=farm['tiles'][target[1]][target[0]]
    ops,_,need=_dp_bundle(tile,day,hour,obs['market']['prices'],crop)
    for item,quantity in need.items():
        if item.startswith('seed:'):requests[item]=requests.get(item,0)+quantity
    if not ops:return ['PASS']
    op=ops[0]
    missing=[i for i in ('WHEAT','FERTILIZER') if need.get(i,0)>bag.get(i,0)]
    # A useful on-tile operation may precede a supply trip.
    if pos==target and op[0] not in ('FEED','FERTILIZE','PLANT'):return op
    if missing:
        item=missing[0];requests[item]=requests.get(item,0)+2
        if pos!=home:return _dp_walk(pos,home)
        quantity=min(2,max(0,int(private['shed'].get(item,0))))
        return ['PICKUP',item,quantity] if quantity else ['PASS']
    if pos!=target:return _dp_walk(pos,target)
    if op[0]=='PLANT' and private['seeds'].get(op[1],0)<=0:return ['PASS']
    return op

def _dp_market(obs,action,requests,crop,unfed):
    farm=obs['farms'][int(obs['player'])];prices=obs['market']['prices'];day,hour=divmod(int(obs['step']),24)
    stock=projected_shed(action,FarmView(obs));orders=[];cash=float(farm['money'])
    grain=sum(b.get('WHEAT',0) for b in obs['private']['inventories'])
    reserve=max(0,unfed-grain)+2 if day<29 else 0
    fert_reserve=min(8,max(3,requests.get('FERTILIZER',0))) if day<29 else 0
    quantities={item:max(0,int(stock.get(item,0))-(reserve if item=='WHEAT' else fert_reserve if item=='FERTILIZER' else 0)) for item in PRODUCTS}
    sells=[['SELL',i,q] for i,q in quantities.items() if q>0]
    sells.sort(key=lambda o:(-prices[o[1]]*o[2],o[1]))
    orders.extend(sells[:3] if hour<2 else sells[:7])
    buys=[]
    for item,wanted in [('WHEAT',reserve),('FERTILIZER',min(8,requests.get('FERTILIZER',0)+2) if requests.get('FERTILIZER') else 0)]:
        amount=max(0,wanted-int(stock.get(item,0)))
        if amount>0:
            unit=int(prices[item])+10;amount=min(amount,int(max(0,cash-200)//max(1,unit)))
            if amount:buys.append(['BUY_PRODUCT',item,amount]);cash-=amount*unit
    if crop is not None:
        wanted=min(8,max(2,requests.get('seed:'+crop,0)))
        amount=max(0,wanted-int(obs['private']['seeds'].get(crop,0)))
        amount=min(amount,int(max(0,cash-200)//_DP_CROPS[crop][3]))
        if amount:buys.append(['BUY_SEED',crop,amount]);cash-=amount*_DP_CROPS[crop][3]
    orders.extend(buys[:max(0,10-len(orders))])
    count=len(farm['hands']);hires=int(farm['hires_today'])
    if hour<4:
        for index in range(count,_DP_WORKERS):
            cost=_v219_fib(hires)
            if len(orders)>=10 or cash<cost+100:break
            orders.append(['HIRE']);cash-=cost;hires+=1
    return orders[:10]

def daily_dispatch_agent(observation,configuration=None):
    step=int(observation['step']);seat=int(observation['player']);day,hour=divmod(step,24)
    if day<_DP_START:return _DP_PARENT(observation,configuration)
    state=_DP_STATE.get(seat)
    if state is None or step<=state['last']:
        state=_DP_STATE[seat]={'last':-1,'day':-1,'goals':{},'returning':set()}
        for key in _DP_REPORT:_DP_REPORT[key]=0
    if state['day']!=day:state.update(day=day,goals={},returning=set())
    state['last']=step;_DP_REPORT['turns']+=1
    farm=copy.deepcopy(observation['farms'][seat]);private=copy.deepcopy(observation['private'])
    animals=sum(isinstance(t,dict) and 'animal' in t for row in farm['tiles'] for t in row)
    unfed=sum(isinstance(t,dict) and 'animal' in t and not t.get('fed_today') for row in farm['tiles'] for t in row)
    grain=private['shed'].get('WHEAT',0)+sum(b.get('WHEAT',0) for b in private['inventories'])
    crop=_dp_crop(day,observation['market']['prices'],grain,animals)
    claimed=set(state['goals'].values());requests={};commands=[]
    for actor in range(len(farm['hands'])+1):
        load=sum(private['shed'].values())+sum(sum(b.values()) for b in private['inventories'])
        cmd=_dp_command(observation,farm,private,actor,state,claimed,crop,requests,load)
        if cmd[0]=='DROP':
            bag=private['inventories'][actor];room=max(0,100-sum(private['shed'].values()))
            if sum(bag.values())>room:
                candidates=[i for i in PRODUCTS if bag.get(i,0)>0]
                item=max(candidates,key=lambda i:observation['market']['prices'][i]) if candidates else None
                cmd=['PLACE',item,min(room,bag[item])] if room and item else ['PASS']
        commands.append(cmd)
        _UNIT_NS['_apply_unit_action'](farm,private,actor,cmd,10,day,24,100)
        key={'FEED':'feed','WATER':'water','HARVEST':'harvest','DROP':'drop','PLANT':'plant','PASS':'idle'}.get(cmd[0])
        if key:_DP_REPORT[key]+=1
    action={'farmer':commands[0],'hands':commands[1:],'market':[]}
    action['market']=_dp_market(observation,action,requests,crop,unfed)
    return action

daily_dispatch_agent.telemetry=_DP_REPORT
