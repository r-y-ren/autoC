
# Product-wise supply extension. Existing R2 decisions are retained.
_PP_OLD_PLAN = _v10_pressure_plan
_PP_REPORT = {'additional_plans': 0, 'eligible_turns': 0, 'errors': 0}
_PP_ANIMAL_PRODUCT = {'COW': 'MILK', 'SHEEP': 'WOOL', 'GOOSE': 'EGG'}
def _pp_counts(farm):
    counts = {}
    for row in farm['tiles']:
        for tile in row:
            if not isinstance(tile, dict):
                continue
            item = tile.get('crop') or _PP_ANIMAL_PRODUCT.get(tile.get('animal'))
            if item:
                counts[item] = counts.get(item, 0) + 1
    return counts

def _v10_pressure_plan(obs, plan, protected):
    step = int(obs['step']); seat = int(obs['player'])
    original = [(t, item, qty) for t, item, qty in plan if t-step <= _V10_ADV_HORIZON]
    result = _PP_OLD_PLAN(obs, original, protected)
    if step < 216 or _V10_ADV_HORIZON >= _PP_MAX_LOOK:
        return result
    own = _pp_counts(obs['farms'][seat])
    rival = _pp_counts(obs['farms'][1-seat])
    stored = _OR2_STATE.get(seat, {}).get('stock', {})
    shops = obs.get('town', {}).get('unlocked_shops', [])
    _PP_REPORT['eligible_turns'] += 1
    for t, item, quantity in plan:
        if t-step <= _V10_ADV_HORIZON or item not in _ADV_ITEMS:
            continue
        if rival.get(item, 0) < max(1, own.get(item, 0)*_PP_CAPACITY_RATIO):
            continue
        demand_per_tick = sum((2 if len(_OR2_SHOPS.get(s, ())) == 1 else 1)
                              for s in shops if item in _OR2_SHOPS.get(s, ()))
        demand = demand_per_tick * sum(x % 4 == 0 for x in range(step, t))
        demand += sum(x % 24 == 0 for x in range(step, t))
        if int(stored.get(item, 0)) < max(_PP_STOCK_MIN, 2*demand):
            continue
        result.append((t, item, quantity))
        _PP_REPORT['additional_plans'] += 1
    return result

_PP_ENTRY_PARENT = phase_b_agent
def product_pressure_agent(observation, configuration=None):
    if int(observation['step']) == 0:
        for key in _PP_REPORT:
            _PP_REPORT[key] = 0
    return _PP_ENTRY_PARENT(observation, configuration)
product_pressure_agent.telemetry = _PP_REPORT
agent = product_pressure_agent
kaggle_submission_agent = product_pressure_agent
