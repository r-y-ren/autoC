
# Alternative development arm: bounded conditional sale size, with optional holds.
def _mbc_advise(obs, action, context):
    cfg=_MBC_CFG;step=int(obs['step']);available=context['projected']
    if not cfg['start']<=step<cfg['end'] or step%24==23:
        return action
    if context['own']['money']<cfg['min_cash']:
        return action
    orders=[list(o) for o in action.get('market',[])];initial=_mbc_sold(action,available)
    if any(o and o[0]=='BUY_PRODUCT' for o in orders):return action
    picked={c[1] for c in [action.get('farmer',[])]+action.get('hands',[])
            if len(c)>1 and c[0]=='PICKUP'}
    can_hold=cfg.get('hold',False) and context['total']<60 and not context['non_sells']
    proposals=[]
    for item in _MBC_ITEMS:
        stock=max(0,int(available.get(item,0)))
        if not stock or item in picked:continue
        p,fraction=_mbc_predict(_mbc_features(context,item));_MBC_STATS['predictions']+=1
        indices=[i for i,o in enumerate(orders) if len(o)>=3 and o[:2]==['SELL',item]]
        if len(indices)>1 or int(obs['market']['prices'][item])<=1:continue
        original=initial[item];desired=original
        if p>=cfg['threshold']:
            estimate=fraction/max(.05,p) if cfg.get('conditional',False) else fraction
            target=min(stock,int(stock*estimate+.5))
            desired=min(original+cfg['max_extra'],target)
            if not can_hold:desired=max(original,desired)
        elif can_hold and p<=cfg['hold_threshold']:
            desired=0
        desired=max(original-cfg.get('max_hold',8),desired)
        desired=max(0,min(stock,desired))
        if desired!=original:proposals.append((p,item,original,desired,indices))
    _MBC_STATS['proposed']+=len(proposals)
    if cfg['mode']=='shadow':return action
    for p,item,original,desired,indices in sorted(proposals,reverse=True):
        if indices:orders[indices[0]]=['SELL',item,desired] if desired else []
        elif desired and len(orders)<10:orders.append(['SELL',item,desired])
        else:continue
        _MBC_STATS['extra']+=max(0,desired-original)
        _MBC_STATS['held']+=max(0,original-desired)
    if orders==action.get('market',[]):return action
    _MBC_STATS['changed']+=1
    return dict(action,market=orders)
