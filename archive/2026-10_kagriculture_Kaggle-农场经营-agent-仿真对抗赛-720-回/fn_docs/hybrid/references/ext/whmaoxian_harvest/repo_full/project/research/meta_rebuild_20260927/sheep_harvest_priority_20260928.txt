
# Research-only service priority on confirmed cow-to-sheep substitutions.
_SH29_PARENT=agent
_SH29_STATE={}
_SH29_REPORT={'extra_harvests':0,'extra_units':0,'extra_sale_requests':0,'fertilizer_skipped':0}
_MP_REPORT=_SH29_REPORT

def sheep_harvest_priority_agent(observation,configuration=None):
    step=int(observation['step']);seat=int(observation['player'])
    if step==0:
        _SH29_STATE.clear()
        for key in _SH29_REPORT:_SH29_REPORT[key]=0
    action=_SH29_PARENT(observation,configuration)
    state=_SH29_STATE.setdefault(seat,{'credit':0})
    farm=observation['farms'][seat];sites=_CS_STATE.get(seat,{}).get('sites',{})
    commands=[list(action.get('farmer') or ['PASS'])]+[list(c) for c in action.get('hands',[])]
    positions=[farm['farmer']]+list(farm['hands']);claimed=set()
    wool_price=float(observation['market']['prices']['WOOL'])
    fert_price=float(observation['market']['prices']['FERTILIZER'])
    if 240<=step<648:
        for actor,pos in enumerate(positions[:len(commands)]):
            xy=tuple(pos);tile=farm['tiles'][pos[1]][pos[0]];cmd=commands[actor]
            if xy not in sites or xy in claimed or not isinstance(tile,dict) or tile.get('animal')!='SHEEP':continue
            units=int(tile.get('yield_units',0))
            if cmd==['HARVEST'] and units>0:claimed.add(xy);continue
            if cmd!=['COLLECT_FERTILIZER'] or units<2:continue
            opportunity=fert_price if tile.get('fertilizer_available') else 0
            if units*wool_price<_SH29_RATIO*max(opportunity,1):continue
            commands[actor]=['HARVEST'];claimed.add(xy);state['credit']+=units
            _SH29_REPORT['extra_harvests']+=1;_SH29_REPORT['extra_units']+=units
            _SH29_REPORT['fertilizer_skipped']+=int(bool(tile.get('fertilizer_available')))
    result=dict(action,farmer=commands[0],hands=commands[1:])
    market=[list(order) for order in result.get('market',[])]
    if state['credit']>0:
        stock=projected_shed(result,FarmView(observation))
        planned=sum(max(0,int(o[2])) for o in market if len(o)>=3 and o[:2]==['SELL','WOOL'])
        extra=min(state['credit'],max(0,int(stock.get('WOOL',0))-planned))
        if extra>0:
            for order in market:
                if len(order)>=3 and order[:2]==['SELL','WOOL']:
                    order[2]=int(order[2])+extra
                    break
            else:
                if len(market)<10:market.append(['SELL','WOOL',extra])
                else:extra=0
            state['credit']-=extra
            _SH29_REPORT['extra_sale_requests']+=extra
    result['market']=market
    return result
sheep_harvest_priority_agent.telemetry=_SH29_REPORT
agent=sheep_harvest_priority_agent
kaggle_submission_agent=sheep_harvest_priority_agent
