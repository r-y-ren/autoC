
# Original Round9: reclaim only a hired worker's finished native day.
# Current and future mandatory physical commands always take precedence.
_R9S_PARENT=round9_final_market_agent
_R9S_STATE={}
_R9S_REPORT={'crew_turns':0,'care':0,'fertilizer':0,'harvest':0,'sales':0,'errors':0}
_R9S_MOVES={'NORTH':(0,-1),'SOUTH':(0,1),'EAST':(1,0),'WEST':(-1,0)}
_R9S_PRODUCT={'COW':'MILK','SHEEP':'WOOL','GOOSE':'EGG'}
_R9S_PERIOD={'COW':2,'SHEEP':3,'GOOSE':1}
_R9S_FIRST={'COW':8,'SHEEP':6,'GOOSE':4}

def _r9s_reservations(obs,native):
    step=int(obs['step']);positions=[list(obs['farms'][obs['player']]['farmer'])]+[list(p) for p in obs['farms'][obs['player']]['hands']]
    reserved=set();busy=set()
    for t in range(step+1,min((step//24+1)*24,719)):
        tape=_IMPL.chassis.routes[2 if t>=648 else native['route']]
        a=tape[t];cmds=[a.get('farmer') or ['PASS']]+list(a.get('hands') or [])
        for i,c in enumerate(cmds[:len(positions)]):
            if c and c!=['PASS']:
                busy.add(i)
                if c[0] in _R9S_MOVES:
                    dx,dy=_R9S_MOVES[c[0]];positions[i]=[max(0,min(9,positions[i][0]+dx)),max(0,min(9,positions[i][1]+dy))]
                elif c[0] in ('CARE','HARVEST','COLLECT_FERTILIZER'):reserved.add((tuple(positions[i]),c[0]))
    return reserved,busy

def _r9s_apply(obs,action):
    step=int(obs['step']);day=step//24;seat=int(obs['player'])
    if day<12 or day>=29 or step%24<8:return action
    farm=obs['farms'][seat];private=obs['private'];native=_IMPL.chassis.players.get(seat)
    if not native or _V219_STATES.get(seat,{}).get('committed'):return action
    state=_R9S_STATE.setdefault(seat,{'day':day,'credit':{},'previous':{}})
    if state['day']!=day:state['day']=day;state['previous']={}
    for actor,past in state['previous'].items():
        if past['step']==step-1 and actor<len(private['inventories']):
            inv=private['inventories'][actor]
            for item in ('MILK','WOOL','EGG','FERTILIZER'):
                gain=max(0,int(inv.get(item,0))-past['stock'].get(item,0))
                if gain:state['credit'][item]=state['credit'].get(item,0)+gain
    state['previous']={}
    cmds=[action.get('farmer') or ['PASS']]+list(action.get('hands') or []);positions=[farm['farmer']]+farm['hands']
    cmds+=[['PASS'] for _ in range(len(positions)-len(cmds))]
    free=[i for i in range(1,len(positions)) if cmds[i]==['PASS']]
    if not free:return action
    reserved,busy=_r9s_reservations(obs,native)
    for i,c in enumerate(cmds):
        if i<len(positions) and c and c[0] in ('CARE','HARVEST','COLLECT_FERTILIZER'):reserved.add((tuple(positions[i]),c[0]))
    stock_total=sum(private['shed'].values())+sum(sum(x.values()) for x in private['inventories'])
    targets=set();changed=False
    for actor in free:
        if actor in busy:continue
        pos=positions[actor];best=None
        for y,row in enumerate(farm['tiles']):
            for x,tile in enumerate(row):
                if not isinstance(tile,dict) or tile.get('animal') not in _R9S_PRODUCT:continue
                site=(x,y);kind=tile['animal'];product=_R9S_PRODUCT[kind]
                distance=abs(pos[0]-x)+abs(pos[1]-y)
                if distance+1>23-step%24:continue
                tasks=[];yield_units=int(tile.get('yield_units',0));pending=int(tile.get('pending_care_bonus',0))
                first=int(tile['placed_day'])+_R9S_FIRST[kind]
                next_yield=first if day<first else day+((_R9S_PERIOD[kind]-(day-first)%_R9S_PERIOD[kind])%_R9S_PERIOD[kind])
                if tile.get('fed_today') and not tile.get('cared_today') and next_yield<30 and yield_units+pending<5:
                    tasks.append(('CARE',float(obs['market']['prices'][product]),0))
                if stock_total<85 and tile.get('fertilizer_available'):
                    tasks.append(('COLLECT_FERTILIZER',float(obs['market']['prices']['FERTILIZER']),1))
                if yield_units>0 and stock_total+yield_units<=85:
                    tasks.append(('HARVEST',.3*yield_units*float(obs['market']['prices'][product]),yield_units))
                for op,value,quantity in tasks:
                    if (site,op) in reserved or site in targets:continue
                    score=value/(distance+1)
                    if best is None or score>best[0]:best=(score,site,op,quantity)
        if best is None:continue
        _,site,op,quantity=best;targets.add(site)
        command=_v219_walk(pos,site) or [op];cmds[actor]=command;changed=True
        _R9S_REPORT['crew_turns']+=1
        if len(command)==1 and command[0]==op:
            _R9S_REPORT[{'CARE':'care','HARVEST':'harvest','COLLECT_FERTILIZER':'fertilizer'}[op]]+=1
            stock_total+=quantity
        state['previous'][actor]={'step':step,'stock':dict(private['inventories'][actor])}
    if not changed:return action
    result=dict(action,farmer=cmds[0],hands=cmds[1:]);market=[list(o) for o in result.get('market',[])]
    # Liquidate credited extra cash crops only. Fertilizer stays available to
    # the parent; its future resource obligations take priority over sale value.
    projected=projected_shed(result,FarmView(obs))
    for item in ('MILK','WOOL','EGG'):
        planned=sum(int(o[2]) for o in market if o and len(o)>=3 and o[:2]==['SELL',item])
        extra=min(state['credit'].get(item,0),max(0,int(projected.get(item,0))-planned))
        if extra>0:
            order=next((o for o in market if o and o[:2]==['SELL',item]),None)
            if order is not None:order[2]+=extra
            elif len(market)<10:market.append(['SELL',item,extra])
            else:continue
            state['credit'][item]-=extra;_R9S_REPORT['sales']+=extra
    result['market']=market
    return result

def round9_slack_agent(observation,configuration=None):
    if int(observation['step'])==0:
        _R9S_STATE.clear()
        for k in _R9S_REPORT:_R9S_REPORT[k]=0
    return _r9s_apply(observation,_R9S_PARENT(observation,configuration))
round9_slack_agent.telemetry=_R9S_REPORT
