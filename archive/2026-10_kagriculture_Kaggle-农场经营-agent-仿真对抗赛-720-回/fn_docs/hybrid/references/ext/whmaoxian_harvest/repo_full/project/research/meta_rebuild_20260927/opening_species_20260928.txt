
# Research opening alternative; first cow carrier only, exact geometry retained.
_OS29_PARENT=phase_b_agent
_OS29_REPORT={'plans':0,'plan_errors':0,'wheat_reduced':0}
_MP_REPORT=_OS29_REPORT
_CS_FROM=720
_CS_TO=720

def opening_species_agent(observation,configuration=None):
    step=int(observation['step']);seat=int(observation['player'])
    if step==0:
        for key in _OS29_REPORT:_OS29_REPORT[key]=0
    action=_OS29_PARENT(observation,configuration)
    if step==0 and _OS29_MODE=='SHEEP':
        market=[list(o) for o in action.get('market',[])]
        for order in market:
            if order[:2]==['BUY_PRODUCT','WHEAT'] and len(order)>=3 and order[2]==5:
                order[2]=1;_OS29_REPORT['wheat_reduced']+=4
        return dict(action,market=market)
    if step!=1:return action
    market=[list(o) for o in action.get('market',[])]
    buys=[o for o in market if len(o)>=3 and o[:2]==['BUY_ANIMAL','COW'] and o[2]>=1]
    if not buys:return action
    plan=_cs_plan(observation,action,1,_OS29_MODE)
    if plan is None:
        _OS29_REPORT['plan_errors']+=1
        return action
    buys[0][2]-=1
    for order in market:
        if len(order)>=3 and order[:2]==['BUY_ANIMAL',_OS29_MODE]:
            order[2]+=1
            break
    else:market.append(['BUY_ANIMAL',_OS29_MODE,1])
    state=_CS_STATE[seat]
    state['mode']=_OS29_MODE;state['plan']=plan;state['decided']=True
    _CS_REPORT['cs_decision']=_OS29_MODE+'@opening'
    _OS29_REPORT['plans']+=1
    return dict(action,market=market)
opening_species_agent.telemetry=_OS29_REPORT
agent=opening_species_agent
kaggle_submission_agent=opening_species_agent
