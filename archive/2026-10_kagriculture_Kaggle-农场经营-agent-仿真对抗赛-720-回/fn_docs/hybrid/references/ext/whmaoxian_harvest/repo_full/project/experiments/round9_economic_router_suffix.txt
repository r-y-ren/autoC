
# Round9 original decision layer: robust marginal cash-flow routing.
# No world seed, opponent identity, future shop draw or private rival state is read.
_R9R_PARENT=round8_production_fusion_agent
_R9R_OLD_ROUTER=_IMPL.chassis.router
_R9R_REPORT={'decisions':0,'switches':0,'errors':0}
_R9R_EVENTS=[]
_R9R_MARGIN=500.0
_R9R_SHOPS={'BAKERY':{'EGG':1,'WHEAT':1},'PIZZA_SHOP':{'MILK':1,'TOMATO':1,'WHEAT':1},
 'BRUNCH_SPOT':{'EGG':1,'WHEAT':1,'STRAWBERRY':1},'YARN_STORE':{'WOOL':2},
 'ICE_CREAM_SHOP':{'STRAWBERRY':1,'MILK':1,'WHEAT':1},'PET_CAFE':{'CARROT':2},
 'SMOOTHIE_SHOP':{'STRAWBERRY':1,'MILK':1},'FARMERS_MARKET':{'WHEAT':1,'CARROT':1,'TOMATO':1,'STRAWBERRY':1}}
_R9R_ITEMS=list(_R37_MARKET_PARAMS)
_R9R_AVG={k:sum(s.get(k,0) for s in _R9R_SHOPS.values())/8 for k in _R9R_ITEMS}

def _r9r_counts(farm):
    out={}
    for row in farm['tiles']:
        for t in row:
            if isinstance(t,dict):
                k=t.get('animal') or t.get('crop')
                if k:out[k]=out.get(k,0)+1
    return out

def _r9r_opponent(obs,step):
    counts=_r9r_counts(obs['farms'][1-int(obs['player'])])
    distances={}
    for route,profile in _R9R_PROFILES.items():
        other=profile['counts'][str(step)]
        distances[route]=sum((3 if k in ('COW','SHEEP','GOOSE') else 1)*abs(counts.get(k,0)-other.get(k,0)) for k in set(counts)|set(other))
    best=min(distances.values());ids=[r for r,v in distances.items() if v<=best+2]
    output=[]
    for d in range(30):
        output.append({key:{k:sum(_R9R_PROFILES[r]['days'][d][key].get(k,0) for r in ids)/len(ids) for k in _R9R_ITEMS} for key in ('sell','buy')})
    return output

def _r9r_value(obs,route,opponent,supply,demand):
    day=int(obs['step'])//24;inventory={k:float(v) for k,v in obs['market']['inventory'].items()}
    current={k:sum(_R9R_SHOPS.get(s,{}).get(k,0) for s in obs['town']['unlocked_shops']) for k in _R9R_ITEMS}
    known=len(obs['town']['unlocked_shops']);value=0.0
    for d in range(day,30):
        own=_R9R_PROFILES[route]['days'][d];opp=opponent[d]
        unknown=max(0,min(8,d//3)-known)
        value-=own['fixed']
        for k in _R9R_ITEMS:
            drain=(0 if k=='FERTILIZER' else 1)+6*(current[k]+unknown*_R9R_AVG[k]*demand)
            os=opp['sell'].get(k,0)*supply;ob=opp['buy'].get(k,0)*supply
            sell=own['sell'].get(k,0);buy=own['buy'].get(k,0)
            # Four midpoint slices approximate the marginal price integral.
            # $1 sales do not add supply; purchases still reduce inventory.
            for z in range(4):
                delta=(sell+os-buy-ob-drain)/4
                quote=_r37_market_price(k,inventory[k]+delta/2)
                value+=(sell-buy)*quote/4
                if quote<=1:delta=(-buy-ob-drain)/4
                inventory[k]+=delta
    return value

def _r9r_router(obs,step,state):
    base=_R9R_OLD_ROUTER(obs,step,state)
    if step not in (144,216,288,360,432):return base
    if base not in _R9R_PROFILES:return base
    candidates=next((group for group in _R9R_GROUPS[str(step)] if base in group),[base])
    if len(candidates)<2:return base
    opponent=_r9r_opponent(obs,step)
    scenarios=[(.75,1.25),(1,1),(1.25,.75)]
    base_values=[_r9r_value(obs,base,opponent,*s) for s in scenarios]
    best=base;best_mean=0;best_worst=0
    for route in candidates:
        if route==base:continue
        gains=[_r9r_value(obs,route,opponent,*s)-v for s,v in zip(scenarios,base_values)]
        mean=sum(gains)/len(gains);worst=min(gains)
        if worst>_R9R_MARGIN and mean>best_mean:
            best=route;best_mean=mean;best_worst=worst
    _R9R_REPORT['decisions']+=1
    if best!=base:
        state['route']=best;_R9R_REPORT['switches']+=1
    _R9R_EVENTS.append({'step':step,'base':base,'selected':best,'expected_increment':round(best_mean,2),'worst_scenario_increment':round(best_worst,2)})
    return best

_IMPL.chassis.router=_r9r_router

def round9_economic_router_agent(observation,configuration=None):
    if int(observation['step'])==0:
        for k in _R9R_REPORT:_R9R_REPORT[k]=0
        _R9R_EVENTS.clear()
    return _R9R_PARENT(observation,configuration)
round9_economic_router_agent.telemetry=_R9R_REPORT
