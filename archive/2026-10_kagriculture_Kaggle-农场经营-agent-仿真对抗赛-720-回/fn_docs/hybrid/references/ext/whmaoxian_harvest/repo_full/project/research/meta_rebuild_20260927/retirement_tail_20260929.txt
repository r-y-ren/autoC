
# Development policy: exit persistently unprofitable mature animal sites.
_K29R_PARENT=phase_b_agent
_K29R_STATE={}
_K29R_REPORT={'chosen':0,'feed_removed':0,'crop_plants':0,'crop_harvest_units':0,'sale_units':0,'delivery_rewrites':0}
_MP_REPORT=_K29R_REPORT
_K29R_HOME=((4,4),(5,4),(4,5),(5,5))
_K29R_SPEC={'COW':('MILK',8,2,3),'SHEEP':('WOOL',6,3,4),'GOOSE':('EGG',4,1,2)}

def _k29r_price_ceiling(obs,animal):
    item,first,period,yield_n=_K29R_SPEC[animal]
    day=int(obs['step'])//24;stock=float(obs['market']['inventory'][item]);prices=[]
    params={k:dict(v) for k,v in _R37_MARKET_PARAMS.items()}
    for k,v in (obs['market'].get('params') or {}).items():
        if k in params:params[k].update(v)
    shops=obs['town']['unlocked_shops'];demand=1+_hd2_daily_shop_demand(item,shops)
    fields=[t for f in obs['farms'] for row in f['tiles'] for t in row if isinstance(t,dict) and t.get('animal')==animal]
    for d in range(day,min(30,day+_K29R_CFG['forecast']+1)):
        if d>day:
            future=min(max(0,8-len(shops)),max(0,d//3-day//3))
            stock-=demand+future*_hd2_future_demand_per_day(item)
            stock+=sum(yield_n for t in fields if d-t['placed_day']>=first and (d-t['placed_day']-first)%period==0)
        prices.append(_r37_market_price(item,int(round(stock)),params))
    return max(prices)

def _k29r_choose(obs,st):
    step=int(obs['step']);day=step//24;seat=int(obs['player']);farm=obs['farms'][seat]
    if step%24 or not _K29R_CFG['start']<=day<=_K29R_CFG['end']:return
    if len(st['sites'])>=_K29R_CFG['cap']:return
    prices=obs['market']['prices'];candidates=[]
    ceilings={animal:_k29r_price_ceiling(obs,animal) for animal in ('COW','SHEEP')}
    feed=float(prices['WHEAT']);fert=float(prices['FERTILIZER'])
    for y,row in enumerate(farm['tiles']):
        for x,t in enumerate(row):
            if (x,y) in st['sites'] or not isinstance(t,dict):continue
            animal=t.get('animal')
            if animal not in ceilings or day-t.get('placed_day',day)<_K29R_SPEC[animal][1]:continue
            ceiling=ceilings[animal]
            if ceiling>_K29R_CFG['price']:continue
            production=_K29R_SPEC[animal][3]/_K29R_SPEC[animal][2]
            value=production*ceiling+fert-feed
            crop=_K29R_CFG['crop']
            alternative=(4*float(prices[crop])-(20 if crop=='CARROT' else 10))/4 if crop else 0
            if value>=alternative*_K29R_CFG['crop_factor']:continue
            distance=min(abs(x-a)+abs(y-b) for a,b in _K29R_HOME)
            candidates.append((value-alternative,distance,x,y,animal))
    counts={a:sum(isinstance(t,dict) and t.get('animal')==a and (x,y) not in st['sites'] for y,row in enumerate(farm['tiles']) for x,t in enumerate(row)) for a in ceilings}
    added=0
    for _,_,x,y,animal in sorted(candidates):
        if len(st['sites'])>=_K29R_CFG['cap'] or added>=_K29R_CFG['daily']:break
        if counts[animal]<=2:continue
        st['sites'][(x,y)]=dict(animal=animal,selected=day)
        counts[animal]-=1;added+=1;_K29R_REPORT['chosen']+=1

def _k29r_work(obs,farm,private,actor,command,site):
    step=int(obs['step']);day=step//24;pos=([farm['farmer']]+farm['hands'])[actor]
    tile=farm['tiles'][pos[1]][pos[0]];op=command[0] if command else 'PASS'
    if op in ('NORTH','SOUTH','EAST','WEST','PICKUP','PLACE','DROP'):return command
    if isinstance(tile,dict) and tile.get('animal'):
        if op=='FEED':_K29R_REPORT['feed_removed']+=1
        if tile.get('yield_units',0)>0:return ['HARVEST']
        if tile.get('fertilizer_available') and obs['market']['prices']['FERTILIZER']>1:return ['COLLECT_FERTILIZER']
        return ['PASS']
    crop=_K29R_CFG['crop']
    if not crop:return ['PASS']
    if tile is None:
        if day<=26 and private['seeds'].get(crop,0)>0:return ['PLANT',crop]
        return ['PASS']
    if isinstance(tile,dict) and tile.get('kind') in ('WEED','PASTURE','COOP'):
        return ['DIG'] if day<=26 else ['PASS']
    if isinstance(tile,dict) and tile.get('kind')=='PLANT':
        age=day-int(tile['planted_day']);yield_n=int(tile.get('yield_units',0))
        if not tile.get('watered_today') and (day<29 or age>=2 and yield_n>0 and step<717):return ['WATER']
        ripe=age>=3 if crop=='CARROT' else age>=4
        if yield_n>0 and (ripe or day>=28 and age>=2):return ['HARVEST']
    return ['PASS']

def retirement_agent(observation,configuration=None):
    step=int(observation['step']);seat=int(observation['player']);day=step//24
    if step==0 or seat not in _K29R_STATE:
        _K29R_STATE[seat]={'sites':{},'credit':0}
        for key in _K29R_REPORT:_K29R_REPORT[key]=0
    st=_K29R_STATE[seat];_k29r_choose(observation,st)
    original=_K29R_PARENT(observation,configuration)
    if not st['sites']:return original
    farm=copy.deepcopy(observation['farms'][seat]);private=copy.deepcopy(observation['private'])
    positions=[farm['farmer']]+farm['hands'];units=[list(original.get('farmer') or ['PASS'])]+[list(c) for c in original.get('hands',[])]
    units+=[['PASS'] for _ in range(max(0,len(positions)-len(units)))];crop=_K29R_CFG['crop']
    for actor,pos in enumerate(positions):
        command=units[actor];tile=farm['tiles'][pos[1]][pos[0]]
        if tuple(pos) in st['sites']:command=_k29r_work(observation,farm,private,actor,command,st['sites'][tuple(pos)])
        bag=private['inventories'][actor]
        if crop and tuple(pos) in _K29R_HOME and len(command)>=2 and command[0]=='PLACE' and command[1] in PRODUCTS and bag.get(command[1],0)==0 and bag.get(crop,0)>0:
            command=['PLACE',crop,int(bag[crop])];_K29R_REPORT['delivery_rewrites']+=1
        before=bag.get(crop,0) if crop else 0
        _UNIT_NS['_apply_unit_action'](farm,private,actor,command,len(farm['tiles']),day,24,100)
        if tuple(pos) in st['sites'] and crop:
            gained=max(0,private['inventories'][actor].get(crop,0)-before)
            if command[0]=='HARVEST':st['credit']+=gained;_K29R_REPORT['crop_harvest_units']+=gained
            if command[:2]==['PLANT',crop] and isinstance(farm['tiles'][pos[1]][pos[0]],dict) and farm['tiles'][pos[1]][pos[0]].get('crop')==crop:_K29R_REPORT['crop_plants']+=1
        units[actor]=command
    action=dict(original,farmer=units[0],hands=units[1:],market=[list(o) for o in original.get('market',[])])
    market=action['market'];stock=private['shed']
    if crop and st['credit']>0:
        planned=sum(max(0,int(o[2])) for o in market if len(o)>=3 and o[:2]==['SELL',crop])
        quantity=min(st['credit'],max(0,int(stock.get(crop,0))-planned))
        hit=next((o for o in market if len(o)>=3 and o[:2]==['SELL',crop]),None)
        if quantity>0 and (observation['market']['prices'][crop]>1 or day>=29):
            if hit is not None:quantity=min(quantity,max(0,100-int(hit[2])));hit[2]+=quantity
            elif len(market)<10:market.append(['SELL',crop,quantity])
            else:quantity=0
            st['credit']-=quantity;_K29R_REPORT['sale_units']+=quantity
    if crop and day<=26 and len(market)<10:
        planned=sum(max(0,int(o[2])) for o in market if len(o)>=3 and o[:2]==['BUY_SEED',crop])
        missing=max(0,min(4,len(st['sites']))-private['seeds'].get(crop,0)-planned)
        seed_cost={'WHEAT':10,'CARROT':20,'TOMATO':50,'STRAWBERRY':100,'MELON':80}
        cash=float(farm['money']);hires=int(farm['hires_today']);land=len(farm['unlocked_quadrants'])
        uncertain=False
        for order in market:
            if not order:continue
            if order[0]=='HIRE':cash-=_v219_fib(hires);hires+=1
            elif order[0]=='BUY_LAND' and land<4:cash-=(1000,2000,4000)[land-1];land+=1
            elif order[0]=='BUY_PRODUCT':uncertain=True
            elif len(order)>=3 and order[0]=='BUY_SEED':cash-=seed_cost.get(order[1],0)*max(0,int(order[2]))
            elif len(order)>=3 and order[0]=='BUY_ANIMAL':cash-=_HD2_SPEC[order[1]]['cost']*max(0,int(order[2]))
        if missing and not uncertain and cash>=missing*seed_cost[crop]+200:market.append(['BUY_SEED',crop,missing])
    previous=_OR2_STATE.get(seat,{}).get('prev')
    if previous and previous.get('step')==step:
        own=previous.setdefault('own',{})
        for item in _ADV_ITEMS:
            requested=sum(min(100,max(0,int(o[2]))) for o in market if len(o)>=3 and o[:2]==['SELL',item])
            own[item]=min(int(stock.get(item,0)),requested)
    return action
