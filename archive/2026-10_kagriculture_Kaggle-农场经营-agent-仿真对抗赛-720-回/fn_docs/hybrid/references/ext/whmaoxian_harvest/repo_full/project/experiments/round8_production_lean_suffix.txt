def _t19_distance(a,b):
    return abs(a[0]-b[0])+abs(a[1]-b[1])

def _t19_qualifies(obs, state):
    farm=obs['farms'][int(obs['player'])]
    if not state.get('eligible') or farm['money']<18000:
        return False
    if any(farm['tiles'][y][x]!='LOCKED' for x,y in _T19_TARGETS):
        return False
    shops=obs['town']['unlocked_shops']
    demand=sum(s in ('PIZZA_SHOP','FARMERS_MARKET') for s in shops)
    if demand<4:
        return False
    # Ignore future unknown shop draws; reserve at least another nineteen-
    # plant investment, since an opponent's expansion was already observed.
    remaining=719-int(obs['step'])
    other=obs['farms'][1-int(obs['player'])]
    rival_plants=sum(isinstance(t,dict) and t.get('crop')=='TOMATO' for r in other['tiles'] for t in r)
    forecast=int(obs['market']['inventory']['TOMATO']) - (remaining//4)*demand - remaining//24 + 152 + max(152,8*rival_plants)
    quote=max(1,_r37_market_price('TOMATO',forecast))
    # Price the extra labor from our own known native hiring schedules.
    extra_labor=0
    baseline={18:2,19:0,20:1,21:0,22:1,23:0,24:2,25:1,26:3,27:3,28:2,29:2}
    for day in range(18,30):
        n=max(len(a.get('hands',[])) for a in _v219_native_day(_IMPL.chassis.players[int(obs['player'])],day))
        proposed=0 if day in (19,21,23) else len(_t19_groups(day))
        extra_labor+=sum(_v219_fib(i) for i in range(n,n+proposed))-sum(_v219_fib(i) for i in range(n,n+baseline[day]))
    incremental_value=72*quote-450-extra_labor-18*int(obs['market']['prices']['FERTILIZER'])
    state['t19_forecast_value']=incremental_value
    state['t19_forecast_extra_labor']=extra_labor
    return incremental_value>=500

def _t19_groups(day):
    if day==18:return _T19_PLANT
    if day in (25,26):return _T19_FOUR
    if day==27:return _T19_PLANT
    if day==29:return _T19_FOUR
    return _T19_WATER

def _t19_assignment(positions,first,groups,day):
    best=None
    for order in _t19_it.permutations(range(len(groups))):
        durations=[]
        for i,k in enumerate(order):
            path=groups[k]
            start=positions[first+i]
            entry=_t19_distance(start,path[0])
            within=sum(_t19_distance(a,b) for a,b in zip(path,path[1:]))
            ops=len(path)*(2 if day in (18,25,26,27) else 1)
            pickup=int(day in (25,26))
            finish=(_t19_distance(path[-1],_v219_home(path[-1]))+1) if day==29 else 0
            durations.append(entry+within+ops+pickup+finish)
        score=(max(durations),sum(durations),order)
        if best is None or score<best[0]:best=(score,order)
    return best

def _t19_assign(obs, state, pending):
    farm=obs['farms'][int(obs['player'])]
    first=pending['first'];day=int(obs['step'])//24
    groups=_t19_groups(day)
    if len(farm['hands'])+1<first+len(groups) or 'SE' not in farm['unlocked_quadrants']:
        _T19_REPORT['hire_shortfalls']+=len(groups)
        return
    positions=[farm['farmer'],*farm['hands']]
    best=_t19_assignment(positions,first,groups,day)
    state['workers']={first+i:{'kind':'crop','targets':list(groups[k]),'t19':True,
                              'needs_fertilizer':day in (25,26),'loaded':False}
                      for i,k in enumerate(best[1])}
    _T19_REPORT['confirmed_workers']+=len(groups)

def _v219_worker(obs, state, actor, role):
    if not role.get('t19'):
        return _T19_WORKER(obs,state,actor,role)
    day=int(obs['step'])//24;step=int(obs['step']);view=FarmView(obs)
    pos=tuple(view.positions[actor]);inv=view.inventory(actor)
    home=_v219_home(pos)
    if day==29 and step>=718-_t19_distance(pos,home) and inv.get('TOMATO',0):
        return _v219_walk(pos,home) or ['PLACE','TOMATO',int(inv['TOMATO'])]
    if role.get('needs_fertilizer') and not role.get('loaded'):
        if inv.get('FERTILIZER',0)>=len(role['targets']):
            role['loaded']=True
        elif role.get('pickup_requested'):
            role['loaded']=True
        else:
            walk=_v219_walk(pos,home)
            if walk:return walk
            quantity=min(len(role['targets']),int(view.shed.get('FERTILIZER',0)))
            role['pickup_requested']=True
            if quantity:return ['PICKUP','FERTILIZER',quantity]
            role['loaded']=True
    for target in role['targets']:
        x,y=target;tile=view.tiles[y][x]
        tomato=isinstance(tile,dict) and tile.get('crop')=='TOMATO'
        if tomato:state['seen_plants'].add(target)
        command=None
        if day==18 and not tomato:
            if tile is None and view.seeds.get('TOMATO',0)>0:
                command=['PLANT','TOMATO']
            elif isinstance(tile,dict) and tile.get('kind')=='WEED':command=['DIG']
        elif tomato:
            if day<29 and not tile.get('watered_today'):command=['WATER']
            elif day in (25,26) and tile.get('fertilized_until_day',-1)<day+2 and inv.get('FERTILIZER',0):
                command=['FERTILIZE']
            elif day in (27,29) and tile.get('yield_units',0)>0:command=['HARVEST']
        if command:
            work=_t19_distance(pos,target)+1
            back=(_t19_distance(target,_v219_home(target))+1) if day==29 else 0
            deadline=718 if day==29 else (day+1)*24-1
            if work+back>deadline-step+1:
                _T19_REPORT['partial_route_deferrals']=_T19_REPORT.get('partial_route_deferrals',0)+1
                continue
            walk=_v219_walk(pos,target)
            if walk:return walk
            if command[0]=='PLANT':_T19_REPORT['plants']+=1
            if command[0]=='HARVEST':_T19_REPORT['harvest_requests']+=1
            if command[0]=='FERTILIZE':_T19_REPORT['fertilizer_requests']+=1
            return command
    if inv.get('TOMATO',0):
        return _v219_walk(pos,home) or ['PLACE','TOMATO',int(inv['TOMATO'])]
    if any(inv.values()):return _v219_walk(pos,home) or ['DROP']
    return ['PASS']

def round8_production_agent(observation,configuration=None):
    if int(observation['step'])==0:
        for k in _T19_REPORT:_T19_REPORT[k]=0
    try:
        return _T19_PARENT(observation,configuration)
    except Exception:
        _T19_REPORT['errors']+=1
        raise
