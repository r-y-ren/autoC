_TS_PARENT=phase_b_agent
_TS_ORIGINAL=_PLANNER_NS['plan_terminal']
_TS_REPORT={'calls':0,'accepted':0,'simulations':0,'max_ms':0.0,'errors':0}
def _ts_plan(obs,config,baseline_remaining,**options):
    if _TS_MODE=='control':return _TS_ORIGINAL(obs,config,baseline_remaining,**options)
    result=_TS_ORIGINAL(obs,config,baseline_remaining,max_simulations=_TS_SIMULATIONS,passes=_TS_PASSES,proposals_per_actor=_TS_PROPOSALS)
    _TS_REPORT['calls']+=1
    _TS_REPORT['accepted']+=int(bool(result.get('accepted')))
    _TS_REPORT['simulations']+=int(result.get('simulations',0))
    _TS_REPORT['max_ms']=max(_TS_REPORT['max_ms'],float(result.get('planning_ms',0)))
    return result
_PLANNER_NS['plan_terminal']=_ts_plan

def terminal_search_agent(observation,configuration=None):
    if int(observation['step'])==0:
        for key in _TS_REPORT:_TS_REPORT[key]=0
    return _TS_PARENT(observation,configuration)
_MP_REPORT=_TS_REPORT
terminal_search_agent.telemetry=_TS_REPORT
agent=terminal_search_agent
kaggle_submission_agent=terminal_search_agent
