_TI_PARENT = round8_fullfusion_agent
_TI_REPORT = {}
_TI_CACHE = {}
_TI_LAST = {}
_TI_TELEMETRY = {}
_TI_SEEDS = ('WHEAT','CARROT')


def _ti_standard(config):
    return config is None or all(config.get(k,v)==v for k,v in
        [('boardSize',10),('turnsPerDay',24),('episodeSteps',720),('shedCapacity',100),('maxMarketOrdersPerTurn',10)])


def _ti_fert_value(tile,day):
    # On the final day an ongoing crop's next refresh occurs after the match.
    # But a mature wheat/carrot can still gain one yield unit before harvest.
    if not isinstance(tile,dict) or tile.get('crop') not in ('WHEAT','CARROT'):
        return False
    crop=tile['crop'];last,cap=(4,6) if crop=='WHEAT' else (3,4)
    age=day-int(tile['planted_day'])
    return (2<=age<=last and not tile.get('watered_today')
            and int(tile.get('fertilized_until_day',-1))<day
            and int(tile.get('yield_units',0))<cap-1)


def _ti_monitor(obs,action):
    step=int(obs['step']);day=step//24
    if step<648:return
    units=[action.get('farmer') or ['PASS'],*(action.get('hands') or [])]
    farm=obs['farms'][int(obs['player'])];private=obs['private']
    positions=[farm['farmer'],*farm['hands']]
    demanded={p:sum(c[:2]==['PLANT',p] for c in units) for p in _TI_SEEDS}
    # Engine's atomic rule rejects the whole crop group when seeds are short.
    for p in _TI_SEEDS:
        if step<=671 and demanded[p]>int(private['seeds'].get(p,0)):
            eligible=sum(c[:2]==['PLANT',p] and farm['tiles'][pos[1]][pos[0]] is None
                         for c,pos in zip(units,positions))
            _TI_REPORT['productive_seed_shortfall_units']+=eligible
    for i,(c,pos) in enumerate(zip(units,positions)):
        if c==['FERTILIZE'] and day==29 and _ti_fert_value(farm['tiles'][pos[1]][pos[0]],day):
            if i<len(private['inventories']) and int(private['inventories'][i].get('FERTILIZER',0))<1:
                _TI_REPORT['productive_fertilizer_shortfall_units']+=1


def _ti_seed_reserve(native,step):
    # Preserve enough of EACH species for every remaining seed opportunity,
    # regardless of future relative prices and wheat/carrot conversions.
    if step>=671:return 0
    key=step
    if key not in _TI_CACHE:
        tape=_IMPL.chassis.routes[2]
        _TI_CACHE[key]=sum(bool(c and len(c)>1 and c[0]=='PLANT' and c[1] in _TI_SEEDS)
                          for t in range(step+1,672)
                          for c in [tape[t].get('farmer') or ['PASS'],*(tape[t].get('hands') or [])])
    retries=sum(bool(c and len(c)>1 and c[0]=='PLANT' and c[1] in _TI_SEEDS)
                for queue in native.get('pending',{}).values() for pos,c in queue)
    return _TI_CACHE[key]+retries


def _ti_fert_reserve(native,step):
    # Reserve every entire remaining planned pickup and sale; never discount
    # carried inventory, expected collection, a future buy, or a price forecast.
    tape=_IMPL.chassis.routes[2]
    needed=0
    for t in range(step+1,719):
        a=tape[t]
        for c in [a.get('farmer') or ['PASS'],*(a.get('hands') or [])]:
            if c[:2]==['PICKUP','FERTILIZER']:
                needed+=max(0,int(c[2]) if len(c)>2 else 1)
        for o in a.get('market') or []:
            if len(o)>2 and o[:2]==['SELL','FERTILIZER']:needed+=max(0,int(o[2]))
    for queue in native.get('pending',{}).values():
        for pos,c in queue:
            if c[:2]==['PICKUP','FERTILIZER']:
                needed+=max(0,int(c[2]) if len(c)>2 else 1)
    return needed


def _ti_apply(obs,action):
    step=int(obs['step']);seat=int(obs['player'])
    if step<648:return action
    orders=[list(o) for o in action.get('market') or []]
    if not any(o and o[0] in ('BUY_SEED','BUY_PRODUCT') for o in orders):return action
    if not _r97_budget(obs,orders):
        _TI_REPORT['unfunded_abstentions']+=1
        return action
    farm,private=_r127_fields(obs,action)
    native=_IMPL.chassis.players[seat]
    seeds=dict(private['seeds']);need=_ti_seed_reserve(native,step)
    changed=False
    for i,o in enumerate(orders):
        if len(o)<3 or o[0]!='BUY_SEED' or o[1] not in _TI_SEEDS:continue
        crop=o[1];quantity=max(0,int(o[2]));keep=min(quantity,max(0,need-int(seeds.get(crop,0))))
        seeds[crop]=int(seeds.get(crop,0))+keep
        if keep==quantity:continue
        cut=quantity-keep;changed=True
        orders[i]=['BUY_SEED',crop,keep] if keep else []
        _TI_REPORT['seed_cut_'+crop.lower()]+=cut
        _TI_REPORT['seed_nominal_savings']+=cut*(10 if crop=='WHEAT' else 20)
        if crop=='CARROT':
            st=_CA_STATE.get(seat)
            if st is not None:st['spare_carrot']=max(0,st.get('spare_carrot',0)-cut)
    if 696<=step<=718 and any(len(o)>2 and o[:2]==['BUY_PRODUCT','FERTILIZER'] for o in orders):
        useful=any(_ti_fert_value(tile,29) for row in farm['tiles'] for tile in row)
        reactive=(_R51_INPUT_STATES.get(seat) or {}).get('workers',{})
        # Same-turn fertilizer trading may be an intentional price sequence.
        trading=any(o[:2]==['SELL','FERTILIZER'] for o in orders)
        if useful or reactive or trading:
            _TI_REPORT['fertilizer_value_abstentions']+=1
        else:
            reserve=_ti_fert_reserve(native,step)
            available=max(0,int(private['shed'].get('FERTILIZER',0)))
            for i,o in enumerate(orders):
                if len(o)<3 or o[:2]!=['BUY_PRODUCT','FERTILIZER']:continue
                qty=max(0,int(o[2]));keep=min(qty,max(0,reserve-available))
                available+=keep
                if keep==qty:continue
                changed=True;cut=qty-keep
                orders[i]=['BUY_PRODUCT','FERTILIZER',keep] if keep else []
                _TI_REPORT['fertilizer_cut_units']+=cut
                _TI_REPORT['fertilizer_quote_savings']+=cut*int(obs['market']['prices']['FERTILIZER'])
    if changed:
        _TI_REPORT['changed_turns']+=1
        return dict(action,market=orders)
    return action


def round8_terminal_inputs_agent(observation,configuration=None):
    step=int(observation['step']);seat=int(observation['player'])
    if step<=_TI_LAST.get(seat,-1):_TI_CACHE.clear()
    if step==0:
        _TI_REPORT.clear()
        _TI_REPORT.update(seed_cut_wheat=0,seed_cut_carrot=0,seed_nominal_savings=0,
                          fertilizer_cut_units=0,fertilizer_quote_savings=0,changed_turns=0,
                          unfunded_abstentions=0,fertilizer_value_abstentions=0,
                          productive_seed_shortfall_units=0,productive_fertilizer_shortfall_units=0,errors=0)
    _TI_LAST[seat]=step
    action=_TI_PARENT(observation,configuration)
    try:
        if _ti_standard(configuration):
            _ti_monitor(observation,action)
            action=_ti_apply(observation,action)
    except Exception:
        _TI_REPORT['errors']+=1
    st=_RACE_STATE.get(seat)
    if st is not None and st.get('prev_action') is not None and st.get('step')==step:st['prev_action']=action
    _TI_TELEMETRY.clear();_TI_TELEMETRY.update(getattr(_TI_PARENT,'telemetry',{}))
    _TI_TELEMETRY.update({'terminal_inputs_'+k:v for k,v in _TI_REPORT.items()})
    return action
round8_terminal_inputs_agent.telemetry=_TI_TELEMETRY
agent=round8_terminal_inputs_agent
