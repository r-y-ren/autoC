
# Distinguish expanding an already-profitable ten-plot plan from new investment.
_T3_BASE_QUAL=_v219_qualifies
_T3_COMPARE=_t19_qualifies
_T3_FORCED={}
def _v219_qualifies(obs,native):
    seat=int(obs['player']);_T3_FORCED.pop(seat,None)
    original=_T3_BASE_QUAL(obs,native)
    if original:return True
    farm=obs['farms'][seat]
    if set(farm['unlocked_quadrants'])!={'NW','NE','SW'}:return False
    if farm['money']<18000:return False
    if obs['private']['seeds'].get('TOMATO',0) or obs['private']['shed'].get('TOMATO',0):return False
    if any(isinstance(t,dict) and t.get('crop')=='TOMATO' for row in farm['tiles'] for t in row):return False
    for tape in _IMPL.chassis.routes.values():
        for a in tape[432:719]:
            if any(o and o[0]=='BUY_LAND' for o in a.get('market',[])):return False
            if any(c==['PLANT','TOMATO'] for c in [a.get('farmer')]+a.get('hands',[])):return False
    _T3_COMPARE(obs,{'eligible':True})
    value=_TG_DECISIONS.get(seat)
    if not value or value['fullnet']<_TG_MIN_NET:return False
    _T3_FORCED[seat]=True
    return True

def _t19_qualifies(obs,state):
    if _T3_FORCED.get(int(obs['player'])):return True
    if _TG_ORIGINAL(obs,state):return True
    return _T3_COMPARE(obs,state)

def tomato_gate3_agent(observation,configuration=None):
    if int(observation['step'])==0:_T3_FORCED.pop(int(observation['player']),None)
    return tomato_gate2_agent(observation,configuration)
tomato_gate3_agent.telemetry=_TG_REPORT
agent=tomato_gate3_agent
kaggle_submission_agent=tomato_gate3_agent
