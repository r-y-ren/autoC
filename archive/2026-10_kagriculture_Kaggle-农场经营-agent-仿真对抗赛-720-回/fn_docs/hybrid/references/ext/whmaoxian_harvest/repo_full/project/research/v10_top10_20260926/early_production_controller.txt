
def _ep_virtual(obs):
    view=copy.deepcopy(obs);view['step']=int(obs['step'])+_EP_SHIFT*24
    view['day']=int(obs['day'])+_EP_SHIFT
    for row in view['farms'][int(obs['player'])]['tiles']:
        for tile in row:
            if isinstance(tile,dict) and int(tile.get('fertilized_until_day',-1))>=0:
                tile['fertilized_until_day']+=_EP_SHIFT
    return view
_EP_SLACK_PARENT=_r9s_apply
def _r9s_apply(obs,action):
    state=_EP_STATE.get(int(obs['player']),{})
    if state.get('committed') and int(obs['step'])//24<_EP_START+12:return action
    return _EP_SLACK_PARENT(obs,action)
_EP_SHADOW_PARENT=_shadow_terminal
def _shadow_terminal(obs,config):
    state=_EP_STATE.get(int(obs['player']),{})
    if state.get('committed') and int(obs['step'])//24<_EP_START+12:return None
    return _EP_SHADOW_PARENT(obs,config)

def early_production_agent(observation,configuration=None):
    step=int(observation['step']);day=step//24;seat=int(observation['player'])
    state=_EP_STATE.get(seat)
    if state is None or step<=state['last']:
        state=_EP_STATE[seat]={'last':-1,'day':-1,'workers':{},'seen_plants':set(),'targets':list(_T19_TARGETS),'eligible':False,'t19_enabled':True}
        if step==0:
            for key in _EP_REPORT:_EP_REPORT[key]=0
            _EP_LAST_FORECAST.pop(seat,None)
    state['last']=step
    action=_EP_PARENT(observation,configuration)
    native=_IMPL.chassis.players.get(seat)
    if step==_EP_START*24 and native:
        state['eligible']=_ep_qualifies(observation,native)
        if state['eligible']:_EP_REPORT['eligible']+=1
    if not state.get('eligible') or not _EP_START<=day<_EP_START+12:return action
    if state['day']!=day:state['day']=day;state['workers']={}
    virtual=_ep_virtual(observation)
    action=_ep_request(virtual,action,state,native)
    farm=observation['farms'][seat]
    commands=[action.get('farmer') or ['PASS']]+list(action.get('hands') or [])
    commands+=[['PASS'] for _ in range(len(farm['hands'])+1-len(commands))]
    for actor,role in state['workers'].items():
        if actor<len(commands):commands[actor]=_ep_worker(virtual,state,actor,role)
    result=dict(action,farmer=commands[0],hands=commands[1:])
    if state.get('committed'):
        stock=projected_shed(result,FarmView(observation));market=[list(o) for o in action.get('market',[])]
        sold=sum(max(0,int(o[2])) for o in market if len(o)>=3 and o[:2]==['SELL','TOMATO'])
        extra=max(0,int(stock.get('TOMATO',0))-sold)
        if extra and len(market)<10:market.append(['SELL','TOMATO',extra])
        result['market']=market
    return result
early_production_agent.telemetry=_EP_REPORT
agent=early_production_agent
kaggle_submission_agent=early_production_agent
