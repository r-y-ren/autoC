def _k29_choose_route(observation,step,state):
    selected=_K29_ROUTE_NATIVE(observation,step,state)
    if 144<=step<648:
        selected=_K29_FORCED_ROUTE
        state['route']=selected
        state['expert']='EXP240' if selected>=100 else 'V39'
        _K29_ROUTE_REPORT['selected']=selected
        _K29_ROUTE_REPORT['selection_step']=144
    return selected
_IMPL.chassis.router=_k29_choose_route

def route_choice_agent(observation,configuration=None):
    if int(observation['step'])==0:
        _K29_ROUTE_REPORT.update(selected=-1,selection_step=-1,turns=0)
    _K29_ROUTE_REPORT['turns']+=1
    return _K29_ROUTE_PARENT(observation,configuration)
route_choice_agent.telemetry=_K29_ROUTE_REPORT
