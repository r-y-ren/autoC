
# Animal substitutions must be valued on the inherited physical labor contract.
_R9C_PARENT_TARGET=_y_target
_R9C_FLOW=None
_R9C_REPORT={'contracts':0,'unresolved':0,'errors':0}

def _r9c_contract(obs,kind):
    seat=int(obs['player']);step=int(obs['step']);farm=obs['farms'][seat];private=obs['private']
    if len(farm['tiles'])!=10:return None
    if private['shed'].get(kind,0) or any(inv.get(kind,0) for inv in private['inventories']):return None
    native=_IMPL.chassis.players[seat];route=native['route'];tape=_IMPL.chassis.routes[route]
    q=sum(int(o[2]) for o in tape[step].get('market',[]) if o and o[:2]==['BUY_ANIMAL',kind])
    if not 1<=q<=2:return None
    pos=[list(farm['farmer'])]+[list(p) for p in farm['hands']]
    placements=[];events=[];sales={'MILK':[],'WOOL':[]};move={'NORTH':(0,-1),'SOUTH':(0,1),'EAST':(1,0),'WEST':(-1,0)}
    for t in range(step,719):
        raw=_IMPL.chassis.routes[2 if t>=648 else route][t]
        cs=[raw.get('farmer') or ['PASS']]+list(raw.get('hands') or [])
        ev=[]
        for i,c in enumerate(cs[:len(pos)]):
            if not c:continue
            if c[0] in move:
                dx,dy=move[c[0]];pos[i]=[max(0,min(9,pos[i][0]+dx)),max(0,min(9,pos[i][1]+dy))]
            else:
                site=tuple(pos[i])
                if c[:2]==['PLACE',kind] and len(placements)<q and t<step+48:placements.append((t,site))
                if c[0] in ('FEED','CARE','HARVEST','DROP','PLACE'):ev.append((i,list(c),site))
        events.append((t,ev))
        for order in raw.get('market',[]):
            if order==['HIRE']:pos.append(list(_ca_spawn(pos,10)))
            elif order and len(order)>=3 and order[0]=='SELL' and order[1] in sales:sales[order[1]].append(t)
        if t%24==23:pos=pos[:1]
    if len(placements)!=q:return None
    next_sales={}
    for product,ts in sales.items():
        available=set(ts);next_t=None;table={}
        for t in range(718,step-1,-1):
            if t in available:next_t=t
            table[t]=next_t
        next_sales[product]=table
    result={kind:[0.0]*30 for kind in ('COW','SHEEP')}
    for animal in ('COW','SHEEP'):
        product=_R9H_PRODUCT[animal];first=_R9H_FIRST[animal];period=_R9H_PERIOD[animal]
        for placed,site in placements:
            born=placed//24;held=0;pending=0;fed=False;cared=False;missed=0;escaped=False;cargo={}
            def deliver(t,quantity):
                sale=next_sales[product].get(t)
                if sale is not None:result[animal][sale//24]+=quantity/q
            for t,ev in events:
                if t<placed:continue
                for actor,c,p in ev:
                    op=c[0]
                    if p==site and not escaped:
                        if op=='FEED':fed=True
                        elif op=='CARE':cared=True
                        elif op=='HARVEST' and held>0:cargo[actor]=cargo.get(actor,0)+held;held=0
                    if p in ((4,4),(5,4),(4,5),(5,5)) and (op=='DROP' or c[:2]==['PLACE',product]):
                        quantity=cargo.get(actor,0)
                        if op=='PLACE' and len(c)>2:quantity=min(quantity,int(c[2]))
                        if quantity:deliver(t,quantity);cargo[actor]-=quantity
                if t%24==23:
                    day=t//24;missed=0 if fed else missed+1
                    if missed>=2:escaped=True
                    if not escaped:
                        age=day+1-born
                        if age>=first and (age-first)%period==0:
                            held=min(6,held+1+(pending if fed else 0));pending=0
                        if fed and cared:pending+=1
                    fed=False;cared=False
                    for quantity in cargo.values():
                        if quantity:deliver(t+1,quantity)
                    cargo={}
    return result

def _y_target(kind,shops,cfg):
    global _R9C_FLOW
    if _R9H_OBS is None or kind not in _R9H_PRODUCT:return _R9H_OLD_TARGET(kind,shops,cfg)
    _R9C_FLOW=_r9c_contract(_R9H_OBS,kind)
    if _R9C_FLOW is None:
        _R9C_REPORT['unresolved']+=1
        return _R9H_OLD_TARGET(kind,shops,cfg)
    _R9C_REPORT['contracts']+=1
    return _R9C_PARENT_TARGET(kind,shops,cfg)

def _r9h_value(obs,kind,supply,factor,future_demand):
    day=int(obs['step'])//24;shops=obs['town']['unlocked_shops'];value=-_Y_COST[kind]
    for animal,product in _R9H_PRODUCT.items():
        inventory=float(obs['market']['inventory'][product])
        demand=sum(s in _R9H_DEMAND[product] for s in shops)*(2 if animal=='SHEEP' else 1)
        expected_new=(2/8 if animal=='SHEEP' else 3/8)
        for d in range(day,30):
            unknown=max(0,min(8,d//3)-len(shops))
            drain=1+6*(demand+unknown*expected_new*future_demand)
            own=supply['own'][animal][d];rival=supply['rival'][animal][d]*factor
            if animal==kind:own+=_R9C_FLOW[kind][d]
            for z in range(4):
                delta=(own+rival-drain)/4;quote=_r37_market_price(product,inventory+delta/2)
                value+=(own-rival)*quote/4;inventory+=delta if quote>1 else -drain/4
    return value

def round9_herd_contract_agent(observation,configuration=None):
    if int(observation['step'])==0:
        for k in _R9C_REPORT:_R9C_REPORT[k]=0
    return round9_herd_agent(observation,configuration)
round9_herd_contract_agent.telemetry=_R9C_REPORT
