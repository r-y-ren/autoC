
# Just-in-time purchase experiment. Never use opponent private state or future observations.
_JT28_PARENT=herd_repeat_agent
_JT28_CASH_PLAN=_cs_plan
_JT28_PENDING={}
_JT28_REPORT={'deferred':0,'paid_sheep':0,'reverted_to_cow':0,'jt_errors':0,'herd':_HG28_REPORT}
_MP_REPORT=_JT28_REPORT

def _cs_plan(obs,action,k,mode):
    normal=_JT28_CASH_PLAN(obs,action,k,mode)
    if normal is not None or mode!='SHEEP':return normal
    step=int(obs['step']);seat=int(obs['player'])
    if not (72<=step<144) or seat in _JT28_PENDING:return None
    if _HG28_STATE[seat]['allocated']+k>_HG28_CFG['max_swaps']:return None
    animal_orders=[o for o in action.get('market',[]) if o and o[0]=='BUY_ANIMAL']
    if animal_orders!=[['BUY_ANIMAL','COW',k]]:return None
    plan=_HG28_PLAN_BASE(obs,action,k,mode)
    if plan is None:return None
    deadline=min(t for t,i in plan['pickups'])-1
    if not step<deadline<=step+5:return None
    old=_CS_STATE[seat]
    _JT28_PENDING[seat]={'step':step,'deadline':deadline,'count':k,'old_mode':old.get('mode'),'old_plan':old.get('plan'),'plan':plan}
    _JT28_REPORT['deferred']+=1
    return plan

def _jt28_spend(action,farm):
    total=0;hires=int(farm['hires_today'])
    for o in action.get('market',[]):
        if not o:continue
        if o[0] in ('BUY_PRODUCT','BUY_LAND'):return None
        if o[0]=='HIRE':
            a,b=1,1
            for _ in range(hires):a,b=b,a+b
            total+=a;hires+=1
        elif len(o)>2 and o[0]=='BUY_SEED':
            total+=max(0,int(o[2]))*{'WHEAT':10,'CARROT':20,'TOMATO':50,'STRAWBERRY':100,'MELON':80}.get(o[1],0)
        elif len(o)>2 and o[0]=='BUY_ANIMAL':
            total+=max(0,int(o[2]))*{'COW':400,'SHEEP':500,'GOOSE':300}.get(o[1],0)
    return total

def _jt28_restore(seat,pending):
    state=_CS_STATE[seat];state['mode']=pending['old_mode'];state['plan']=pending['old_plan']
    state['decided']=True
    _HG28_STATE[seat]['allocated']-=pending['count']
    _HG28_STATE[seat]['plan']=pending['old_plan']

def just_in_time_herd_agent(observation,configuration=None):
    step=int(observation['step']);seat=int(observation['player'])
    if step==0:
        _JT28_PENDING.pop(seat,None)
        for key in ('deferred','paid_sheep','reverted_to_cow','jt_errors'):_JT28_REPORT[key]=0
    action=_JT28_PARENT(observation,configuration)
    pending=_JT28_PENDING.get(seat)
    if pending is None:return action
    orders=[list(o) for o in action.get('market',[])]
    if step==pending['step']:
        matches=[i for i,o in enumerate(orders) if o==['BUY_ANIMAL','SHEEP',pending['count']]]
        if len(matches)!=1:
            _JT28_REPORT['jt_errors']+=1;return action
        orders.pop(matches[0]);return dict(action,market=orders)
    farm=observation['farms'][seat];spend=_jt28_spend(action,farm)
    available=float(farm['money'])-(spend if spend is not None else float('inf'))
    count=pending['count'];room=len(orders)<10
    if room and available>=500*count+_JT28_RESERVE:
        orders.append(['BUY_ANIMAL','SHEEP',count])
        _JT28_REPORT['paid_sheep']+=count;del _JT28_PENDING[seat]
    elif step>=pending['deadline']:
        _jt28_restore(seat,pending)
        if room and available>=400*count:
            orders.append(['BUY_ANIMAL','COW',count]);_JT28_REPORT['reverted_to_cow']+=count
        else:_JT28_REPORT['jt_errors']+=1
        del _JT28_PENDING[seat]
    return dict(action,market=orders)
just_in_time_herd_agent.telemetry=_JT28_REPORT
agent=just_in_time_herd_agent
kaggle_submission_agent=just_in_time_herd_agent
