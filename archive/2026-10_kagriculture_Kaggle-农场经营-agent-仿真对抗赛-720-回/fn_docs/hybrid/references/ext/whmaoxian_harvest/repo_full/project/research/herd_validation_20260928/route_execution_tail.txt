def _rp28_compatible(actual,reference_key):
    reference=json.loads(reference_key)
    for key in ('farmer','hands','unlocked_quadrants','hires_today'):
        if actual.get(key)!=reference.get(key):return False
    for y,row in enumerate(actual['tiles']):
        for x,tile in enumerate(row):
            old=reference['tiles'][y][x]
            if not isinstance(tile,dict) or not isinstance(old,dict):
                if tile!=old:return False
                continue
            for key in ('kind','animal','crop','placed_day','planted_day','max_lifespan_step'):
                if tile.get(key)!=old.get(key):return False
            for key in ('consecutive_unwatered','consecutive_unfed'):
                if tile.get(key,0)>old.get(key,0):return False
            for key in ('yield_units','pending_care_bonus','fertilized_until_day'):
                if tile.get(key,0)<old.get(key,0):return False
    return True
