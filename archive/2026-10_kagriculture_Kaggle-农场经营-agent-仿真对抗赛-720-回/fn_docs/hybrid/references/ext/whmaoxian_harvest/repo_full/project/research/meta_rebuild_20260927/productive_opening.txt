_PO_DROP_STEP=57 if _PO_HARVEST_DAY==2 else 91
def productive_opening_agent(observation,configuration=None):
    step=int(observation['step']);seat=int(observation['player'])
    if step==0:
        for key in _PO_REPORT:_PO_REPORT[key]=0
    action=_PO_PARENT(observation,configuration)
    farm=observation['farms'][seat];private=observation['private']
    if step in (54,88) and len(private['inventories'])>1:
        _PO_REPORT['harvest_units']=max(_PO_REPORT['harvest_units'],int(private['inventories'][1].get('WHEAT',0)))
    if step==_PO_DROP_STEP:
        tile=farm['tiles'][4][2]
        _PO_REPORT['pasture_restored']=int(isinstance(tile,dict) and tile.get('kind')=='PASTURE')
        if farm['hands'] and tuple(farm['hands'][0])==(4,4) and action.get('hands',[[]])[0]==['DROP']:
            quantity=max(0,int(private['inventories'][1].get('WHEAT',0)))
            if quantity:
                orders=[list(o) for o in action.get('market',[])]
                hit=next((o for o in orders if len(o)>=3 and o[:2]==['SELL','WHEAT']),None)
                if hit is not None:hit[2]=int(hit[2])+quantity
                elif len(orders)<10:orders.append(['SELL','WHEAT',quantity])
                else:quantity=0
                if quantity:
                    action=dict(action,market=orders);_PO_REPORT['delivered_units']+=quantity
    return action
productive_opening_agent.telemetry=_PO_REPORT
