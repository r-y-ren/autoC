
# Forecast ablation: correct the initial care payout and model expected later shops.
_HF29_PARENT_EV=_hd2_ev
_HF29_PARENT_SCHEDULE=_hd2_schedule
_HF29_REPORT={'evaluations':0}

def _hf29_schedule(animal,placed_day,day_from):
    spec=_HD2_SPEC[animal];cap={'GOOSE':4,'COW':6,'SHEEP':6}[animal]
    result={}
    for day in range(max(day_from,placed_day+spec['first']),30):
        age=day-placed_day
        if (age-spec['first'])%spec['interval']:continue
        care_days=spec['first']-1 if age==spec['first'] else spec['interval']
        result[day]=min(float(cap),1.0+care_days*_HD2_CARE)
    return result

def _hd2_ev(option,k,obs,state):
    global _HD2_FUTURE,_hd2_schedule
    if state is not _CS_STATE.get(int(obs['player'])):
        return _HF29_PARENT_EV(option,k,obs,state)
    previous_future=_HD2_FUTURE;previous_schedule=_hd2_schedule
    try:
        _HD2_FUTURE=_HF29_FUTURE
        if _HF29_FIRST_CARE:_hd2_schedule=_hf29_schedule
        _HF29_REPORT['evaluations']+=1
        return _HF29_PARENT_EV(option,k,obs,state)
    finally:
        _HD2_FUTURE=previous_future;_hd2_schedule=previous_schedule
