
def _tier_value(job,core,obs):
    day=int(obs['step'])//24;seat=int(obs['player']);x,y=job['xy']
    tile=obs['farms'][seat]['tiles'][y][x];prices=obs['market']['prices'];value=0.0
    for cmd in job['full_ops']:
        if cmd in core:continue
        op=cmd[0]
        if op=='PLANT':
            spec=_DP_CROPS[cmd[1]];value+=_DP_PLANT_WEIGHT*max(0,spec[2]*prices[cmd[1]]-spec[3])
        elif op=='COLLECT_FERTILIZER':value+=prices['FERTILIZER']
        elif isinstance(tile,dict):
            crop=tile.get('crop');animal=tile.get('animal')
            item=crop or (_DP_ANIMAL[animal][0] if animal else None)
            if op=='HARVEST' and item:value+=tile.get('yield_units',0)*prices[item]
            elif op=='CARE' and animal:
                _,first,interval,_=_DP_ANIMAL[animal];next_production=tile['placed_day']+first
                while next_production<=day+1:next_production+=interval
                if next_production<=29:value+=prices[item]
            elif op=='FERTILIZE' and crop:value+=max(0,1.5*prices[crop]-prices['FERTILIZER'])
            elif op=='WATER' and crop:value+=prices[crop]
    return max(0.0,value)

def _vr_plan(obs):
    jobs,crop=_tier_jobs(obs)
    day,hour=divmod(int(obs['step']),24);farm=obs['farms'][int(obs['player'])]
    starts=[tuple(farm['farmer'])]+[tuple(p) for p in farm['hands']]
    routes=[[] for _ in starts];costs=[0]*len(starts);owners={}
    budget=719-int(obs['step']) if day==29 else 23-hour
    mandatory=set();pending=set(range(len(jobs)))
    for index,job in enumerate(jobs):
        x,y=job['xy'];tile=farm['tiles'][y][x];full=list(job['ops']);core=[]
        job['full_ops']=full;job['full_need']=dict(job['need'])
        if isinstance(tile,dict) and 'animal' in tile:
            if ['FEED'] in full:core.append(['FEED'])
            item,first,interval,cap=_DP_ANIMAL[tile['animal']]
            due=(day+1-tile['placed_day']-first)>=0 and (day+1-tile['placed_day']-first)%interval==0
            if ['HARVEST'] in full and (day==29 or due or tile.get('yield_units',0)>=cap):core.insert(0,['HARVEST'])
        elif isinstance(tile,dict) and tile.get('crop'):
            if ['WATER'] in full and tile.get('consecutive_unwatered',0)>0:core.append(['WATER'])
            urgent=tile['crop'] not in ('STRAWBERRY','TOMATO') or tile.get('max_lifespan_step',-1)>=0 or tile.get('yield_units',0)>=3
            if ['HARVEST'] in full and (urgent or day==29):core.append(['HARVEST'])
        job['extra_value']=_tier_value(job,core,obs)
        if core:
            mandatory.add(index);job['ops']=core;job['need']={'WHEAT':1} if ['FEED'] in core else {}
    while mandatory:
        best=None
        for index in mandatory:
            for actor,route in enumerate(routes):
                for offset in range(len(route)+1):
                    cost=_vr_insert_cost(route,jobs,starts[actor],index,offset,costs[actor])
                    if cost>budget:continue
                    key=(cost-costs[actor],cost,actor,offset,index)
                    if best is None or key<best[0]:best=(key,actor,offset,index,cost)
        if best is None:break
        _,actor,offset,index,cost=best;routes[actor].insert(offset,index);costs[actor]=cost
        owners[index]=actor;mandatory.remove(index);pending.remove(index)
    _VR_REPORT['essential_unassigned']+=len(mandatory)
    upgrades={i for i in owners if jobs[i]['ops']!=jobs[i]['full_ops']}
    available=pending-mandatory
    while available or upgrades:
        best=None
        for index in upgrades:
            job=jobs[index];actor=owners[index];old_ops,old_need=job['ops'],job['need']
            job['ops'],job['need']=job['full_ops'],job['full_need']
            cost=_vr_cost(routes[actor],jobs,starts[actor]);job['ops'],job['need']=old_ops,old_need
            if cost>budget:continue
            gain=job['extra_value'];added=max(1,cost-costs[actor]);key=(gain/added,-cost,-actor,-index)
            if gain>0 and (best is None or key>best[0]):best=(key,index,actor,None,cost)
        for index in available:
            job=jobs[index];gain=job['extra_value']
            if gain<=0:continue
            for actor,route in enumerate(routes):
                for offset in range(len(route)+1):
                    cost=_vr_insert_cost(route,jobs,starts[actor],index,offset,costs[actor])
                    if cost>budget:continue
                    key=(gain/max(1,cost-costs[actor]),-cost,-actor,-index)
                    if best is None or key>best[0]:best=(key,index,actor,offset,cost)
        if best is None:break
        _,index,actor,offset,cost=best;costs[actor]=cost
        if offset is None:
            jobs[index]['ops']=jobs[index]['full_ops'];jobs[index]['need']=jobs[index]['full_need']
            upgrades.remove(index);_VR_REPORT['optional_upgrades']+=1
        else:
            routes[actor].insert(offset,index);owners[index]=actor;available.remove(index);pending.remove(index)
