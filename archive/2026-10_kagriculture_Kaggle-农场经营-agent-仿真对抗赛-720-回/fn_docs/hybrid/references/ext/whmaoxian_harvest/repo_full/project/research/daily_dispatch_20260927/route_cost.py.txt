
def _vr_insert_cost(route,jobs,start,index,offset,current):
    job=jobs[index];position=job['xy']
    before=start if offset==0 else jobs[route[offset-1]]['xy']
    added=_dp_dist(before,position)+len(job['ops'])
    if offset<len(route):
        after=jobs[route[offset]]['xy'];added+=_dp_dist(position,after)-_dp_dist(before,after)
    else:
        added+=_dp_dist(position,_dp_home(position))
        if route:added-=_dp_dist(before,_dp_home(before))
        else:added+=1
    for item in ('WHEAT','FERTILIZER'):
        if job['need'].get(item,0) and not any(jobs[i]['need'].get(item,0) for i in route):added+=1
    return current+added
