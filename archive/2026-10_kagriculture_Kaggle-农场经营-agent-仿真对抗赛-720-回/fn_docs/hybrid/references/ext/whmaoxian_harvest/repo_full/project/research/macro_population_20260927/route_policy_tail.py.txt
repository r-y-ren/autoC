
# Select a complete opening-compatible native macro route, not a splice of tasks.
_RP_ROUTER_PARENT=_IMPL.chassis.router
_RP_REPORT={'decisions':0,'selected_route':-1}
def _rp_router(observation,step,state):
    original=_RP_ROUTER_PARENT(observation,step,state)
    if step==0:
        _RP_REPORT['decisions']=0;_RP_REPORT['selected_route']=-1
    if 144<=step<648 and not state.get('rp_latched'):
        target=_RP_FIXED_ROUTE
        if target is not None:
            if target not in _IMPL.chassis.routes:raise ValueError('Unknown macro route')
            state['route']=target
            _RP_REPORT['selected_route']=target
        else:_RP_REPORT['selected_route']=original
        state['rp_latched']=True;_RP_REPORT['decisions']+=1
    return state.get('route',original)
_IMPL.chassis.router=_rp_router
_RP_PARENT=phase_b_agent
def macro_route_agent(observation,configuration=None):
    return _RP_PARENT(observation,configuration)
macro_route_agent.telemetry=_RP_REPORT
agent=macro_route_agent
kaggle_submission_agent=macro_route_agent
