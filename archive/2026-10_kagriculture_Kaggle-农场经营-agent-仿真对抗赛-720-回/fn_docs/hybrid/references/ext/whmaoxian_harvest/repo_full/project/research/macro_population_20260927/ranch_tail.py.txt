
# Separate southeast ranch: observation-driven chores and actual hired-worker IDs.
_RX_PARENT=phase_b_agent
_RX_GROUPS=(((5,5),(6,5)),((5,6),(6,6)))
_RX_HOME=((4,4),(5,4),(4,5),(5,5))
_RX_STATE={}
_RX_REPORT={'eligible':0,'requests':0,'shortfalls':0,'placed_requests':0,'harvest_requests':0,'errors':0}
def _rx_walk(pos,target):
    if pos[0]!=target[0]:return ['EAST' if target[0]>pos[0] else 'WEST']
    if pos[1]!=target[1]:return ['SOUTH' if target[1]>pos[1] else 'NORTH']
    return None
def _rx_home(pos):return min(_RX_HOME,key=lambda p:abs(p[0]-pos[0])+abs(p[1]-pos[1]))
def _rx_schedule(animal,born,start):
    spec=_HD2_SPEC[animal];cap={'SHEEP':6,'COW':6,'GOOSE':4}[animal];out={}
    for day in range(max(start,born+spec['first']),30):
        age=day-born
        if (age-spec['first'])%spec['interval']==0:
            n=spec['first']-1 if age==spec['first'] else spec['interval']
            out[day]=min(cap,1+.8*n)
    return out

def _rx_forecast(obs,native):
    seat=int(obs['player']);farm=obs['farms'][seat];day=int(obs['step'])//24
    if set(farm['unlocked_quadrants'])!={'NW','NE','SW'}:return None
    if any(farm['tiles'][y][x]!='LOCKED' for group in _RX_GROUPS for x,y in group):return None
    item=_HD2_SPEC[_RX_ANIMAL]['product'];shops=obs['town']['unlocked_shops']
    demand=1+_hd2_daily_shop_demand(item,shops)
    if demand<9:return None
    supply={d:0.0 for d in range(day+1,30)}
    for f in obs['farms']:
        for row in f['tiles']:
            for tile in row:
                if isinstance(tile,dict) and tile.get('animal')==_RX_ANIMAL:
                    for d,n in _rx_schedule(_RX_ANIMAL,int(tile['placed_day']),day+1).items():supply[d]+=n
    added={d:4*n for d,n in _rx_schedule(_RX_ANIMAL,day,day+1).items()}
    inventory=float(obs['market']['inventory'][item]);revenue=0.0
    params=_v44y_params(obs)
    for d in range(day+1,30):
        inventory-=demand;inventory+=supply[d]
        for _ in range(int(added.get(d,0))):
            price=max(1,_r37_market_price(item,int(round(inventory)),params))
            revenue+=price
            if price>1:inventory+=1
    wages=0
    for d in range(day,30):
        plan=_v219_native_day(native,d);n=max(len(a.get('hands',[])) for a in plan)
        wages+=_v219_fib(n)+_v219_fib(n+1)
    days=30-day;feed=4*days*max(20,int(obs['market']['prices']['WHEAT'])+15)
    fertilizer=4*max(0,days-1)*min(25,int(obs['market']['prices']['FERTILIZER']))
    capital=4000+4*_HD2_SPEC[_RX_ANIMAL]['cost']
    net=revenue+fertilizer-capital-feed-wages-2500
    return dict(net=net,revenue=revenue,wages=wages,feed=feed,capital=capital,demand=demand)

def _rx_worker(obs,actor,group,stock):
    seat=int(obs['player']);farm=obs['farms'][seat];pos=([farm['farmer']]+farm['hands'])[actor]
    inv=obs['private']['inventories'][actor];home=_rx_home(pos)
    if int(obs['step'])%24>=22-abs(pos[0]-home[0])-abs(pos[1]-home[1]):
        return _rx_walk(pos,home) or (['DROP'] if sum(inv.values()) else ['PASS'])
    tiles=[farm['tiles'][y][x] for x,y in group]
    missing=sum(not isinstance(t,dict) or not t.get('animal') for t in tiles)
    feed=sum(isinstance(t,dict) and t.get('animal') and not t.get('fed_today') for t in tiles)+missing
    for item,need in ((_RX_ANIMAL,missing),('WHEAT',feed)):
        need=max(0,need-int(inv.get(item,0)))
        if need:
            if tuple(pos) not in _RX_HOME:return _rx_walk(pos,home)
            q=min(need,max(0,int(stock.get(item,0))))
            if q:stock[item]-=q;return ['PICKUP',item,q]
            return ['PASS']
    for target,tile in zip(group,tiles):
        cmd=None
        if tile is None:cmd=['BUILD_COOP' if _RX_ANIMAL=='GOOSE' else 'BUILD_PASTURE']
        elif isinstance(tile,dict) and tile.get('kind')=='WEED':cmd=['DIG']
        elif isinstance(tile,dict) and not tile.get('animal'):cmd=['PLACE',_RX_ANIMAL,1]
        elif isinstance(tile,dict) and tile.get('animal')==_RX_ANIMAL:
            if not tile.get('fed_today'):cmd=['FEED']
            elif not tile.get('cared_today'):cmd=['CARE']
            elif tile.get('fertilizer_available'):cmd=['COLLECT_FERTILIZER']
            elif tile.get('yield_units',0)>0:cmd=['HARVEST']
        if cmd:
            walk=_rx_walk(pos,target)
            if walk:return walk
            if cmd[0]=='PLACE':_RX_REPORT['placed_requests']+=1
            if cmd[0]=='HARVEST':_RX_REPORT['harvest_requests']+=1
            return cmd
    return _rx_walk(pos,home) or (['DROP'] if sum(inv.values()) else ['PASS'])

def _rx_apply(obs,action):
    step=int(obs['step']);day=step//24;hour=step%24;seat=int(obs['player']);farm=obs['farms'][seat]
    state=_RX_STATE.get(seat)
    if state is None or step<=state['last']:
        state=_RX_STATE[seat]=dict(last=-1,day=-1,started=False,workers={},pending=None,credit={})
        for key in _RX_REPORT:_RX_REPORT[key]=0
    state['last']=step
    if state['day']!=day:state.update(day=day,workers={},pending=None,requested_today=False)
    pending=state.get('pending')
    if pending and step>pending['step']:
        for actor,group in pending['workers']:
            if actor<=len(farm['hands']):state['workers'][actor]=group
            else:_RX_REPORT['shortfalls']+=1
        state['pending']=None
    native=_IMPL.chassis.players.get(seat)
    if not native or day<_RX_DAY:return action
    orders=[list(o) for o in action.get('market',[])];plan=_v219_native_day(native,day)
    native_count=max(len(a.get('hands',[])) for a in plan)
    last_hire=max((i for i,a in enumerate(plan) if ['HIRE'] in a.get('market',[])),default=-1)
    can_request=max(2,last_hire+1)<=hour<=7 and len(farm['hands'])>=native_count and not state['requested_today']
    if not state['started'] and day==_RX_DAY and can_request:
        forecast=_rx_forecast(obs,native)
        if forecast and forecast['net']>=_RX_MIN_NET and farm['money']>=forecast['capital']+2000 and len(orders)<=5:
            orders += [['BUY_LAND'],['BUY_ANIMAL',_RX_ANIMAL,4]]
            state['started']=True;state['forecast']=forecast;_RX_REPORT['eligible']+=1
    if not state['started']:return action
    if state.get('commit_step') is None:state['commit_step']=step
    if step>state['commit_step'] and 'SE' not in farm['unlocked_quadrants']:
        state['started']=False;state['failed_land']=True;return action
    if can_request and not state.get('failed_land'):
        count=len(_RX_GROUPS)
        need_cash=sum(_v219_fib(int(farm['hires_today'])+j) for j in range(count))
        if len(orders)+count+1<=10 and farm['money']>=need_cash+500:
            first=len(farm['hands'])+1
            orders.extend([['HIRE'] for _ in range(count)])
            state['pending']={'step':step,'workers':[(first+j,group) for j,group in enumerate(_RX_GROUPS)]}
            state['requested_today']=True;_RX_REPORT['requests']+=count
    commands=[list(action.get('farmer') or ['PASS'])]+[list(c) for c in action.get('hands',[])]
    commands += [['PASS'] for _ in range(len(farm['hands'])+1-len(commands))]
    stock=projected_shed(dict(action,farmer=commands[0],hands=commands[1:]),FarmView(obs))
    for actor,group in state['workers'].items():
        if actor>=len(commands):continue
        cmd=_rx_worker(obs,actor,group,stock);commands[actor]=cmd
        x,y=([farm['farmer']]+farm['hands'])[actor];tile=farm['tiles'][y][x]
        if isinstance(tile,dict):
            if cmd==['HARVEST']:
                item=_HD2_SPEC[_RX_ANIMAL]['product'];n=max(0,int(tile.get('yield_units',0)))
                state['credit'][item]=state['credit'].get(item,0)+n
            if cmd==['COLLECT_FERTILIZER'] and tile.get('fertilizer_available'):
                state['credit']['FERTILIZER']=state['credit'].get('FERTILIZER',0)+1
    result=dict(action,farmer=commands[0],hands=commands[1:],market=orders)
    projected=projected_shed(result,FarmView(obs))
    units=sum(len(group) for group in _RX_GROUPS)
    buying=sum(int(o[2]) for o in orders if len(o)>=3 and o[:2]==['BUY_PRODUCT','WHEAT'])
    feed_gap=max(0,units*2-int(projected.get('WHEAT',0))-buying)
    if feed_gap and len(orders)<10:
        orders.append(['BUY_PRODUCT','WHEAT',feed_gap])
    for item,credit in list(state['credit'].items()):
        selling=sum(max(0,int(o[2])) for o in orders if len(o)>=3 and o[:2]==['SELL',item])
        n=min(credit,max(0,int(projected.get(item,0))-selling))
        if n and len(orders)<10:
            orders.insert(0,['SELL',item,n]);state['credit'][item]-=n
    result['market']=orders
    return result

def ranch_agent(observation,configuration=None):
    action=_RX_PARENT(observation,configuration)
    action=_rx_apply(observation,action)
    seat=int(observation['player']);step=int(observation['step'])
    previous=_OR2_STATE.get(seat,{}).get('prev')
    if previous and previous.get('step')==step:
        stock=projected_shed(action,FarmView(observation))
        for item in _ADV_ITEMS:
            n=sum(max(0,int(o[2])) for o in action.get('market',[]) if len(o)>=3 and o[:2]==['SELL',item])
            previous.setdefault('own',{})[item]=min(max(0,int(stock.get(item,0))),n)
    return action
ranch_agent.telemetry=_RX_REPORT
agent=ranch_agent
kaggle_submission_agent=ranch_agent
