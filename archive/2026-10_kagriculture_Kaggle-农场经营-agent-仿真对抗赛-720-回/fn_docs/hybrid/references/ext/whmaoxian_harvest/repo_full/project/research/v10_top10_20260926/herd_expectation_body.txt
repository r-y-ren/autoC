    scenarios=_he_scenarios(item,day,shops)
    price_cache={}
    def price(inventory):
        inventory=int(inventory)
        if inventory not in price_cache:price_cache[inventory]=max(1,_r37_market_price(item,inventory,params))
        return price_cache[inventory]
    held=[0,0]
    if _HE_INCLUDE_STOCK:
        held[seat]=int(shed.get(item,0))+sum(int(b.get(item,0)) for b in obs['private']['inventories'])
        held[1-seat]=max(0,int(_OR2_STATE.get(seat,{}).get('stock',{}).get(item,0)))

            held[index]+=sum(int(t.get('yield_units',0)) for row in farm['tiles'] for t in row if isinstance(t,dict) and t.get('animal')==animal_of)
    def rollout(arrivals,adding):
        inventory=int(market['inventory'][item]);me=0;other=0;extra=0
        for d in range(day+1,30):
            extra+=arrivals.get(d,0)
            inventory-=int(base_demand+6*extra)
            mine=int(round(ours_existing.get(d,0)+(new.get(d,0) if adding else 0)))

            if d==day+1:mine+=held[seat];theirs+=held[1-seat]
            a,b,inventory=_he_trade(inventory,mine,theirs,price)

        return me,other
    values=[];mean_own=0.0
    for arrivals,probability in scenarios:
        before=rollout(arrivals,False);after=rollout(arrivals,True)

        values.append((gain,probability));mean_own+=probability*(after[0]-before[0])
    mean=sum(v*p for v,p in values)
    variance=sum(p*(v-mean)**2 for v,p in values)
    _HE_REPORT['evaluations']+=1;_HE_REPORT['scenario_paths']+=len(scenarios)
    return mean-_HE_RISK*_he_math.sqrt(max(0.0,variance)),mean_own
