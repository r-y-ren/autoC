
# Observation-driven daily dispatch after a preserved opening.
_DP_PARENT=phase_b_agent
_DP_STATE={}
_DP_REPORT={'dispatch_turns':0,'new_projects':0,'hires_requested':0,'harvests':0,'waterings':0,'feeds':0,'cares':0,'errors':0}
_DP_HOME=((4,4),(5,4),(4,5),(5,5))
_DP_CROPS={'WHEAT':(2,4,6,0,10),'CARROT':(2,3,4,0,20),'TOMATO':(8,8,4,1,50),'STRAWBERRY':(10,10,4,2,100),'MELON':(10,12,6,0,80)}
_DP_ANIMALS={'GOOSE':(4,1,'EGG'),'COW':(8,2,'MILK'),'SHEEP':(6,3,'WOOL')}
def _dp_distance(a,b):return abs(a[0]-b[0])+abs(a[1]-b[1])
def _dp_home(pos):return min(_DP_HOME,key=lambda h:(_dp_distance(pos,h),h))
def _dp_walk(pos,target):
    if pos[0]!=target[0]:return ['EAST' if target[0]>pos[0] else 'WEST']
    if pos[1]!=target[1]:return ['SOUTH' if target[1]>pos[1] else 'NORTH']
    return None

def _dp_next_animal(tile,day):
    first,interval,product=_DP_ANIMALS[tile['animal']]
    born=int(tile['placed_day']);next_day=max(day+1,born+first)
    return next_day+(-(next_day-born-first))%interval

def _dp_crop_flags(tile,day):
    crop=tile['crop'];first,last,capacity,interval,cost=_DP_CROPS[crop]
    age=day-int(tile['planted_day']);watered=bool(tile.get('watered_today'))
    productive=age>=first-1 and interval and (age+1-first)%interval==0 and (age+1-first)//interval<4
    growing=not interval and (last+1)//2<=age<=last and int(tile.get('yield_units',0))<capacity
    needs_water=not watered and (int(tile.get('consecutive_unwatered',0))>=1 or productive or growing)
    return age,capacity,interval,bool(productive),bool(growing),needs_water
