def _k28r_cost(obs,orders):
    farm=obs['farms'][int(obs['player'])];cost=0;h=int(farm['hires_today'])
    land=len(farm['unlocked_quadrants'])-1
    for order in orders:
        if not order:continue
        op=order[0]
        if op=='HIRE':cost+=_v219_fib(h);h+=1
        elif op=='BUY_LAND':
            if land<3:cost+=(1000,2000,4000)[land];land+=1
        elif len(order)>=3 and op in ('BUY_ANIMAL','BUY_SEED','BUY_PRODUCT'):
            prices=({'COW':400,'SHEEP':500,'GOOSE':300} if op=='BUY_ANIMAL' else
                    {'WHEAT':10,'CARROT':20,'TOMATO':50,'STRAWBERRY':100,'MELON':80} if op=='BUY_SEED' else {'WHEAT':25,'FERTILIZER':100})
            cost+=max(0,int(order[2]))*prices.get(order[1],10000)
    return cost
