
# Reactive procurement coupled to actual task progress, not a second tape executor.
_NV_REPORT={'cash_repairs':0,'purchases':0,'errors':0}
def _market(obs,commands,needs,target_hands):
    seat=int(obs['player']);step=int(obs['step']);day=step//24
    farm=obs['farms'][seat];prices=obs['market']['prices']
    stock=_project_stock(obs,commands)
    reserve={i:max(0,int(needs.get('keep:'+i,0))) for i in _ITEMS}
    reserve['WHEAT']=max(reserve.get('WHEAT',0),_NV_FEED_RESERVE)
    reserve['FERTILIZER']=max(reserve.get('FERTILIZER',0),_NV_FERT_RESERVE)
    if day==29:reserve={i:0 for i in _ITEMS}
    planned=_TASK_ACTIONS[min(step,718)].get('market',[])
    sells={};orderkeys=[]
    for o in planned:
        if len(o)>=3 and o[0]=='SELL' and o[1] in _ITEMS:
            item=o[1];available=max(0,int(stock.get(item,0))-reserve.get(item,0))
            sells[item]=min(available,sells.get(item,0)+max(0,int(o[2])))
            if item not in orderkeys:orderkeys.append(item)
    if _NV_LEAD and step%4 and step<718:
        for o in _TASK_ACTIONS[step+1].get('market',[]):
            if len(o)>=3 and o[0]=='SELL' and o[1] not in ('WHEAT','FERTILIZER'):
                item=o[1];available=max(0,int(stock.get(item,0))-reserve.get(item,0))
                sells[item]=min(available,max(sells.get(item,0),int(o[2])))
                if item not in orderkeys:orderkeys.append(item)
    if step>=696:
        for item in _ITEMS:
            sells[item]=max(0,int(stock.get(item,0))-reserve.get(item,0))
            if item not in orderkeys:orderkeys.append(item)
    buying=[];cash_need=0.0
    count=len(farm['hands']);hired=int(farm['hires_today'])
    for n in range(count,target_hands):
        cost=_fib(hired+n-count);buying.append((['HIRE'],cost));cash_need+=cost
    for item in ('WHEAT','FERTILIZER'):
        q=max(0,int(needs.get(item,0)));cost=int(prices[item])+4
        if q:buying.append((['BUY_PRODUCT',item,q],q*cost));cash_need+=q*cost
    owned=len(farm['unlocked_quadrants']);wanted=int(needs.get('land',owned))
    if wanted>owned and owned<4:
        cost=(1000,2000,4000)[owned-1];buying.append((['BUY_LAND'],cost));cash_need+=cost
    for item in ('COW','SHEEP','GOOSE'):
        q=max(0,int(needs.get(item,0)))
        if q:buying.append((['BUY_ANIMAL',item,q],q*_ANIMALS[item]));cash_need+=q*_ANIMALS[item]
    for crop in ('WHEAT','MELON','STRAWBERRY','CARROT','TOMATO'):
        q=max(0,int(needs.get('seed:'+crop,0)))
        if q:buying.append((['BUY_SEED',crop,q],q*_SEEDS[crop]));cash_need+=q*_SEEDS[crop]
    cash=float(farm['money'])
    proceeds=sum(q*max(1,float(prices.get(i,1))*.6) for i,q in sells.items())
    short=max(0,cash_need+_NV_CASH_RESERVE-cash-proceeds)
    for item in sorted(_ITEMS,key=lambda i:-int(prices.get(i,0))):
        if short<=0:break
        available=max(0,int(stock.get(item,0))-reserve.get(item,0)-sells.get(item,0))
        price=max(1,float(prices.get(item,1))*.6)
        q=min(available,int(short/price)+1)
        if q:
            sells[item]=sells.get(item,0)+q;short-=q*price
            if item not in orderkeys:orderkeys.append(item)
            _NV_REPORT['cash_repairs']+=q
    overflow=max(0,sum(stock.values())-sum(sells.values())-90)
    if step%24==23 and overflow:
        for item in sorted(_ITEMS,key=lambda i:-int(prices.get(i,0))):
            q=min(overflow,max(0,int(stock.get(item,0))-reserve.get(item,0)-sells.get(item,0)))
            if q:
                sells[item]=sells.get(item,0)+q;overflow-=q
                if item not in orderkeys:orderkeys.append(item)
    orders=[['SELL',i,sells[i]] for i in orderkeys if sells.get(i,0)>0]
    cash+=sum(q*max(1,float(prices.get(i,1))*.6) for i,q in sells.items())
    for order,cost in buying:
        if len(orders)>=10:break
        if cost>cash and len(order)>=3:
            unit=cost/order[2];quantity=min(order[2],int(cash//unit))
            order=list(order);order[2]=quantity;cost=quantity*unit
            if not quantity:continue
        if cost<=cash:
            orders.append(order);cash-=cost;_NV_REPORT['purchases']+=1
    return orders[:10]

def native_value_agent(observation,configuration=None):
    if int(observation['step'])==0:
        for key in _NV_REPORT:_NV_REPORT[key]=0
    return intent_agent(observation,configuration)
native_value_agent.telemetry=_NV_REPORT
agent=native_value_agent
kaggle_submission_agent=native_value_agent
