
# Preserve the production schedule; sell only to prevent imminent day-end discard.
_K29C_PARENT=phase_b_agent
_K29C_REPORT={'risk_turns':0,'changed_turns':0,'extra_units':0,'booked_units':0,'residual_overflow':0}
_MP_REPORT=_K29C_REPORT

def _k29c_capacity_action(obs,action):
    step=int(obs['step']);seat=int(obs['player']);day,hour=divmod(step,24)
    if hour!=23 or not _K29C_FROM<=step<696:return action
    market=[list(o) for o in action.get('market',[])[:10]]
    if any(o and o[0] in ('BUY_PRODUCT','BUY_ANIMAL') for o in market):return action
    farm=copy.deepcopy(obs['farms'][seat]);private=copy.deepcopy(obs['private'])
    units=[action.get('farmer') or ['PASS']]+list(action.get('hands') or [])
    for actor,command in enumerate(units[:len(farm['hands'])+1]):
        _UNIT_NS['_apply_unit_action'](farm,private,actor,command,len(farm['tiles']),day,24,100)
    deposited=dict(private['shed']);after=dict(deposited)
    for order in market:
        if len(order)>=3 and order[0]=='SELL':
            item=order[1];after[item]=max(0,after.get(item,0)-min(100,max(0,int(order[2]))))
    carried={}
    for bag in private['inventories']:
        for item,n in bag.items():carried[item]=carried.get(item,0)+max(0,int(n))
    excess=max(0,sum(after.values())+sum(carried.values())-100)
    if excess<=0:return action
    _K29C_REPORT['risk_turns']+=1
    animals=sum(isinstance(t,dict) and t.get('animal') is not None for row in farm['tiles'] for t in row)
    reserve={'WHEAT':animals+_K29C_WHEAT_BUFFER,'FERTILIZER':_K29C_FERT_RESERVE}
    prices=obs['market']['prices'];extra={}
    order_items=sorted(prices,key=lambda i:(_K29C_PRICE_SIGN*float(prices[i]),i))
    for item in order_items:
        if excess<=0:break
        stock=max(0,int(after.get(item,0)))
        available=min(stock,max(0,stock+carried.get(item,0)-reserve.get(item,0)))
        quantity=min(excess,available)
        if quantity<=0:continue
        hit=next((o for o in market if len(o)>=3 and o[:2]==['SELL',item] and int(o[2])<100),None)
        if hit is None:
            if len(market)>=10:continue
            quantity=min(100,quantity);market.append(['SELL',item,quantity])
        else:
            quantity=min(quantity,100-max(0,int(hit[2])));hit[2]=max(0,int(hit[2]))+quantity
        extra[item]=extra.get(item,0)+quantity;after[item]-=quantity;excess-=quantity
    _K29C_REPORT['residual_overflow']+=excess
    if not extra:return action
    native=_IMPL.chassis.players.get(seat)
    if _K29C_BOOK and native:
        debts=native['sell_state'].setdefault('r36_debts',{})
        for item,quantity in extra.items():
            remaining=quantity
            for turn in range(step+1,min(719,step+49)):
                planned=sum(min(100,max(0,int(o[2]))) for o in _adv_future(seat,turn) if len(o)>=3 and o[:2]==['SELL',item])
                already=debts.get(turn,{}).get(item,0);take=min(remaining,max(0,planned-already))
                if take:debts.setdefault(turn,{})[item]=already+take;remaining-=take;_K29C_REPORT['booked_units']+=take
                if not remaining:break
    previous=_OR2_STATE.get(seat,{}).get('prev')
    if previous and previous.get('step')==step:
        own=previous.setdefault('own',{})
        for item in extra:
            requested=sum(min(100,max(0,int(o[2]))) for o in market if len(o)>=3 and o[:2]==['SELL',item])
            own[item]=min(deposited.get(item,0),requested)
    _K29C_REPORT['changed_turns']+=1;_K29C_REPORT['extra_units']+=sum(extra.values())
    return dict(action,market=market)

def capacity_guard_agent(observation,configuration=None):
    if int(observation['step'])==0:
        for key in _K29C_REPORT:_K29C_REPORT[key]=0
    return _k29c_capacity_action(observation,_K29C_PARENT(observation,configuration))
capacity_guard_agent.telemetry=_K29C_REPORT
agent=capacity_guard_agent
kaggle_submission_agent=capacity_guard_agent
