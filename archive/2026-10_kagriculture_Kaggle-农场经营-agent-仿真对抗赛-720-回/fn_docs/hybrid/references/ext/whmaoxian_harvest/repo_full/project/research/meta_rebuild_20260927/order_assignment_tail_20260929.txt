
# Exact per-product settlement model; optimize only a permutation of existing sales.
_K29O_PARENT=phase_b_agent
_K29O_REPORT={'eligible':0,'changed':0,'expected_margin_gain':0.0,'risk_rejections':0}
_MP_REPORT=_K29O_REPORT

def _k29o_pair(prices,initial,q,op,r,relation,alpha):
    inventory=initial;own=other=0.0
    def price(index):
        if index not in prices:prices[index]=prices['fn'](index)
        return prices[index]
    if relation==0:
        for index in range(max(q,r)):
            a=price(inventory) if index<q else None
            b=price(inventory-1 if op=='BUY_PRODUCT' else inventory) if index<r else None
            if a is not None:own+=a;inventory+=int(a>1)
            if b is not None:
                other+=b if op=='SELL' else -b
                inventory+=int(b>1) if op=='SELL' else -1
    else:
        for side in ((0,1) if relation<0 else (1,0)):
            for _ in range(q if side==0 else r):
                buying=side==1 and op=='BUY_PRODUCT';value=price(inventory-1 if buying else inventory)
                if side==0:own+=value
                else:other+=-value if buying else value
                inventory+=-1 if buying else int(value>1)
    return own-alpha*other

def _k29o_scenarios(obs,action):
    seat=int(obs['player']);step=int(obs['step']);books=[]
    def pack(orders):
        book={}
        for slot,o in enumerate(orders[:10]):
            if len(o)<3 or o[0] not in ('SELL','BUY_PRODUCT') or o[1] not in PRODUCTS:continue
            if int(o[2])>0 and o[1] not in book:book[o[1]]=(slot,o[0],min(100,int(o[2])))
        return book
    native=pack(_cs_tape(seat,step).get('market',[]));mirror=pack(action.get('market',[]))
    mode=_K29O_CFG['mode']
    if mode=='native':return [(1.0,native)]
    if mode=='mirror':return [(1.0,mirror)]
    stock=_OR2_STATE.get(seat,{}).get('stock',{});quotes=obs['market']['prices']
    quantities={i:min(100,max(0,int(stock.get(i,0)))) for i in PRODUCTS if quotes[i]>1}
    scale=min(1.,100/max(1,sum(quantities.values())))
    quantities={i:int(n*scale) for i,n in quantities.items() if int(n*scale)>0}
    by_price=sorted(quantities,key=lambda i:(-quotes[i],i))
    by_value=sorted(quantities,key=lambda i:(-quotes[i]*quantities[i],i))
    price_book={i:(slot,'SELL',quantities[i]) for slot,i in enumerate(by_price)}
    value_book={i:(slot,'SELL',quantities[i]) for slot,i in enumerate(by_value)}
    return [(.35,native),(.35,mirror),(.15,price_book),(.15,value_book)]

def order_assignment_agent(observation,configuration=None):
    step=int(observation['step']);seat=int(observation['player'])
    if step==0:
        for key in _K29O_REPORT:_K29O_REPORT[key]=0
    action=_K29O_PARENT(observation,configuration);market=action.get('market',[])
    if step<216 or observation['farms'][seat]['money']<3000:return action
    if any(o and o[0]!='SELL' for o in market):return action
    if _K29O_CFG['gate'] and _r37_similarity(observation)<_K29O_CFG['gate']:return action
    stock=projected_shed(action,FarmView(observation));slots=[];items=[];amounts=[]
    for slot,o in enumerate(market):
        if len(o)>=3 and o[0]=='SELL' and int(o[2])>0 and int(stock.get(o[1],0))>0:
            slots.append(slot);items.append(o[1]);amounts.append(min(100,int(o[2]),int(stock[o[1]])))
    n=len(slots)
    if not 2<=n<=6 or len(set(items))!=n:return action
    scenarios=_k29o_scenarios(observation,action);params={k:dict(v) for k,v in _R37_MARKET_PARAMS.items()}
    for k,v in (observation['market'].get('params') or {}).items():
        if k in params:params[k].update(v)
    matrices=[[[0.0]*n for _ in range(n)] for _ in scenarios];cache={}
    for i,item in enumerate(items):
        initial=int(observation['market']['inventory'][item]);prices={'fn':lambda inventory,item=item:_r37_market_price(item,inventory,params)}
        for s,(_,book) in enumerate(scenarios):
            rival_slot,op,r=book.get(item,(0,'SELL',0))
            for j,slot in enumerate(slots):
                relation=(slot>rival_slot)-(slot<rival_slot);key=(item,op,r,relation)
                if key not in cache:cache[key]=_k29o_pair(prices,initial,amounts[i],op,r,relation,_K29O_CFG['alpha'])
                matrices[s][i][j]=cache[key]
    cost=[[sum(weight*matrices[s][i][j] for s,(weight,_) in enumerate(scenarios)) for j in range(n)] for i in range(n)]
    states={0:(0.,())}
    for i in range(n):
        following={}
        for used,(score,path) in states.items():
            for j in range(n):
                if used&(1<<j):continue
                new=used|(1<<j);value=score+cost[i][j]
                if new not in following or value>following[new][0]+1e-9:following[new]=(value,path+(j,))
        states=following
    best,path=states[(1<<n)-1];before=sum(cost[i][i] for i in range(n));gain=best-before
    _K29O_REPORT['eligible']+=1
    if gain<_K29O_CFG['gain']:return action
    worst=min(sum(matrix[i][path[i]]-matrix[i][i] for i in range(n)) for matrix in matrices)
    if worst < -_K29O_CFG['risk']:
        _K29O_REPORT['risk_rejections']+=1;return action
    result=[list(o) for o in market]
    for i,j in enumerate(path):result[slots[j]]=list(market[slots[i]])
    assert sorted(tuple(o) for o in result)==sorted(tuple(o) for o in market)
    _K29O_REPORT['changed']+=1;_K29O_REPORT['expected_margin_gain']+=gain
    return dict(action,market=result)
order_assignment_agent.telemetry=_K29O_REPORT
