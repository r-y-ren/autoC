
# Research-only sequential cow-to-sheep allocation. No future world information.
_H29_PARENT=phase_b_agent
_H29_PLAN_PARENT=_cs_plan
_H29_NATIVE=(_CS_OPTIONS,_CS_SHOP_RULE,_CS_RATIO,_CS_MIN_GAIN,_CS_FROM,_CS_TO)
_H29_ACTIVE=False
_H29_REPORT={'eligible_turns':0,'plan_reopens':0,'cash_veto':0,'capacity_veto':0}
_MP_REPORT=_H29_REPORT

def _cs_plan(obs,action,k,mode):
    if _H29_ACTIVE and mode=='SHEEP':
        state=_CS_STATE.get(int(obs['player']),{})
        occupied=len(state.get('sites',{}))+len(state.get('pending',[]))
        if occupied+k>_H29_CFG['cap']:
            _H29_REPORT['capacity_veto']+=1
            return None
        # Deliberately do not count unexecuted sale proceeds as guaranteed cash.
        cash=float(obs['farms'][int(obs['player'])]['money'])
        if cash<500*k+_H29_CFG['reserve']:
            _H29_REPORT['cash_veto']+=1
            return None
    return _H29_PLAN_PARENT(obs,action,k,mode)

def herd_sequence_agent(observation,configuration=None):
    global _CS_OPTIONS,_CS_SHOP_RULE,_CS_RATIO,_CS_MIN_GAIN,_CS_FROM,_CS_TO,_H29_ACTIVE
    step=int(observation['step']);seat=int(observation['player'])
    if step==0:
        for key in _H29_REPORT:_H29_REPORT[key]=0
    _CS_OPTIONS,_CS_SHOP_RULE,_CS_RATIO,_CS_MIN_GAIN,_CS_FROM,_CS_TO=_H29_NATIVE
    shops=observation['town']['unlocked_shops']
    milk=any(s in ('PIZZA_SHOP','ICE_CREAM_SHOP','SMOOTHIE_SHOP') for s in shops)
    _H29_ACTIVE=('YARN_STORE' in shops and (not _H29_CFG['nomilk'] or not milk)
        and _H29_CFG['start']<=step<_H29_CFG['end'])
    state=_CS_STATE.get(seat)
    if _H29_ACTIVE and (not state or state.get('mode') in (None,'SHEEP')):
        _CS_OPTIONS=('SHEEP',);_CS_SHOP_RULE=''
        _CS_RATIO=_H29_CFG['ratio'];_CS_MIN_GAIN=_H29_CFG['gain']
        _CS_FROM=_H29_CFG['start'];_CS_TO=_H29_CFG['end']
        _H29_REPORT['eligible_turns']+=1
        if state and not state.get('broken') and not state.get('pending'):
            last=max((t for t,i in (state.get('plan') or {}).get('places',{})),default=-1)
            if step>last and len(state.get('sites',{}))<_H29_CFG['cap']:
                if state.get('decided'):_H29_REPORT['plan_reopens']+=1
                state['decided']=False
    result=_H29_PARENT(observation,configuration)
    return result
herd_sequence_agent.telemetry=_H29_REPORT
agent=herd_sequence_agent
kaggle_submission_agent=herd_sequence_agent
